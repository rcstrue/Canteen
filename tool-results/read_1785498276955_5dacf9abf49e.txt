   141→const ITEMS_PER_PAGE = 10;
   142→
   143→type SortField =
   144→  | "name"
   145→  | "category"
   146→  | "unit"

   148→  | "minStock"

   152→type SortDirection = "asc" | "desc";
   153→
   154→// ─── Zod Schema ──────────────────────────────────────────────────────────────
   155→
   156→const ingredientSchema = z.object({
   157→  name: z.string().min(1, "Name is required"),
   158→  unit: z.string().min(1, "Unit is required"),
   159→  category: z.string().min(1, "Category is required"),
   160→  currentStock: z.coerce.number().min(0, "Must be 0 or more"),
   161→  minStock: z.coerce.number().min(0, "Must be 0 or more"),
   162→  lastPurchasePrice: z.coerce.number().min(0, "Must be 0 or more"),
   163→  avgCost: z.coerce.number().min(0, "Must be 0 or more"),
   164→  supplier: z.string().optional().default(""),
   165→});
   166→
   167→type IngredientFormValues = z.infer<typeof ingredientSchema>;
   168→
   169→// ─── Helpers ─────────────────────────────────────────────────────────────────
   170→
   171→function formatCurrency(value: number): string {
   172→  return `₹${value.toFixed(2)}`;
   173→}
   174→
   175→function formatStockWithUnit(quantity: number, unit: string): string {
   176→  return `${quantity} ${unit}`;
   177→}
   178→
   179→function isLowStock(item: Ingredient): boolean {
   180→  return item.currentStock < item.minStock;
   181→}
   182→
   183→// ─── Component ───────────────────────────────────────────────────────────────
   184→
   185→export function StockView() {
   186→  // State
   187→  const [ingredients, setIngredients] = useState<Ingredient[]>([]);
   188→  const [loading, setLoading] = useState(true);
   189→  const [search, setSearch] = useState("");
   190→  const [categoryFilter, setCategoryFilter] = useState("All");
   191→  const [showLowStockOnly, setShowLowStockOnly] = useState(false);
   192→
   193→  // Dialog state
   194→  const [dialogOpen, setDialogOpen] = useState(false);
   195→  const [editingItem, setEditingItem] = useState<Ingredient | null>(null);
   196→  const [deleteItem, setDeleteItem] = useState<Ingredient | null>(null);
   197→  const [detailItem, setDetailItem] = useState<IngredientDetail | null>(null);
   198→  const [detailOpen, setDetailOpen] = useState(false);
   199→
   200→  // Sorting
   201→  const [sortField, setSortField] = useState<SortField>("name");
   202→  const [sortDirection, setSortDirection] = useState<SortDirection>("asc");
   203→
   204→  // Pagination
   205→  const [currentPage, setCurrentPage] = useState(1);
   206→
   207→  // Submitting state
   208→  const [submitting, setSubmitting] = useState(false);
   209→
   210→  // ─── Form ────────────────────────────────────────────────────────────────
   211→
   212→  const form = useForm<IngredientFormValues>({
   213→    resolver: zodResolver(ingredientSchema),
   214→    defaultValues: {
   215→      name: "",
   216→      unit: "",
   217→      category: "",
   218→      currentStock: 0,
   219→      minStock: 0,
   220→      lastPurchasePrice: 0,
   221→      avgCost: 0,
   222→      supplier: "",
   223→    },
   224→  });
   225→
   226→  // ─── Fetch ───────────────────────────────────────────────────────────────
   227→
   228→  const fetchIngredients = useCallback(async () => {
   229→    setLoading(true);
   230→    try {
   231→      const params = new URLSearchParams();
   232→      if (search) params.set("search", search);
   233→      if (categoryFilter !== "All") params.set("category", categoryFilter);
   234→      if (showLowStockOnly) params.set("lowStock", "true");
   235→
   236→      const res = await fetch(`/api/ingredients?${params.toString()}`);
   237→      if (res.ok) {
   238→        const data = await res.json();
   239→        setIngredients(data);
   240→      }
   241→    } catch (error) {
   242→      console.error("Failed to fetch ingredients:", error);
   243→    } finally {
   244→      setLoading(false);
   245→    }
   246→  }, [search, categoryFilter, showLowStockOnly]);
   247→
   248→  useEffect(() => {
   249→    fetchIngredients();
   250→  }, [fetchIngredients]);
   251→
   252→  // Reset page when filters change
   253→  useEffect(() => {
   254→    setCurrentPage(1);
   255→  }, [search, categoryFilter, showLowStockOnly]);
   256→
   257→  // ─── Sorting ─────────────────────────────────────────────────────────────
   258→
   259→  const handleSort = (field: SortField) => {
   260→    if (sortField === field) {
   261→      setSortDirection(sortDirection === "asc" ? "desc" : "asc");
   262→    } else {
   263→      setSortField(field);
   264→      setSortDirection("asc");
   265→    }
   266→  };
   267→
   268→  const sortedIngredients = [...ingredients].sort((a, b) => {
   269→    let aVal: string | number = a[sortField] ?? "";
   270→    let bVal: string | number = b[sortField] ?? "";
   271→
   272→    if (typeof aVal === "string") aVal = aVal.toLowerCase();
   273→    if (typeof bVal === "string") bVal = bVal.toLowerCase();
   274→
   275→    if (aVal < bVal) return sortDirection === "asc" ? -1 : 1;
   276→    if (aVal > bVal) return sortDirection === "asc" ? 1 : -1;
   277→    return 0;
   278→  });
   279→
   280→  // ─── Pagination ──────────────────────────────────────────────────────────
   281→
   282→  const totalPages = Math.max(1, Math.ceil(sortedIngredients.length / ITEMS_PER_PAGE));
   283→  const paginatedIngredients = sortedIngredients.slice(
   284→    (currentPage - 1) * ITEMS_PER_PAGE,
   285→    currentPage * ITEMS_PER_PAGE
   286→  );
   287→
   288→  // ─── CRUD Handlers ──────────────────────────────────────────────────────
   289→
   290→  const openAddDialog = () => {
   291→    setEditingItem(null);
   292→    form.reset({
   293→      name: "",
   294→      unit: "",
   295→      category: "",
   296→      currentStock: 0,
   297→      minStock: 0,
   298→      lastPurchasePrice: 0,
   299→      avgCost: 0,
   300→      supplier: "",
   301→    });
   302→    setDialogOpen(true);
   303→  };
   304→
   305→  const openEditDialog = (item: Ingredient) => {
   306→    setEditingItem(item);
   307→    form.reset({
   308→      name: item.name,
   309→      unit: item.unit,
   310→      category: item.category,
   311→      currentStock: item.currentStock,
   312→      minStock: item.minStock,
   313→      lastPurchasePrice: item.lastPurchasePrice,
   314→      avgCost: item.avgCost,
   315→      supplier: item.supplier ?? "",
   316→    });
   317→    setDialogOpen(true);
   318→  };
   319→
   320→  const onSubmit = async (values: IngredientFormValues) => {
   321→    setSubmitting(true);
   322→    try {
   323→      const body = {
   324→        ...values,
   325→        supplier: values.supplier || null,
   326→      };
   327→
   328→      if (editingItem) {
   329→        const res = await fetch(`/api/ingredients/${editingItem.id}`, {
   330→          method: "PUT",
   331→          headers: { "Content-Type": "application/json" },
   332→          body: JSON.stringify(body),
   333→        });
   334→        if (!res.ok) throw new Error("Failed to update ingredient");
   335→      } else {
   336→        const res = await fetch("/api/ingredients", {
   337→          method: "POST",
   338→          headers: { "Content-Type": "application/json" },
   339→          body: JSON.stringify(body),
   340→        });
   341→        if (!res.ok) throw new Error("Failed to create ingredient");
   342→      }
   343→
   344→      setDialogOpen(false);
   345→      form.reset();
   346→      await fetchIngredients();
   347→    } catch (error) {
   348→      console.error("Failed to save ingredient:", error);
   349→    } finally {
   350→      setSubmitting(false);
   351→    }
   352→  };
   353→
   354→  const handleDelete = async () => {
   355→    if (!deleteItem) return;
   356→    try {
   357→      const res = await fetch(`/api/ingredients/${deleteItem.id}`, {
   358→        method: "DELETE",
   359→      });
   360→      if (!res.ok) throw new Error("Failed to delete ingredient");
   361→      setDeleteItem(null);
   362→      await fetchIngredients();
   363→    } catch (error) {
   364→      console.error("Failed to delete ingredient:", error);
   365→    }
   366→  };
   367→
   368→  const openDetailDialog = async (item: Ingredient) => {
   369→    try {
   370→      const res = await fetch(`/api/ingredients/${item.id}`);
   371→      if (res.ok) {
   372→        const data = await res.json();
   373→        setDetailItem(data);
   374→        setDetailOpen(true);
   375→      }
   376→    } catch (error) {
   377→      console.error("Failed to fetch ingredient detail:", error);
   378→    }
   379→  };
   380→
   381→  // ─── Sortable Header Helper ──────────────────────────────────────────────
   382→
   383→  const SortableHeader = ({
   384→    field,
   385→    children,
   386→  }: {
   387→    field: SortField;
   388→    children: React.ReactNode;
   389→  }) => (
   390→    <Button
   391→      variant="ghost"
   392→      size="sm"
   393→      className="-ml-3 h-8"
   394→      onClick={() => handleSort(field)}
   395→    >
   396→      {children}
   397→      <ArrowUpDown className="ml-1 h-3 w-3" />
   398→      {sortField === field && (
   399→        <span className="ml-1 text-[10px] text-muted-foreground">
   400→          {sortDirection === "asc" ? "↑" : "↓"}
   401→        </span>
   402→      )}
   403→    </Button>
   404→  );
   405→
   406→  // ─── Low stock count ────────────────────────────────────────────────────
   407→
   408→  const lowStockCount = ingredients.filter(isLowStock).length;
   409→
   410→  // ─── Render ──────────────────────────────────────────────────────────────
   411→
   412→  return (
   413→    <div className="space-y-6">
   414→      {/* Header */}
   415→      <div className="flex items-center gap-3">
   416→        <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-orange-100 dark:bg-orange-900/40">
   417→          <Package className="h-5 w-5 text-orange-600 dark:text-orange-400" />
   418→        </div>
   419→        <div>
   420→          <h1 className="text-2xl font-bold tracking-tight">
   421→            Stock / Raw Materials
   422→          </h1>
   423→          <p className="text-muted-foreground">
   424→            Manage raw material inventory, track quantities, and monitor stock
   425→            levels
   426→          </p>
   427→        </div>
   428→      </div>
   429→
   430→      {/* Summary Cards */}
   431→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-3">
   432→        <Card>
   433→          <CardContent className="p-4">
   434→            <div className="flex items-center gap-3">
   435→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-900/40">
   436→                <Package className="h-4 w-4 text-amber-600 dark:text-amber-400" />
   437→              </div>
   438→              <div>
   439→                <p className="text-sm text-muted-foreground">Total Items</p>
   440→                <p className="text-xl font-bold">{ingredients.length}</p>
   441→              </div>
   442→            </div>
   443→          </CardContent>
   444→        </Card>
   445→        <Card>
   446→          <CardContent className="p-4">
   447→            <div className="flex items-center gap-3">
   448→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-red-100 dark:bg-red-900/40">
   449→                <AlertTriangle className="h-4 w-4 text-red-600 dark:text-red-400" />
   450→              </div>
   451→              <div>
   452→                <p className="text-sm text-muted-foreground">Low Stock</p>
   453→                <p className="text-xl font-bold text-red-600 dark:text-red-400">
   454→                  {lowStockCount}
   455→                </p>
   456→              </div>
   457→            </div>
   458→          </CardContent>
   459→        </Card>
   460→        <Card>
   461→          <CardContent className="p-4">
   462→            <div className="flex items-center gap-3">
   463→              <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-green-100 dark:bg-green-900/40">
   464→                <Package className="h-4 w-4 text-green-600 dark:text-green-400" />
   465→              </div>
   466→              <div>
   467→                <p className="text-sm text-muted-foreground">In Stock</p>
   468→                <p className="text-xl font-bold text-green-600 dark:text-green-400">
   469→                  {ingredients.length - lowStockCount}
   470→                </p>
   471→              </div>
   472→            </div>
   473→          </CardContent>
   474→        </Card>
   475→      </div>
   476→
   477→      {/* Filters & Actions */}
   478→      <Card>
   479→        <CardHeader className="pb-4">
   480→          <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   481→            <div>
   482→              <CardTitle>Inventory</CardTitle>
   483→              <CardDescription>
   484→                Add, edit, and track all raw materials used in the canteen
   485→              </CardDescription>
   486→            </div>
   487→            <Button
   488→              onClick={openAddDialog}
   489→              className="bg-orange-600 hover:bg-orange-700 text-white shrink-0"
   490→            >
   491→              <Plus className="h-4 w-4" />
   492→              Add Ingredient
   493→            </Button>
   494→          </div>
   495→        </CardHeader>
   496→        <CardContent className="space-y-4">
   497→          {/* Search & Filter Row */}
   498→          <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   499→            <div className="relative flex-1">
   500→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   501→              <Input
   502→                placeholder="Search by name..."
   503→                value={search}
   504→                onChange={(e) => setSearch(e.target.value)}
   505→                className="pl-9"
   506→              />
   507→              {search && (
   508→                <button
   509→                  onClick={() => setSearch("")}
   510→                  className="absolute right-3 top-1/2 -translate-y-1/2 text-muted-foreground hover:text-foreground"
   511→                >
   512→                  <X className="h-4 w-4" />
   513→                </button>
   514→              )}
   515→            </div>
   516→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   517→              <SelectTrigger className="w-full sm:w-[180px]">
   518→                <SelectValue placeholder="Category" />
   519→              </SelectTrigger>
   520→              <SelectContent>
   521→                {CATEGORIES.map((cat) => (
   522→                  <SelectItem key={cat} value={cat}>
   523→                    {cat}
   524→                  </SelectItem>
   525→                ))}
   526→              </SelectContent>
   527→            </Select>
   528→            <Button
   529→              variant={showLowStockOnly ? "default" : "outline"}
   530→              size="sm"
   531→              onClick={() => setShowLowStockOnly(!showLowStockOnly)}
   532→              className={
   533→                showLowStockOnly
   534→                  ? "bg-red-600 hover:bg-red-700 text-white"
   535→                  : ""
   536→              }
   537→            >
   538→              <AlertTriangle className="h-4 w-4" />
   539→              Low Stock
   540→            </Button>
   541→          </div>
   542→
   543→          {/* Table */}
   544→          {loading ? (
   545→            <div className="flex h-48 items-center justify-center">
   546→              <Loader2 className="h-8 w-8 animate-spin text-muted-foreground" />
   547→            </div>
   548→          ) : sortedIngredients.length === 0 ? (
   549→            <div className="flex h-48 flex-col items-center justify-center rounded-lg border border-dashed border-muted-foreground/25">
   550→              <Package className="mb-2 h-8 w-8 text-muted-foreground" />
   551→              <p className="text-sm text-muted-foreground">
   552→                No ingredients found. Add your first ingredient to get started.
   553→              </p>
   554→            </div>
   555→          ) : (
   556→            <>
   557→              <div className="rounded-md border">
   558→                <Table>
   559→                  <TableHeader>
   560→                    <TableRow>
   561→                      <TableHead>
   562→                        <SortableHeader field="name">Name</SortableHeader>
   563→                      </TableHead>
   564→                      <TableHead>
   565→                        <SortableHeader field="category">
   566→                          Category
   567→                        </SortableHeader>
   568→                      </TableHead>
   569→                      <TableHead>
   570→                        <SortableHeader field="unit">Unit</SortableHeader>
   571→                      </TableHead>
   572→                      <TableHead className="text-right">
   573→                        <SortableHeader field="currentStock">
   574→                          Current Stock
   575→                        </SortableHeader>
   576→                      </TableHead>
   577→                      <TableHead className="text-right">
   578→                        <SortableHeader field="minStock">
   579→                          Min Stock
   580→                        </SortableHeader>
   581→                      </TableHead>
   582→                      <TableHead className="text-right">
   583→                        <SortableHeader field="lastPurchasePrice">
   584→                          Last Price
   585→                        </SortableHeader>
   586→                      </TableHead>
   587→                      <TableHead className="text-right">
   588→                        <SortableHeader field="avgCost">
   589→                          Avg Cost
   590→                        </SortableHeader>
   591→                      </TableHead>
   592→                      <TableHead>
   593→                        <SortableHeader field="supplier">
   594→                          Supplier
   595→                        </SortableHeader>
   596→                      </TableHead>
   597→                      <TableHead>Status</TableHead>
   598→                      <TableHead className="text-right">Actions</TableHead>
   599→                    </TableRow>
   600→                  </TableHeader>
   601→                  <TableBody>
   602→                    {paginatedIngredients.map((item) => {
   603→                      const low = isLowStock(item);
   604→                      return (
   605→                        <TableRow
   606→                          key={item.id}
   607→                          className={`cursor-pointer ${
   608→                            low
   609→                              ? "bg-red-50/50 hover:bg-red-100/50 dark:bg-red-950/20 dark:hover:bg-red-950/30"
   610→                              : ""
   611→                          }`}
   612→                          onClick={() => openDetailDialog(item)}
   613→                        >
   614→                          <TableCell className="font-medium">
   615→                            {item.name}
   616→                          </TableCell>
