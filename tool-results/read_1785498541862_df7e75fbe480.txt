   104→// ─── Helpers ─────────────────────────────────────────────────────────────────
   105→
   106→/** Format number as Indian Rupee with 2 decimal places and Indian comma grouping */
   107→function formatCurrency(value: number): string {
   108→  return new Intl.NumberFormat("en-IN", {
   109→    style: "currency",
   110→    currency: "INR",
   111→    minimumFractionDigits: 2,
   112→    maximumFractionDigits: 2,
   113→  }).format(value);
   114→}
   115→
   116→/** Format number with Indian comma grouping, no decimals */
   117→function formatNumber(value: number): string {
   118→  return new Intl.NumberFormat("en-IN").format(value);
   119→}
   120→
   121→/** Format number with Indian comma grouping, 1 decimal */
   122→function formatNumberDecimal(value: number): string {
   123→  return new Intl.NumberFormat("en-IN", {
   124→    minimumFractionDigits: 1,
   125→    maximumFractionDigits: 1,
   126→  }).format(value);
   127→}
