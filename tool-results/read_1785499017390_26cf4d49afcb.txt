    98→const MEAL_TYPES = ["All", "Breakfast", "Lunch", "Dinner", "Snack"] as const;
    99→
   100→function getMealTypeBadgeClass(mealType: string): string {
   101→  switch (mealType) {
   102→    case "Breakfast":
   103→      return "bg-yellow-100 text-yellow-800 border-yellow-300 dark:bg-yellow-900/30 dark:text-yellow-400 dark:border-yellow-700";
   104→    case "Lunch":
   105→      return "bg-orange-100 text-orange-800 border-orange-300 dark:bg-orange-900/30 dark:text-orange-400 dark:border-orange-700";
   106→    case "Dinner":
   107→      return "bg-green-100 text-green-800 border-green-300 dark:bg-green-900/30 dark:text-green-400 dark:border-green-700";
   108→    case "Snack":
   109→      return "bg-purple-100 text-purple-800 border-purple-300 dark:bg-purple-900/30 dark:text-purple-400 dark:border-purple-700";
   110→    default:
   111→      return "bg-gray-100 text-gray-800 border-gray-300 dark:bg-gray-900/30 dark:text-gray-400 dark:border-gray-700";
   112→  }
   113→}
   114→
   115→function formatRupee(amount: number): string {
   116→  return "₹" + amount.toLocaleString("en-IN", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
   117→}
   118→
   119→function calcTotalIngredientCost(ingredients: RecipeIngredient[]): number {
   120→  return ingredients.reduce((sum, ri) => sum + ri.quantity * ri.ingredient.avgCost, 0);
   121→}
   122→
   123→function calcCostPerMeal(ingredients: RecipeIngredient[], baseServings: number): number {
   124→  if (baseServings <= 0) return 0;
   125→  return calcTotalIngredientCost(ingredients) / baseServings;
   126→}
   127→
   128→function calcCostFor600(ingredients: RecipeIngredient[], baseServings: number): number {
   129→  return calcCostPerMeal(ingredients, baseServings) * 600;
   130→}
