   142→
   143→const formatCurrency = (amount: number) =>
   144→  new Intl.NumberFormat('en-IN', {
   145→    style: 'currency',
   146→    currency: 'INR',
   147→    minimumFractionDigits: 2,
   148→    maximumFractionDigits: 2,
   149→  }).format(amount);
   150→
   151→const formatDate = (dateStr: string) => {
   152→  const d = new Date(dateStr);
   153→  return format(d, 'dd/MM/yyyy');
   154→};
   155→
   156→const formatDateISO = (date: Date) => format(date, 'yyyy-MM-dd');
   157→
   158→const getToday = () => {
   159→  const d = new Date();
   160→  d.setHours(0, 0, 0, 0);
   161→  return d;
   162→};
   163→
   164→const mealTypeBadgeVariant = (type: string) => {
   165→  switch (type) {
   166→    case 'Breakfast':
   167→      return 'bg-yellow-100 text-yellow-800 dark:bg-yellow-900/30 dark:text-yellow-400 hover:bg-yellow-100';
   168→    case 'Lunch':
   169→      return 'bg-orange-100 text-orange-800 dark:bg-orange-900/30 dark:text-orange-400 hover:bg-orange-100';
   170→    case 'Dinner':
   171→      return 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-400 hover:bg-green-100';
   172→    default:
   173→      return 'bg-gray-100 text-gray-800 dark:bg-gray-900/30 dark:text-gray-400 hover:bg-gray-100';
   174→  }
   175→};
   176→
   177→const movementTypeBadge = (type: string) => {
   178→  switch (type) {
   179→    case 'PURCHASE':
   180→      return (
   181→        <Badge className="bg-blue-100 text-blue-800 dark:bg-blue-900/30 dark:text-blue-400 hover:bg-blue-100">
   182→          <ArrowUp className="mr-1 h-3 w-3" />
   183→          Purchase
   184→        </Badge>
   185→      );
   186→    case 'CONSUMPTION':
   187→      return (
   188→        <Badge className="bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-400 hover:bg-amber-100">
   189→          <ArrowDown className="mr-1 h-3 w-3" />
   190→          Consumption
   191→        </Badge>
   192→      );
   193→    case 'WASTAGE':
   194→      return (
   195→        <Badge className="bg-red-100 text-red-800 dark:bg-red-900/30 dark:text-red-400 hover:bg-red-100">
   196→          <AlertTriangle className="mr-1 h-3 w-3" />
   197→          Wastage
   198→        </Badge>
   199→      );
   200→    case 'ADJUSTMENT':
   201→      return (
   202→        <Badge className="bg-purple-100 text-purple-800 dark:bg-purple-900/30 dark:text-purple-400 hover:bg-purple-100">
   203→          <RefreshCw className="mr-1 h-3 w-3" />
   204→          Adjustment
   205→        </Badge>
   206→      );
   207→    default:
   208→      return <Badge variant="secondary">{type}</Badge>;
   209→  }

   212→// ─── Component ───────────────────────────────────────────────────────────────
   213→
   214→export function DailyEntryView() {
   215→  // ── Shared state ─────────────────────────────────────────────────────────
   216→  const [activeTab, setActiveTab] = useState('meals');
   217→
   218→  // ── Meals Served state ───────────────────────────────────────────────────
   219→  const [selectedDate, setSelectedDate] = useState<Date>(getToday());
   220→  const [mealTypeFilter, setMealTypeFilter] = useState<string>('all');
   221→  const [meals, setMeals] = useState<DailyMeal[]>([]);
   222→  const [mealsLoading, setMealsLoading] = useState(true);
   223→  const [showAddMealDialog, setShowAddMealDialog] = useState(false);
   224→  const [mealForm, setMealForm] = useState({
   225→    date: formatDateISO(getToday()),
   226→    mealType: '',
   227→    recipeId: '',
   228→    mealsServed: '',
   229→    notes: '',
   230→  });
   231→  const [recipes, setRecipes] = useState<Recipe[]>([]);
   232→  const [selectedRecipe, setSelectedRecipe] = useState<Recipe | null>(null);
   233→  const [mealSubmitting, setMealSubmitting] = useState(false);
   234→  const [showMealConfirmation, setShowMealConfirmation] = useState(false);
   235→  const [mealDeduction, setMealDeduction] = useState<
   236→    { name: string; unit: string; consumed: number; currentStock: number }[]
   237→  >([]);
   238→
   239→  // ── Stock Adjustment state ───────────────────────────────────────────────
   240→  const [adjustments, setAdjustments] = useState<StockMovement[]>([]);
   241→  const [adjustmentsLoading, setAdjustmentsLoading] = useState(true);
   242→  const [showAddAdjustmentDialog, setShowAddAdjustmentDialog] = useState(false);
   243→  const [ingredients, setIngredients] = useState<Ingredient[]>([]);
   244→  const [adjustmentForm, setAdjustmentForm] = useState({
   245→    ingredientId: '',
   246→    type: '',
   247→    quantity: '',
   248→    unitPrice: '',
   249→    date: formatDateISO(getToday()),
   250→    notes: '',
   251→  });
   252→  const [adjustmentSubmitting, setAdjustmentSubmitting] = useState(false);
   253→
   254→  // ── Calendar popovers ────────────────────────────────────────────────────
   255→  const [mealCalendarOpen, setMealCalendarOpen] = useState(false);
   256→  const [mainCalendarOpen, setMainCalendarOpen] = useState(false);
   257→
   258→  // ── Fetch Meals ──────────────────────────────────────────────────────────
   259→  const fetchMeals = useCallback(async () => {
   260→    setMealsLoading(true);
   261→    try {
   262→      const params = new URLSearchParams();
   263→      params.set('date', formatDateISO(selectedDate));
   264→      if (mealTypeFilter && mealTypeFilter !== 'all') {
   265→        params.set('mealType', mealTypeFilter);
   266→      }
   267→      const res = await fetch(`/api/daily-meals?${params.toString()}`);
   268→      if (res.ok) {
   269→        const json = await res.json();
   270→        setMeals(json.data || []);
   271→      }
   272→    } catch (err) {
   273→      console.error('Error fetching meals:', err);
   274→    } finally {
   275→      setMealsLoading(false);
   276→    }
   277→  }, [selectedDate, mealTypeFilter]);
   278→
   279→  // ── Fetch Recipes ────────────────────────────────────────────────────────
   280→  const fetchRecipes = useCallback(async () => {
   281→    try {
   282→      const res = await fetch('/api/recipes');
   283→      if (res.ok) {
   284→        const data = await res.json();
   285→        setRecipes(data);
   286→      }
   287→    } catch (err) {
   288→      console.error('Error fetching recipes:', err);
   289→    }
   290→  }, []);
   291→
   292→  // ── Fetch Adjustments ────────────────────────────────────────────────────
   293→  const fetchAdjustments = useCallback(async () => {
   294→    setAdjustmentsLoading(true);
   295→    try {
   296→      const params = new URLSearchParams();
   297→      params.set('type', 'CONSUMPTION,WASTAGE,ADJUSTMENT');
   298→      const today = formatDateISO(getToday());
   299→      params.set('startDate', today);
   300→      params.set('endDate', today);
   301→      params.set('limit', '50');
   302→      const res = await fetch(`/api/stock-movements?${params.toString()}`);
   303→      if (res.ok) {
   304→        const json = await res.json();
   305→        setAdjustments(json.data || []);
   306→      }
   307→    } catch (err) {
   308→      console.error('Error fetching adjustments:', err);
   309→    } finally {
   310→      setAdjustmentsLoading(false);
   311→    }
   312→  }, []);
   313→
   314→  // ── Fetch Ingredients ────────────────────────────────────────────────────
   315→  const fetchIngredients = useCallback(async () => {
   316→    try {
   317→      const res = await fetch('/api/ingredients');
   318→      if (res.ok) {
   319→        const data = await res.json();
   320→        setIngredients(data);
   321→      }
   322→    } catch (err) {
   323→      console.error('Error fetching ingredients:', err);
   324→    }
   325→  }, []);
   326→
   327→  // ── Effects ──────────────────────────────────────────────────────────────
   328→  useEffect(() => {
   329→    fetchMeals();
   330→  }, [fetchMeals]);
   331→
   332→  useEffect(() => {
   333→    if (activeTab === 'adjustments') {
   334→      fetchAdjustments();
   335→    }
   336→  }, [activeTab, fetchAdjustments]);
   337→
   338→  useEffect(() => {
   339→    fetchRecipes();
   340→    fetchIngredients();
   341→  }, [fetchRecipes, fetchIngredients]);
   342→
   343→  // ── Calculate stock deduction preview ────────────────────────────────────
   344→  const calculateDeduction = () => {
   345→    if (!selectedRecipe || !mealForm.mealsServed) return [];
   346→    const ratio = parseInt(mealForm.mealsServed) / selectedRecipe.baseServings;
   347→    return selectedRecipe.ingredients.map((ri) => ({
   348→      name: ri.ingredient.name,
   349→      unit: ri.ingredient.unit,
   350→      consumed: ratio * ri.quantity,
   351→      currentStock: ri.ingredient.currentStock,
   352→    }));
   353→  };
   354→
   355→  // ── Handle Add Meal ─────────────────────────────────────────────────────
   356→  const handleAddMeal = () => {
   357→    if (!mealForm.mealType || !mealForm.recipeId || !mealForm.mealsServed) return;
   358→    const deduction = calculateDeduction();
   359→    setMealDeduction(deduction);
   360→    setShowMealConfirmation(true);
   361→  };
   362→
   363→  const confirmAddMeal = async () => {
   364→    setMealSubmitting(true);
   365→    try {
   366→      const res = await fetch('/api/daily-meals', {
   367→        method: 'POST',
   368→        headers: { 'Content-Type': 'application/json' },
   369→        body: JSON.stringify({
   370→          date: mealForm.date,
   371→          mealType: mealForm.mealType,
   372→          mealsServed: parseInt(mealForm.mealsServed),
   373→          recipeId: mealForm.recipeId,
   374→          notes: mealForm.notes || undefined,
   375→        }),
   376→      });
   377→      if (res.ok) {
   378→        setShowMealConfirmation(false);
   379→        setShowAddMealDialog(false);
   380→        resetMealForm();
   381→        fetchMeals();
   382→      } else {
   383→        const err = await res.json();
   384→        alert(err.error || 'Failed to record meal');
   385→      }
   386→    } catch (err) {
   387→      console.error('Error creating meal:', err);
   388→      alert('Failed to record meal');
   389→    } finally {
   390→      setMealSubmitting(false);
   391→    }
   392→  };
   393→
   394→  const resetMealForm = () => {
   395→    setMealForm({
   396→      date: formatDateISO(getToday()),
   397→      mealType: '',
   398→      recipeId: '',
   399→      mealsServed: '',
   400→      notes: '',
   401→    });
   402→    setSelectedRecipe(null);
   403→  };
   404→
   405→  // ── Handle Add Adjustment ────────────────────────────────────────────────
   406→  const handleAddAdjustment = async () => {
   407→    if (
   408→      !adjustmentForm.ingredientId ||
   409→      !adjustmentForm.type ||
   410→      !adjustmentForm.quantity
   411→    )
   412→      return;
   413→
   414→    setAdjustmentSubmitting(true);
   415→    try {
   416→      const res = await fetch('/api/stock-movements', {
   417→        method: 'POST',
   418→        headers: { 'Content-Type': 'application/json' },
   419→        body: JSON.stringify({
   420→          ingredientId: adjustmentForm.ingredientId,
   421→          type: adjustmentForm.type,
   422→          quantity: parseFloat(adjustmentForm.quantity),
   423→          unitPrice: parseFloat(adjustmentForm.unitPrice) || 0,
   424→          totalAmount:
   425→            (parseFloat(adjustmentForm.unitPrice) || 0) *
   426→            parseFloat(adjustmentForm.quantity),
   427→          date: adjustmentForm.date,
   428→          notes: adjustmentForm.notes || undefined,
   429→        }),
   430→      });
   431→      if (res.ok) {
   432→        setShowAddAdjustmentDialog(false);
   433→        resetAdjustmentForm();
   434→        fetchAdjustments();
   435→      } else {
   436→        const err = await res.json();
   437→        alert(err.error || 'Failed to add adjustment');
   438→      }
   439→    } catch (err) {
   440→      console.error('Error creating adjustment:', err);
   441→      alert('Failed to add adjustment');
   442→    } finally {
   443→      setAdjustmentSubmitting(false);
   444→    }
   445→  };
   446→
   447→  const resetAdjustmentForm = () => {
   448→    setAdjustmentForm({
   449→      ingredientId: '',
   450→      type: '',
   451→      quantity: '',
   452→      unitPrice: '',
   453→      date: formatDateISO(getToday()),
   454→      notes: '',
   455→    });
   456→  };
   457→
   458→  // ── Selected ingredient for auto-fill ────────────────────────────────────
   459→  const selectedIngredient = ingredients.find(
   460→    (i) => i.id === adjustmentForm.ingredientId
   461→  );
   462→
   463→  // ── Totals ───────────────────────────────────────────────────────────────
   464→  const totalMealsServed = meals.reduce((sum, m) => sum + m.mealsServed, 0);
   465→
   466→  // ── Render ───────────────────────────────────────────────────────────────
   467→  return (
   468→    <div className="space-y-6">
   469→      {/* Header */}
   470→      <div className="flex items-center gap-3">
   471→        <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-900/30">
   472→          <ClipboardList className="h-5 w-5 text-amber-600 dark:text-amber-400" />
   473→        </div>
   474→        <div>
   475→          <h1 className="text-2xl font-bold tracking-tight">Daily Entry</h1>
   476→          <p className="text-muted-foreground">
   477→            Record meals served and manage stock adjustments
   478→          </p>
   479→        </div>
   480→      </div>
   481→
   482→      {/* Tabs */}
   483→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   484→        <TabsList>
   485→          <TabsTrigger value="meals" className="gap-1.5">
   486→            <UtensilsCrossed className="h-4 w-4" />
   487→            Meals Served
   488→          </TabsTrigger>
   489→          <TabsTrigger value="adjustments" className="gap-1.5">
   490→            <Package className="h-4 w-4" />
   491→            Stock Adjustment
   492→          </TabsTrigger>
   493→        </TabsList>
   494→
   495→        {/* ─── Meals Served Tab ─────────────────────────────────────────── */}
   496→        <TabsContent value="meals" className="space-y-4">
   497→          {/* Filters */}
   498→          <Card>
   499→            <CardContent className="p-4">
   500→              <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   501→                <div className="flex flex-col gap-3 sm:flex-row sm:items-center">
   502→                  {/* Date Picker */}
   503→                  <div className="flex items-center gap-2">
   504→                    <Label className="text-sm font-medium whitespace-nowrap">
   505→                      Date:
   506→                    </Label>
   507→                    <Popover
   508→                      open={mainCalendarOpen}
   509→                      onOpenChange={setMainCalendarOpen}
   510→                    >
   511→                      <PopoverTrigger asChild>
   512→                        <Button
   513→                          variant="outline"
   514→                          className="w-[160px] justify-start text-left font-normal"
   515→                        >
   516→                          <CalendarIcon className="mr-2 h-4 w-4" />
   517→                          {format(selectedDate, 'dd/MM/yyyy')}
   518→                        </Button>
   519→                      </PopoverTrigger>
   520→                      <PopoverContent className="w-auto p-0" align="start">
   521→                        <Calendar
   522→                          mode="single"
   523→                          selected={selectedDate}
   524→                          onSelect={(date) => {
   525→                            if (date) {
   526→                              setSelectedDate(date);
   527→                              setMainCalendarOpen(false);
   528→                            }
   529→                          }}
   530→                          initialFocus
   531→                        />
   532→                      </PopoverContent>
   533→                    </Popover>
   534→                  </div>
   535→
   536→                  {/* Meal Type Filter */}
   537→                  <div className="flex items-center gap-2">
   538→                    <Label className="text-sm font-medium whitespace-nowrap">
   539→                      Meal:
   540→                    </Label>
   541→                    <Select
   542→                      value={mealTypeFilter}
   543→                      onValueChange={setMealTypeFilter}
   544→                    >
   545→                      <SelectTrigger className="w-[140px]">
   546→                        <SelectValue />
   547→                      </SelectTrigger>
   548→                      <SelectContent>
   549→                        <SelectItem value="all">All Meals</SelectItem>
   550→                        <SelectItem value="Breakfast">Breakfast</SelectItem>
   551→                        <SelectItem value="Lunch">Lunch</SelectItem>
   552→                        <SelectItem value="Dinner">Dinner</SelectItem>
   553→                      </SelectContent>
   554→                    </Select>
   555→                  </div>
   556→                </div>
   557→
   558→                <Button
   559→                  onClick={() => setShowAddMealDialog(true)}
   560→                  className="bg-amber-600 hover:bg-amber-700 text-white"
   561→                >
   562→                  <Plus className="mr-2 h-4 w-4" />
   563→                  Record Meal
   564→                </Button>
   565→              </div>
   566→            </CardContent>
   567→          </Card>
   568→
   569→          {/* Meals Table */}
   570→          <Card>
   571→            <CardHeader className="pb-3">
   572→              <CardTitle className="text-lg">
   573→                Meals for {format(selectedDate, 'dd/MM/yyyy')}
   574→              </CardTitle>
   575→              <CardDescription>
   576→                Showing {meals.length} meal{meals.length !== 1 ? 's' : ''}{' '}
   577→                recorded
   578→              </CardDescription>
   579→            </CardHeader>
   580→            <CardContent>
   581→              {mealsLoading ? (
   582→                <div className="space-y-3">
   583→                  {[1, 2, 3].map((i) => (
   584→                    <div key={i} className="flex items-center gap-4">
   585→                      <Skeleton className="h-5 w-24" />
   586→                      <Skeleton className="h-5 w-20" />
   587→                      <Skeleton className="h-5 w-32" />
   588→                      <Skeleton className="h-5 w-16" />
   589→                      <Skeleton className="h-5 w-24" />
   590→                    </div>
   591→                  ))}
   592→                </div>
   593→              ) : meals.length === 0 ? (
   594→                <div className="flex flex-col items-center justify-center py-12 text-center">
   595→                  <UtensilsCrossed className="h-12 w-12 text-muted-foreground/40 mb-3" />
   596→                  <p className="text-muted-foreground text-sm">
   597→                    No meals recorded for this date
   598→                  </p>
   599→                  <p className="text-muted-foreground/60 text-xs mt-1">
   600→                    Click &quot;Record Meal&quot; to add a meal entry
   601→                  </p>
   602→                </div>
   603→              ) : (
   604→                <div className="overflow-x-auto">
   605→                  <Table>
   606→                    <TableHeader>
   607→                      <TableRow>
   608→                        <TableHead>Date</TableHead>
   609→                        <TableHead>Meal Type</TableHead>
   610→                        <TableHead>Recipe</TableHead>
   611→                        <TableHead className="text-right">
   612→                          Meals Served
