    63→
    64→interface CostReport {
    65→  period: { start: string; end: string; type: string };
    66→  foodCost: {
    67→    total: number;
    68→    costPerMeal: number;
    69→    categoryBreakdown: { category: string; totalCost: number; totalQuantity: number }[];
    70→    dailyTrend: { date: string; cost: number }[];
    71→  };

    78→interface ConsumptionReport {

    81→    totalConsumption: number;

    83→    wastagePercentage: number;

    86→    ingredient: { id: string; name: string; unit: string; category: string; avgCost: number };
    87→    consumptionQty: number;
    88→    consumptionCost: number;
    89→    wastageQty: number;
    90→    wastageCost: number;
    91→    totalQty: number;
    92→    totalCost: number;
    93→  }[];
    94→  categoryBreakdown: { category: string; totalQty: number; totalCost: number; wastagePercent: number }[];
    95→  dailyTrend: { date: string; consumption: number; wastage: number }[];
    96→}
    97→
    98→interface VarianceReport {
    99→  period: { start: string; end: string; type: string };
   100→  summary: {
   101→    totalTheoreticalCost: number;
   102→    totalActualCost: number;
   103→    totalVarianceCost: number;
   104→    variancePercentage: number;
   105→    totalWastage: number;
   106→    criticalCount: number;
   107→    warningCount: number;
   108→  };
   109→  varianceByIngredient: {
   110→    ingredient: { id: string; name: string; unit: string; category: string; avgCost: number };
   111→    theoreticalQty: number;
   112→    actualQty: number;
   113→    varianceQty: number;
   114→    variancePercent: number;
   115→    theoreticalCost: number;
   116→    actualCost: number;
   117→    varianceCost: number;
   118→    wastageQty: number;
   119→    status: 'normal' | 'warning' | 'critical';
   120→  }[];
   121→  criticalItems: {
   122→    ingredient: { id: string; name: string; unit: string; category: string; avgCost: number };
   123→    theoreticalQty: number;
   124→    actualQty: number;
   125→    varianceQty: number;
   126→    variancePercent: number;
   127→    status: string;
   128→  }[];
   129→  warningItems: {
   130→    ingredient: { id: string; name: string; unit: string; category: string; avgCost: number };
   131→    theoreticalQty: number;
   132→    actualQty: number;
   133→    varianceQty: number;
   134→    variancePercent: number;
   135→    status: string;
   136→  }[];
   137→}

   156→const PIE_COLORS = [
   157→  '#f59e0b', '#d97706', '#b45309', '#92400e',
   158→  '#fbbf24', '#f59e0b', '#ea580c', '#c2410c',
   159→  '#fb923c', '#fdba74',

   162→const CATEGORY_COLORS: Record<string, string> = {
   163→  Grains: '#f59e0b',
   164→  Pulses: '#d97706',
   165→  Vegetables: '#22c55e',
   166→  Oil: '#eab308',
   167→  Spices: '#ef4444',
   168→  Dairy: '#3b82f6',
   169→  Meat: '#f43f5e',
   170→  Other: '#6b7280',
   171→};
   172→
   173→function getCategoryColor(category: string): string {
   174→  return CATEGORY_COLORS[category] || '#6b7280';
   175→}
   176→
   177→function formatDate(dateStr: string): string {
   178→  const d = new Date(dateStr);
   179→  return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' });
   180→}
   181→
   182→function formatShortDate(dateStr: string): string {
   183→  const d = new Date(dateStr);
   184→  return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short' });
   185→}

   211→// ─── Component ────────────────────────────────────────────────────────────────
   212→
   213→export function ReportsView() {
   214→  const [period, setPeriod] = useState<PeriodType>('month');
   215→  const [startDate, setStartDate] = useState('');
   216→  const [endDate, setEndDate] = useState('');
   217→  const [activeTab, setActiveTab] = useState('cost');
   218→
   219→  const [costData, setCostData] = useState<CostReport | null>(null);
   220→  const [consumptionData, setConsumptionData] = useState<ConsumptionReport | null>(null);
   221→  const [varianceData, setVarianceData] = useState<VarianceReport | null>(null);
   222→
   223→  const [costLoading, setCostLoading] = useState(false);
   224→  const [consumptionLoading, setConsumptionLoading] = useState(false);
   225→  const [varianceLoading, setVarianceLoading] = useState(false);
   226→
   227→  const [costError, setCostError] = useState('');
   228→  const [consumptionError, setConsumptionError] = useState('');
   229→  const [varianceError, setVarianceError] = useState('');
   230→
   231→  const buildQuery = useCallback(() => {
   232→    const params = new URLSearchParams();
   233→    if (startDate && endDate) {
   234→      params.set('startDate', startDate);
   235→      params.set('endDate', endDate);
   236→      params.set('period', period);
   237→    } else {
   238→      params.set('period', period);
   239→    }
   240→    return params.toString();
   241→  }, [period, startDate, endDate]);
   242→
   243→  const fetchCostReport = useCallback(async () => {
   244→    setCostLoading(true);
   245→    setCostError('');
   246→    try {
   247→      const res = await fetch(`/api/reports/cost?${buildQuery()}`);
   248→      if (!res.ok) throw new Error('Failed to fetch cost report');
   249→      const data = await res.json();
   250→      setCostData(data);
   251→    } catch {
   252→      setCostError('Failed to load cost report');
   253→    } finally {
   254→      setCostLoading(false);
   255→    }
   256→  }, [buildQuery]);
   257→
   258→  const fetchConsumptionReport = useCallback(async () => {
   259→    setConsumptionLoading(true);
   260→    setConsumptionError('');
   261→    try {
   262→      const res = await fetch(`/api/reports/consumption?${buildQuery()}`);
   263→      if (!res.ok) throw new Error('Failed to fetch consumption report');
   264→      const data = await res.json();
   265→      setConsumptionData(data);
   266→    } catch {
   267→      setConsumptionError('Failed to load consumption report');
   268→    } finally {
   269→      setConsumptionLoading(false);
   270→    }
   271→  }, [buildQuery]);
   272→
   273→  const fetchVarianceReport = useCallback(async () => {
   274→    setVarianceLoading(true);
   275→    setVarianceError('');
   276→    try {
   277→      const res = await fetch(`/api/reports/variance?${buildQuery()}`);
   278→      if (!res.ok) throw new Error('Failed to fetch variance report');
   279→      const data = await res.json();
   280→      setVarianceData(data);
   281→    } catch {
   282→      setVarianceError('Failed to load variance report');
   283→    } finally {
   284→      setVarianceLoading(false);
   285→    }
   286→  }, [buildQuery]);
   287→
   288→  // Fetch all reports when period changes
   289→  useEffect(() => {
   290→    fetchCostReport();
   291→    fetchConsumptionReport();
   292→    fetchVarianceReport();
   293→  }, [fetchCostReport, fetchConsumptionReport, fetchVarianceReport]);
   294→
   295→  const handlePeriodChange = (p: PeriodType) => {
   296→    setPeriod(p);
   297→    setStartDate('');
   298→    setEndDate('');
   299→  };
   300→
   301→  const handleCustomApply = () => {
   302→    if (startDate && endDate) {
   303→      // Trigger re-fetch by changing the buildQuery deps
   304→      fetchCostReport();
   305→      fetchConsumptionReport();
   306→      fetchVarianceReport();
   307→    }
   308→  };
   309→
   310→  const totalEmployees = 600; // Default employee count for cost per employee calculation
   311→
   312→  // ─── Render ───────────────────────────────────────────────────────────────
   313→
   314→  return (
   315→    <div className="space-y-6">
   316→      {/* Header */}
   317→      <div className="flex items-center gap-3">
   318→        <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-900/30">
   319→          <BarChart3 className="h-5 w-5 text-amber-600 dark:text-amber-400" />
   320→        </div>
   321→        <div>
   322→          <h1 className="text-2xl font-bold tracking-tight">Reports</h1>
   323→          <p className="text-muted-foreground">
   324→            Cost, consumption, and variance analysis for canteen operations
   325→          </p>
   326→        </div>
   327→      </div>
   328→
   329→      {/* Period Selector */}
   330→      <Card>
   331→        <CardContent className="p-4">
   332→          <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   333→            <div className="flex items-center gap-2">
   334→              <CalendarDays className="h-4 w-4 text-amber-600" />
   335→              <span className="text-sm font-medium">Period:</span>
   336→              <div className="flex gap-1">
   337→                {(['today', 'week', 'month'] as PeriodType[]).map((p) => (
   338→                  <Button
   339→                    key={p}
   340→                    variant={period === p && !startDate && !endDate ? 'default' : 'outline'}
   341→                    size="sm"
   342→                    onClick={() => handlePeriodChange(p)}
   343→                    className={
   344→                      period === p && !startDate && !endDate
   345→                        ? 'bg-amber-600 hover:bg-amber-700 text-white'
   346→                        : ''
   347→                    }
   348→                  >
   349→                    {p === 'today' ? 'Today' : p === 'week' ? 'This Week' : 'This Month'}
   350→                  </Button>
   351→                ))}
   352→              </div>
   353→            </div>
   354→            <div className="flex items-center gap-2">
   355→              <span className="text-sm text-muted-foreground">Custom:</span>
   356→              <Input
   357→                type="date"
   358→                value={startDate}
   359→                onChange={(e) => setStartDate(e.target.value)}
   360→                className="h-9 w-auto text-sm"
   361→                placeholder="From"
   362→              />
   363→              <span className="text-sm text-muted-foreground">to</span>
   364→              <Input
   365→                type="date"
   366→                value={endDate}
   367→                onChange={(e) => setEndDate(e.target.value)}
   368→                className="h-9 w-auto text-sm"
   369→                placeholder="To"
   370→              />
   371→              <Button
   372→                size="sm"
   373→                onClick={handleCustomApply}
   374→                disabled={!startDate || !endDate}
   375→                className="bg-amber-600 hover:bg-amber-700 text-white"
   376→              >
   377→                Apply
   378→              </Button>
   379→            </div>
   380→          </div>
   381→        </CardContent>
   382→      </Card>
   383→
   384→      {/* Tabs */}
   385→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   386→        <TabsList className="w-full sm:w-auto">
   387→          <TabsTrigger value="cost" className="gap-1.5">
   388→            <IndianRupee className="h-4 w-4" />
   389→            Cost Report
   390→          </TabsTrigger>
   391→          <TabsTrigger value="consumption" className="gap-1.5">
   392→            <Package className="h-4 w-4" />
   393→            Consumption Report
   394→          </TabsTrigger>
   395→          <TabsTrigger value="variance" className="gap-1.5">
   396→            <AlertTriangle className="h-4 w-4" />
   397→            Variance Report
   398→          </TabsTrigger>
   399→        </TabsList>
   400→
   401→        {/* ─── Cost Report Tab ──────────────────────────────────────────── */}
   402→        <TabsContent value="cost" className="space-y-6">
   403→          {costLoading ? (
   404→            <CostReportSkeleton />
   405→          ) : costError ? (
   406→            <Card>
   407→              <CardContent className="flex flex-col items-center justify-center p-8">
   408→                <XCircle className="h-10 w-10 text-destructive mb-2" />
   409→                <p className="text-destructive font-medium">{costError}</p>
   410→                <Button variant="outline" size="sm" className="mt-3" onClick={fetchCostReport}>
   411→                  Retry
   412→                </Button>
   413→              </CardContent>
   414→            </Card>
   415→          ) : costData ? (
   416→            <>
   417→              {/* Summary Cards */}
   418→              <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   419→                <SummaryCard
   420→                  title="Total Food Cost"
   421→                  value={fmt.format(costData.foodCost.total)}
   422→                  icon={<IndianRupee className="h-5 w-5" />}
   423→                  iconBg="bg-amber-100 dark:bg-amber-900/30"
   424→                  iconColor="text-amber-600 dark:text-amber-400"
   425→                  subtitle={`Period: ${costData.period.type}`}
   426→                />
   427→                <SummaryCard
   428→                  title="Cost Per Meal"
   429→                  value={fmt.format(costData.foodCost.costPerMeal)}
   430→                  icon={<Utensils className="h-5 w-5" />}
   431→                  iconBg="bg-orange-100 dark:bg-orange-900/30"
   432→                  iconColor="text-orange-600 dark:text-orange-400"
   433→                  subtitle={`${fmtNum.format(costData.meals.total)} meals served`}
   434→                />
   435→                <SummaryCard
   436→                  title="Total Meals Served"
   437→                  value={fmtNum.format(costData.meals.total)}
   438→                  icon={<Utensils className="h-5 w-5" />}
   439→                  iconBg="bg-green-100 dark:bg-green-900/30"
   440→                  iconColor="text-green-600 dark:text-green-400"
   441→                  subtitle={`${costData.meals.count} meal entries`}
   442→                />
   443→                <SummaryCard
   444→                  title="Cost Per Employee"
   445→                  value={fmt.format(costData.foodCost.total / totalEmployees)}
   446→                  icon={<Users className="h-5 w-5" />}
   447→                  iconBg="bg-yellow-100 dark:bg-yellow-900/30"
   448→                  iconColor="text-yellow-600 dark:text-yellow-400"
   449→                  subtitle={`Based on ${totalEmployees} employees`}
   450→                />
   451→              </div>
   452→
   453→              {/* Operating Cost Banner */}
   454→              <Card className="border-amber-200 dark:border-amber-800/50">
   455→                <CardContent className="p-4">
   456→                  <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   457→                    <div>
   458→                      <p className="text-sm font-medium text-muted-foreground">Total Operating Cost</p>
   459→                      <p className="text-2xl font-bold text-amber-600 dark:text-amber-400">
   460→                        {fmt.format(costData.totalOperatingCost)}
   461→                      </p>
   462→                    </div>
   463→                    <div className="flex gap-4">
   464→                      <div className="text-center">
   465→                        <p className="text-xs text-muted-foreground">Food Cost</p>
   466→                        <p className="text-sm font-semibold">{fmt.format(costData.foodCost.total)}</p>
   467→                      </div>
   468→                      <div className="text-center">
   469→                        <p className="text-xs text-muted-foreground">Expenses</p>
   470→                        <p className="text-sm font-semibold">{fmt.format(costData.expenses.total)}</p>
   471→                      </div>
   472→                      <div className="text-center">
   473→                        <p className="text-xs text-muted-foreground">Per Meal</p>
   474→                        <p className="text-sm font-semibold">{fmt.format(costData.operatingCostPerMeal)}</p>
   475→                      </div>
   476→                    </div>
   477→                  </div>
   478→                </CardContent>
   479→              </Card>
   480→
   481→              {/* Charts Row */}
   482→              <div className="grid grid-cols-1 gap-6 lg:grid-cols-2">
   483→                {/* Cost Breakdown by Category (Pie Chart) */}
   484→                <Card>
   485→                  <CardHeader className="pb-2">
   486→                    <CardTitle className="text-base">Cost Breakdown by Category</CardTitle>
   487→                    <CardDescription>Food cost distribution across ingredient categories</CardDescription>
   488→                  </CardHeader>
   489→                  <CardContent>
   490→                    {costData.foodCost.categoryBreakdown.length > 0 ? (
   491→                      <ChartContainer config={costBreakdownConfig} className="mx-auto aspect-square max-h-[300px]">
   492→                        <PieChart>
   493→                          <ChartTooltip
   494→                            content={
   495→                              <ChartTooltipContent
   496→                                formatter={(value) => fmt.format(Number(value))}
   497→                              />
   498→                            }
   499→                          />
   500→                          <Pie
   501→                            data={costData.foodCost.categoryBreakdown.map((c) => ({
   502→                              name: c.category,
   503→                              value: c.totalCost,
   504→                            }))}
   505→                            cx="50%"
   506→                            cy="50%"
   507→                            innerRadius={60}
   508→                            outerRadius={100}
   509→                            paddingAngle={2}
   510→                            dataKey="value"
   511→                            nameKey="name"
   512→                          >
   513→                            {costData.foodCost.categoryBreakdown.map((entry, index) => (
   514→                              <Cell
   515→                                key={`cell-${index}`}
   516→                                fill={getCategoryColor(entry.category)}
   517→                              />
   518→                            ))}
   519→                          </Pie>
   520→                          <Legend
   521→                            verticalAlign="bottom"
   522→                            height={36}
   523→                            formatter={(value) => (
   524→                              <span className="text-xs">{value}</span>
   525→                            )}
   526→                          />
   527→                        </PieChart>
   528→                      </ChartContainer>
   529→                    ) : (
   530→                      <div className="flex h-[200px] items-center justify-center text-muted-foreground text-sm">
   531→                        No category data available
   532→                      </div>
   533→                    )}
   534→                  </CardContent>
   535→                </Card>
   536→
   537→                {/* Daily Cost Trend (Line Chart) */}
   538→                <Card>
   539→                  <CardHeader className="pb-2">
   540→                    <CardTitle className="text-base">Daily Cost Trend</CardTitle>
   541→                    <CardDescription>Food cost trend over the selected period</CardDescription>
   542→                  </CardHeader>
   543→                  <CardContent>
   544→                    {costData.foodCost.dailyTrend.length > 0 ? (
   545→                      <ChartContainer config={costTrendConfig} className="aspect-[2/1] w-full">
   546→                        <LineChart data={costData.foodCost.dailyTrend}>
   547→                          <CartesianGrid strokeDasharray="3 3" className="stroke-border/50" />
   548→                          <XAxis
   549→                            dataKey="date"
   550→                            tickFormatter={formatShortDate}
   551→                            tick={{ fontSize: 11 }}
   552→                            className="text-muted-foreground"
   553→                          />
   554→                          <YAxis
   555→                            tickFormatter={(v) => `₹${fmtNum.format(v / 1000)}k`}
   556→                            tick={{ fontSize: 11 }}
   557→                            className="text-muted-foreground"
   558→                          />
   559→                          <ChartTooltip
   560→                            content={
   561→                              <ChartTooltipContent
   562→                                labelFormatter={(label) => formatDate(label as string)}
   563→                                formatter={(value) => fmt.format(Number(value))}
   564→                              />
   565→                            }
   566→                          />
   567→                          <Line
   568→                            type="monotone"
   569→                            dataKey="cost"
   570→                            stroke="#f59e0b"
   571→                            strokeWidth={2}
   572→                            dot={{ r: 3, fill: '#f59e0b' }}
   573→                            activeDot={{ r: 5, fill: '#d97706' }}
   574→                          />
   575→                        </LineChart>
   576→                      </ChartContainer>
   577→                    ) : (
   578→                      <div className="flex h-[200px] items-center justify-center text-muted-foreground text-sm">
   579→                        No daily trend data available
   580→                      </div>
