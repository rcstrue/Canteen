    99→    99→
   100→   100→function getMealTypeBadgeClass(mealType: string): string {
   101→   101→  switch (mealType) {
   102→   102→    case "Breakfast":
   103→   103→      return "bg-yellow-100 text-yellow-800 border-yellow-300 dark:bg-yellow-900/30 dark:text-yellow-400 dark:border-yellow-700";
   104→   104→    case "Lunch":
   105→   105→      return "bg-orange-100 text-orange-800 border-orange-300 dark:bg-orange-900/30 dark:text-orange-400 dark:border-orange-700";
   106→   106→    case "Dinner":
   107→   107→      return "bg-green-100 text-green-800 border-green-300 dark:bg-green-900/30 dark:text-green-400 dark:border-green-700";
   108→   108→    case "Snack":
   109→   109→      return "bg-purple-100 text-purple-800 border-purple-300 dark:bg-purple-900/30 dark:text-purple-400 dark:border-purple-700";
   110→   110→    default:
   111→   111→      return "bg-gray-100 text-gray-800 border-gray-300 dark:bg-gray-900/30 dark:text-gray-400 dark:border-gray-700";
   112→   112→  }
   113→   113→}
   114→   114→
   115→   115→function formatRupee(amount: number): string {
   116→   116→  return "₹" + amount.toLocaleString("en-IN", { minimumFractionDigits: 2, maximumFractionDigits: 2 });
   117→   117→}
   118→   118→
   119→   119→function calcTotalIngredientCost(ingredients: RecipeIngredient[]): number {
   120→   120→  return ingredients.reduce((sum, ri) => sum + ri.quantity * ri.ingredient.avgCost, 0);
   121→   121→}
   122→   122→
   123→   123→function calcCostPerMeal(ingredients: RecipeIngredient[], baseServings: number): number {
   124→   124→  if (baseServings <= 0) return 0;
   125→   125→  return calcTotalIngredientCost(ingredients) / baseServings;
   126→   126→}
   127→   127→
   128→   128→function calcCostFor600(ingredients: RecipeIngredient[], baseServings: number): number {
   129→   129→  return calcCostPerMeal(ingredients, baseServings) * 600;
   130→   130→}
