   106→   106→/** Format number as Indian Rupee with 2 decimal places and Indian comma grouping */

   108→   108→  return new Intl.NumberFormat("en-IN", {
   109→   109→    style: "currency",
   110→   110→    currency: "INR",
   111→   111→    minimumFractionDigits: 2,
   112→   112→    maximumFractionDigits: 2,
   113→   113→  }).format(value);
   114→   114→}
   115→   115→
   116→   116→/** Format number with Indian comma grouping, no decimals */
