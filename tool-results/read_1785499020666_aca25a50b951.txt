   215→   215→  const [startDate, setStartDate] = useState('');
   216→   216→  const [endDate, setEndDate] = useState('');
   217→   217→  const [activeTab, setActiveTab] = useState('cost');
   218→   218→
   219→   219→  const [costData, setCostData] = useState<CostReport | null>(null);
   220→   220→  const [consumptionData, setConsumptionData] = useState<ConsumptionReport | null>(null);
   221→   221→  const [varianceData, setVarianceData] = useState<VarianceReport | null>(null);
   222→   222→
   223→   223→  const [costLoading, setCostLoading] = useState(false);
   224→   224→  const [consumptionLoading, setConsumptionLoading] = useState(false);
   225→   225→  const [varianceLoading, setVarianceLoading] = useState(false);
   226→   226→
   227→   227→  const [costError, setCostError] = useState('');
   228→   228→  const [consumptionError, setConsumptionError] = useState('');
   229→   229→  const [varianceError, setVarianceError] = useState('');
   230→   230→
   231→   231→  const buildQuery = useCallback(() => {
   232→   232→    const params = new URLSearchParams();
   233→   233→    if (startDate && endDate) {
   234→   234→      params.set('startDate', startDate);
   235→   235→      params.set('endDate', endDate);
   236→   236→      params.set('period', period);
   237→   237→    } else {
   238→   238→      params.set('period', period);
   239→   239→    }
   240→   240→    return params.toString();
   241→   241→  }, [period, startDate, endDate]);
   242→   242→
   243→   243→  const fetchCostReport = useCallback(async () => {
   244→   244→    setCostLoading(true);
   245→   245→    setCostError('');
   246→   246→    try {
   247→   247→      const res = await fetch(`/api/reports/cost?${buildQuery()}`);
   248→   248→      if (!res.ok) throw new Error('Failed to fetch cost report');
   249→   249→      const data = await res.json();
   250→   250→      setCostData(data);
   251→   251→    } catch {
   252→   252→      setCostError('Failed to load cost report');
   253→   253→    } finally {
   254→   254→      setCostLoading(false);
   255→   255→    }
   256→   256→  }, [buildQuery]);
   257→   257→
   258→   258→  const fetchConsumptionReport = useCallback(async () => {
   259→   259→    setConsumptionLoading(true);
   260→   260→    setConsumptionError('');
   261→   261→    try {
   262→   262→      const res = await fetch(`/api/reports/consumption?${buildQuery()}`);
   263→   263→      if (!res.ok) throw new Error('Failed to fetch consumption report');
   264→   264→      const data = await res.json();
   265→   265→      setConsumptionData(data);
   266→   266→    } catch {
   267→   267→      setConsumptionError('Failed to load consumption report');
   268→   268→    } finally {
   269→   269→      setConsumptionLoading(false);
   270→   270→    }
   271→   271→  }, [buildQuery]);
   272→   272→
   273→   273→  const fetchVarianceReport = useCallback(async () => {
   274→   274→    setVarianceLoading(true);
   275→   275→    setVarianceError('');

   277→   277→      const res = await fetch(`/api/reports/variance?${buildQuery()}`);
   278→   278→      if (!res.ok) throw new Error('Failed to fetch variance report');
   279→   279→      const data = await res.json();
   280→   280→      setVarianceData(data);
   281→   281→    } catch {
   282→   282→      setVarianceError('Failed to load variance report');
   283→   283→    } finally {
   284→   284→      setVarianceLoading(false);
   285→   285→    }
   286→   286→  }, [buildQuery]);

   288→   288→  // Fetch all reports when period changes
   289→   289→  useEffect(() => {
   290→   290→    fetchCostReport();
   291→   291→    fetchConsumptionReport();

   295→   295→  const handlePeriodChange = (p: PeriodType) => {
   296→   296→    setPeriod(p);
   297→   297→    setStartDate('');
   298→   298→    setEndDate('');

   303→   303→      // Trigger re-fetch by changing the buildQuery deps
   304→   304→      fetchCostReport();
   305→   305→      fetchConsumptionReport();
   306→   306→      fetchVarianceReport();
   307→   307→    }
   308→   308→  };
   309→   309→
   310→   310→  const totalEmployees = 600; // Default employee count for cost per employee calculation
   311→   311→
   312→   312→  // ─── Render ───────────────────────────────────────────────────────────────

   315→   315→    <div className="space-y-6">
   316→   316→      {/* Header */}

   322→   322→          <h1 className="text-2xl font-bold tracking-tight">Reports</h1>
   323→   323→          <p className="text-muted-foreground">
   324→   324→            Cost, consumption, and variance analysis for canteen operations
   325→   325→          </p>
   326→   326→        </div>
   327→   327→      </div>
   328→   328→
   329→   329→      {/* Period Selector */}
   330→   330→      <Card>
   331→   331→        <CardContent className="p-4">
   332→   332→          <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">
   333→   333→            <div className="flex items-center gap-2">
   334→   334→              <CalendarDays className="h-4 w-4 text-amber-600" />
   335→   335→              <span className="text-sm font-medium">Period:</span>
   336→   336→              <div className="flex gap-1">
   337→   337→                {(['today', 'week', 'month'] as PeriodType[]).map((p) => (
   338→   338→                  <Button
   339→   339→                    key={p}
   340→   340→                    variant={period === p && !startDate && !endDate ? 'default' : 'outline'}
   341→   341→                    size="sm"
   342→   342→                    onClick={() => handlePeriodChange(p)}
   343→   343→                    className={
   344→   344→                      period === p && !startDate && !endDate
   345→   345→                        ? 'bg-amber-600 hover:bg-amber-700 text-white'

   350→   350→                  </Button>
   351→   351→                ))}
   352→   352→              </div>

   355→   355→              <span className="text-sm text-muted-foreground">Custom:</span>

   357→   357→                type="date"

   371→   371→              <Button
   372→   372→                size="sm"

   374→   374→                disabled={!startDate || !endDate}
   375→   375→                className="bg-amber-600 hover:bg-amber-700 text-white"
   376→   376→              >
   377→   377→                Apply
   378→   378→              </Button>
   379→   379→            </div>
   380→   380→          </div>
