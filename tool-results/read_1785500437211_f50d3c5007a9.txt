   115→// ─── Helpers ─────────────────────────────────────────────────────────────────
   116→
   117→function formatCurrency(amount: number): string {
   118→  return new Intl.NumberFormat("en-IN", {
   119→    style: "currency",
   120→    currency: "INR",
   121→    minimumFractionDigits: 2,
   122→    maximumFractionDigits: 2,
   123→  }).format(amount);
   124→}
   125→
   126→function formatDate(dateStr: string): string {
   127→  const d = new Date(dateStr);
   128→  const day = String(d.getDate()).padStart(2, "0");
   129→  const month = String(d.getMonth() + 1).padStart(2, "0");
   130→  const year = d.getFullYear();
   131→  return `${day}/${month}/${year}`;
   132→}
   133→
   134→function getTodayStr(): string {
   135→  const d = new Date();
   136→  return d.toISOString().split("T")[0];
   137→}

   387→        </Button>

   393→          <CardContent className="p-4">

   409→                <p className="text-sm text-muted-foreground">Page Total</p>

   411→                  {formatCurrency(totalPurchaseAmount)}

   413→              </div>
   414→              <div className="flex h-10 w-10 items-center justify-center rounded-lg bg-orange-100 dark:bg-orange-900/30">

   416→              </div>
   417→            </div>
   418→          </CardContent>
   419→        </Card>
   420→        <Card>

   422→            <div className="flex items-center justify-between">
   423→              <div>
   424→                <p className="text-sm text-muted-foreground">Items on Page</p>
   425→                <p className="text-2xl font-bold">{purchases.length}</p>
