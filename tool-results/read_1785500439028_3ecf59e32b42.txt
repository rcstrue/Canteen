   126→
   127→function formatCurrency(amount: number): string {
   128→  return new Intl.NumberFormat("en-IN", {
   129→    style: "currency",
   130→    currency: "INR",
   131→    minimumFractionDigits: 2,
   132→    maximumFractionDigits: 2,
   133→  }).format(amount);
   134→}
   135→
   136→function formatDate(dateStr: string): string {
   137→  const d = new Date(dateStr);
   138→  const day = String(d.getDate()).padStart(2, "0");
   139→  const month = String(d.getMonth() + 1).padStart(2, "0");
   140→  const year = d.getFullYear();
   141→  return `${day}/${month}/${year}`;
   142→}
   143→
   144→function getTodayStr(): string {
   145→  const d = new Date();
   146→  return d.toISOString().split("T")[0];
   147→}
   148→
   149→function getMonthStartStr(): string {
   150→  const d = new Date();
   151→  return new Date(d.getFullYear(), d.getMonth(), 1).toISOString().split("T")[0];
   152→}
