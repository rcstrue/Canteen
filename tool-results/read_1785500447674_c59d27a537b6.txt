    97→function getCategoryColor(category: string): string {
    98→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
    99→  // Fallback: hash the category name to pick a consistent color
   100→  let hash = 0;
   101→  for (let i = 0; i < category.length; i++) {
   102→    hash = (hash << 5) - hash + category.charCodeAt(i);
   103→    hash |= 0;
   104→  }
   105→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   106→}

   242→// ─── Reusable: Trend Badge ──────────────────────────────────────────────────
   243→
   244→interface TrendBadgeProps {
   245→  /** Percentage change (positive = increase, negative = decrease) */
   246→  pct: number | null;
   247→  /** For cost metrics, "lower is better" → up arrow = red. For meals/positive metrics, "up is better" */
   248→  lowerIsBetter?: boolean;
   249→  /** Comparison label e.g. "vs avg day" */
   250→  label: string;
   251→}
   252→
   253→function TrendBadge({ pct, lowerIsBetter = true, label }: TrendBadgeProps) {
   254→  if (pct === null || !isFinite(pct)) {
   255→    return (
   256→      <div className="flex items-center gap-1 text-xs text-muted-foreground">
   257→        <Minus className="h-3 w-3" />
   258→        <span>{label}</span>
   259→      </div>
   260→    );
   261→  }
   262→
   263→  const isUp = pct > 0;
   264→  const isFlat = Math.abs(pct) < 0.5;
   265→  const goodDirection = lowerIsBetter ? !isUp : isUp;
   266→  const colorClass = isFlat
   267→    ? "text-muted-foreground"
   268→    : goodDirection
   269→      ? "text-emerald-600 dark:text-emerald-400"
   270→      : "text-rose-600 dark:text-rose-400";
   271→  const Icon = isFlat ? Minus : isUp ? TrendingUp : TrendingDown;
   272→
   273→  return (
   274→    <div className="flex items-center gap-1 text-xs">
   275→      <Icon className={`h-3.5 w-3.5 ${colorClass}`} />
   276→      <span className={`font-semibold tabular-nums ${colorClass}`}>
   277→        {isFlat ? "0%" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   278→      </span>
   279→      <span className="text-muted-foreground">{label}</span>
   280→    </div>
   281→  );
   282→}

   326→// ─── Stock Health Gauge ─────────────────────────────────────────────────────

   328→function CircularGauge({ percent }: { percent: number }) {
   329→  const size = 168;
   330→  const stroke = 14;
   331→  const radius = (size - stroke) / 2;
   332→  const circumference = 2 * Math.PI * radius;
   333→  const clamped = Math.max(0, Math.min(100, percent));
   334→  const dashOffset = circumference - (clamped / 100) * circumference;
   335→
   336→  // Color shifts: red (low) → amber (mid) → emerald (high)
   337→  const color =
   338→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   339→
   340→  return (
   341→    <div
   342→      className="relative"
   343→      style={{ width: size, height: size }}
   344→      role="meter"
   345→      aria-valuenow={Math.round(clamped)}
   346→      aria-valuemin={0}
   347→      aria-valuemax={100}
   348→      aria-label="Stock health percentage"
   349→    >
   350→      <svg width={size} height={size} className="-rotate-90">
   351→        <circle
   352→          cx={size / 2}
   353→          cy={size / 2}
   354→          r={radius}
   355→          fill="none"
   356→          stroke="var(--muted)"
   357→          strokeWidth={stroke}
   358→        />
   359→        <circle
   360→          cx={size / 2}
   361→          cy={size / 2}
   362→          r={radius}
   363→          fill="none"
   364→          stroke={color}
   365→          strokeWidth={stroke}
   366→          strokeLinecap="round"
   367→          strokeDasharray={circumference}
   368→          strokeDashoffset={dashOffset}
   369→          style={{
   370→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   371→          }}
   372→        />
   373→      </svg>
   374→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   375→        <span
   376→          className="text-3xl font-bold tabular-nums"
   377→          style={{ color }}
   378→        >
   379→          {Math.round(clamped)}%
   380→        </span>
   381→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   382→          Above Par
   383→        </span>
   384→      </div>
   385→    </div>
   386→  );
   387→}

   389→// ─── Donut Pie Label Renderer ───────────────────────────────────────────────
   390→
   391→/** Renders percentage labels INSIDE donut slices */
   392→function renderPiePercentLabel(props: {
   393→  cx: number;
   394→  cy: number;
   395→  midAngle: number;
   396→  innerRadius: number;
   397→  outerRadius: number;
   398→  percent: number;
   399→}) {
   400→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   401→  if (percent < 0.05) return null; // hide labels on slices < 5%
   402→
   403→  const RADIAN = Math.PI / 180;
   404→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   405→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   406→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   407→
   408→  return (
   409→    <text
   410→      x={x}
   411→      y={y}
   412→      fill="#fff"
   413→      textAnchor="middle"
   414→      dominantBaseline="central"
   415→      fontSize={12}
   416→      fontWeight={700}
   417→      style={{ pointerEvents: "none" }}
   418→    >
   419→      {(percent * 100).toFixed(0)}%
   420→    </text>
   421→  );
   422→}

   475→export function DashboardView({ onNavigate }: DashboardViewProps = {}) {
   476→  const [data, setData] = useState<DashboardData | null>(null);
   477→  const [allIngredients, setAllIngredients] = useState<IngredientListItem[]>(
   478→    []
   479→  );
   480→  const [loading, setLoading] = useState(true);
   481→  const [error, setError] = useState<string | null>(null);
   482→
   483→  useEffect(() => {
   484→    async function fetchDashboard() {
   485→      try {
   486→        setLoading(true);
   487→        setError(null);
   488→        const [dashRes, ingRes] = await Promise.all([
   489→          fetch("/api/dashboard"),
   490→          fetch("/api/ingredients"),
   491→        ]);
   492→        if (!dashRes.ok) {
   493→          throw new Error(
   494→            `Failed to fetch dashboard data (${dashRes.status})`
   495→          );
   496→        }
   497→        const json = await dashRes.json();
   498→        setData(json);
   499→        // Ingredients fetch is best-effort (used for stock health gauge)
   500→        if (ingRes.ok) {
   501→          const ingJson = (await ingRes.json()) as IngredientListItem[];
   502→          setAllIngredients(ingJson);
   503→        }
   504→      } catch (err) {
   505→        console.error("Dashboard fetch error:", err);
   506→        setError(
   507→          err instanceof Error ? err.message : "Failed to load dashboard"
   508→        );
   509→      } finally {
   510→        setLoading(false);
   511→      }
   512→    }
   513→    fetchDashboard();
   514→  }, []);
   515→
   516→  // ─── Loading State ──────────────────────────────────────────────────────
   517→
   518→  if (loading) {
   519→    return (
   520→      <div className="space-y-6">
   521→        <BannerSkeleton />
   522→        <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
   523→          {Array.from({ length: 4 }).map((_, i) => (
   524→            <MetricCardSkeleton key={i} />
   525→          ))}
   526→        </div>
   527→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2">
   528→          <LargeCardSkeleton />
   529→          <LargeCardSkeleton />
   530→        </div>
   531→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2">
   532→          <LargeCardSkeleton />
   533→          <LargeCardSkeleton />
   534→        </div>
   535→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2">
   536→          <LargeCardSkeleton />
   537→          <LargeCardSkeleton />
   538→        </div>
   539→      </div>
   540→    );
   541→  }
   542→
   543→  // ─── Error State ────────────────────────────────────────────────────────
   544→
   545→  if (error || !data) {
   546→    return (
   547→      <div className="space-y-6">
   548→        <Card className="border-destructive/50">
   549→          <CardHeader>
   550→            <CardTitle className="flex items-center gap-2 text-destructive">
   551→              <AlertTriangle className="h-5 w-5" />
   552→              Error Loading Dashboard
   553→            </CardTitle>
   554→            <CardDescription>
   555→              {error ||
   556→                "Unable to load dashboard data. Please try again later."}
   557→            </CardDescription>
   558→          </CardHeader>
   559→          <CardContent>
   560→            <button
   561→              onClick={() => window.location.reload()}
   562→              className="inline-flex items-center justify-center rounded-md bg-primary px-4 py-2 text-sm font-medium text-primary-foreground hover:bg-primary/90"
   563→            >
   564→              Retry
   565→            </button>
   566→          </CardContent>
   567→        </Card>
   568→      </div>
   569→    );
   570→  }
   571→
   572→  // ─── Computed Values ────────────────────────────────────────────────────
   573→
   574→  // Estimated weekly meals from month + today data (rough heuristic)
   575→  const weekMeals = data.meals.month
   576→    ? Math.round((data.meals.today * 7 + data.meals.month) / 8)
   577→    : data.meals.today * 7;
   578→
   579→  // Trend calculations (vs averages derived from existing data)
   580→  const avgDailyFromWeek = data.foodCost.week / 7;
   581→  const avgWeeklyFromMonth = data.foodCost.month / 4.33;
   582→  const todayVsAvgDay = pctChange(data.foodCost.today, avgDailyFromWeek);
   583→  const weekVsAvgWeek = pctChange(data.foodCost.week, avgWeeklyFromMonth);
   584→
   585→  // Cost per employee (monthly total operating cost / 600 employees)
   586→  const costPerEmployee = data.totalOperatingCost / EMPLOYEE_COUNT;
   587→  const dailyCostPerEmployee = costPerEmployee / 30;
   588→
   589→  // Stock health: % of ingredients at/above par level
   590→  const totalIngredientCount =
   591→    allIngredients.length > 0 ? allIngredients.length : data.lowStockAlerts.length;
   592→  const aboveParCount =
   593→    allIngredients.length > 0
   594→      ? allIngredients.filter((i) => i.currentStock >= i.minStock).length
   595→      : Math.max(0, totalIngredientCount - data.lowStockAlerts.length);
   596→  const stockHealthPct =
   597→    totalIngredientCount > 0
   598→      ? (aboveParCount / totalIngredientCount) * 100
   599→      : 100;
   600→
   601→  // Top consuming ingredients chart data
   602→  const consumptionChartData = data.topConsumingIngredients
   603→    .slice(0, 8)
   604→    .map((item) => ({
   605→      name:
   606→        item.ingredient.name.length > 14
   607→          ? item.ingredient.name.slice(0, 13) + "…"
   608→          : item.ingredient.name,
   609→      fullName: item.ingredient.name,
   610→      totalQuantity: Number(item.totalQuantity.toFixed(1)),
   611→      unit: item.ingredient.unit,
   612→      totalCost: item.totalCost,
   613→      category: item.ingredient.category,
   614→      color: getCategoryColor(item.ingredient.category),
   615→    }));
   616→
