   131→// ─── Helpers ─────────────────────────────────────────────────────────────────
   132→
   133→function formatCurrency(amount: number): string {
   134→  return new Intl.NumberFormat("en-IN", {
   135→    style: "currency",
   136→    currency: "INR",
   137→    minimumFractionDigits: 0,
   138→    maximumFractionDigits: 2,
   139→  }).format(amount);
   140→}
   141→
   142→function formatDate(dateStr: string): string {
   143→  const d = new Date(dateStr);
   144→  const day = String(d.getDate()).padStart(2, "0");
   145→  const month = String(d.getMonth() + 1).padStart(2, "0");
   146→  const year = d.getFullYear();
   147→  return `${day}/${month}/${year}`;
   148→}
   149→
   150→function getTodayStr(): string {
   151→  const d = new Date();
   152→  return d.toISOString().split("T")[0];
   153→}
   154→
   155→function getMonthStartStr(): string {
   156→  const d = new Date();
   157→  d.setDate(1);
   158→  return d.toISOString().split("T")[0];
   159→}

   290→  async function handleSubmit() {

   295→      const body = {
   296→        date: formDate,
   297→        category: formCategory,
   298→        amount: Number(formAmount),
   299→        description: formDescription || null,

   309→      } else {
   310→        const res = await fetch("/api/expenses", {
   311→          method: "POST",
   312→          headers: { "Content-Type": "application/json" },
   313→          body: JSON.stringify(body),
   314→        });

   316→      }
   317→
   318→      setDialogOpen(false);
   319→      fetchExpenses();
   320→    } catch (err) {
   321→      console.error("Error saving expense:", err);
   322→    } finally {
   323→      setSubmitting(false);
   324→    }
   325→  }

   331→        method: "DELETE",
   332→      });

   336→      fetchExpenses();

   338→      console.error("Error deleting expense:", err);
