   133→   133→function formatCurrency(amount: number): string {
   134→   134→  return new Intl.NumberFormat("en-IN", {
   135→   135→    style: "currency",
   136→   136→    currency: "INR",
   137→   137→    minimumFractionDigits: 0,
   138→   138→    maximumFractionDigits: 2,
   139→   139→  }).format(amount);
   140→   140→}
   141→   141→
   142→   142→function formatDate(dateStr: string): string {
   143→   143→  const d = new Date(dateStr);
   144→   144→  const day = String(d.getDate()).padStart(2, "0");
   145→   145→  const month = String(d.getMonth() + 1).padStart(2, "0");
   146→   146→  const year = d.getFullYear();
   147→   147→  return `${day}/${month}/${year}`;
   148→   148→}
   149→   149→
   150→   150→function getTodayStr(): string {
   151→   151→  const d = new Date();
   152→   152→  return d.toISOString().split("T")[0];
   153→   153→}
   154→   154→
   155→   155→function getMonthStartStr(): string {
   156→   156→  const d = new Date();
   157→   157→  d.setDate(1);
   158→   158→  return d.toISOString().split("T")[0];
   159→   159→}
