   123→function formatINR(value: number): string {
   124→  const formatted = new Intl.NumberFormat("en-IN", {
   125→    minimumFractionDigits: 2,
   126→    maximumFractionDigits: 2,
   127→  }).format(value || 0);
   128→  return `₹${formatted}`;
   129→}
   130→
   131→function formatINRShort(value: number): string {
   132→  if (value >= 100000) return `₹${(value / 100000).toFixed(2)}L`;
   133→  if (value >= 1000) return `₹${(value / 1000).toFixed(1)}k`;
   134→  return formatINR(value);
   135→}
   136→
   137→function getBudgetColor(pct: number): string {
   138→  if (pct > 100) return "#ef4444"; // red-500
   139→  if (pct >= 80) return "#f59e0b"; // amber-500
   140→  return "#10b981"; // emerald-500
   141→}
   142→
   143→function getBudgetColorClass(pct: number): string {
   144→  if (pct > 100) return "text-red-500";
   145→  if (pct >= 80) return "text-amber-500";
   146→  return "text-emerald-500";
   147→}
   148→
   149→function getProgressClass(pct: number): string {
   150→  if (pct > 100) return "[&>div]:bg-red-500";
   151→  if (pct >= 80) return "[&>div]:bg-amber-500";
   152→  return "[&>div]:bg-emerald-500";
   153→}

   155→// ─── Circular Budget Gauge ───────────────────────────────────────────────────
   156→
   157→function BudgetGauge({ percent, label, spent, budget }: {
   158→  percent: number;
   159→  label: string;
   160→  spent: number;
   161→  budget: number;
   162→}) {
   163→  const size = 140;
   164→  const stroke = 12;
   165→  const radius = (size - stroke) / 2;
   166→  const circumference = 2 * Math.PI * radius;
   167→  const clamped = Math.max(0, Math.min(percent, 150)); // allow up to 150% for visual
   168→  const dashOffset = circumference - (Math.min(clamped, 100) / 100) * circumference;
   169→  const color = getBudgetColor(percent);
   170→
   171→  return (
   172→    <div className="flex flex-col items-center gap-2">
   173→      <div
   174→        className="relative"
   175→        style={{ width: size, height: size }}
   176→        role="meter"
   177→        aria-valuenow={Math.round(percent)}
   178→        aria-valuemin={0}
   179→        aria-valuemax={100}
   180→        aria-label={`${label} budget utilization`}
   181→      >
   182→        <svg width={size} height={size} className="-rotate-90">
   183→          <circle
   184→            cx={size / 2}
   185→            cy={size / 2}
   186→            r={radius}
   187→            fill="none"
   188→            stroke="var(--muted)"
   189→            strokeWidth={stroke}
   190→          />
   191→          <circle
   192→            cx={size / 2}
   193→            cy={size / 2}
   194→            r={radius}
   195→            fill="none"
   196→            stroke={color}
   197→            strokeWidth={stroke}
   198→            strokeLinecap="round"
   199→            strokeDasharray={circumference}
   200→            strokeDashoffset={dashOffset}
   201→            style={{
   202→              transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   203→            }}
   204→          />
   205→        </svg>
   206→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   207→          <span
   208→            className="text-2xl font-bold tabular-nums"
   209→            style={{ color }}
   210→          >
   211→            {Math.round(percent)}%
   212→          </span>
   213→          <span className="text-[10px] font-medium text-muted-foreground">
   214→            Used
   215→          </span>
   216→        </div>
   217→      </div>
   218→      <div className="text-center">
   219→        <p className="text-sm font-semibold">{label}</p>
   220→        <p className="text-xs text-muted-foreground tabular-nums">
   221→          {formatINRShort(spent)} / {formatINRShort(budget)}
   222→        </p>
   223→      </div>
   224→    </div>
   225→  );
   226→}
