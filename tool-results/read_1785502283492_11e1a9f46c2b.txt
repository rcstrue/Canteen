   100→function getCategoryColor(category: string): string {
   101→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   102→  // Fallback: hash the category name to pick a consistent color
   103→  let hash = 0;
   104→  for (let i = 0; i < category.length; i++) {
   105→    hash = (hash << 5) - hash + category.charCodeAt(i);
   106→    hash |= 0;
   107→  }
   108→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   109→}

   253→interface TrendBadgeProps {
   254→  /** Percentage change (positive = increase, negative = decrease) */
   255→  pct: number | null;
   256→  /** For cost metrics, "lower is better" → up arrow = red. For meals/positive metrics, "up is better" */
   257→  lowerIsBetter?: boolean;
   258→  /** Comparison label e.g. "vs avg day" */
   259→  label: string;
   260→}
   261→
   262→function TrendBadge({ pct, lowerIsBetter = true, label }: TrendBadgeProps) {
   263→  if (pct === null || !isFinite(pct)) {
   264→    return (
   265→      <div className="flex items-center gap-1 text-xs text-muted-foreground">
   266→        <Minus className="h-3 w-3" />
   267→        <span>{label}</span>
   268→      </div>
   269→    );
   270→  }
   271→
   272→  const isUp = pct > 0;
   273→  const isFlat = Math.abs(pct) < 0.5;
   274→  const goodDirection = lowerIsBetter ? !isUp : isUp;
   275→  const colorClass = isFlat
   276→    ? "text-muted-foreground"
   277→    : goodDirection
   278→      ? "text-emerald-600 dark:text-emerald-400"
   279→      : "text-rose-600 dark:text-rose-400";
   280→  const Icon = isFlat ? Minus : isUp ? TrendingUp : TrendingDown;
   281→
   282→  return (
   283→    <div className="flex items-center gap-1 text-xs">
   284→      <Icon className={`h-3.5 w-3.5 ${colorClass}`} />
   285→      <span className={`font-semibold tabular-nums ${colorClass}`}>
   286→        {isFlat ? "0%" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   287→      </span>
   288→      <span className="text-muted-foreground">{label}</span>
   289→    </div>
   290→  );
   291→}

   335→// ─── Stock Health Gauge ─────────────────────────────────────────────────────
   336→
   337→function CircularGauge({ percent }: { percent: number }) {
   338→  const size = 168;
   339→  const stroke = 14;
   340→  const radius = (size - stroke) / 2;
   341→  const circumference = 2 * Math.PI * radius;
   342→  const clamped = Math.max(0, Math.min(100, percent));
   343→  const dashOffset = circumference - (clamped / 100) * circumference;
   344→
   345→  // Color shifts: red (low) → amber (mid) → emerald (high)
   346→  const color =
   347→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   348→
   349→  return (
   350→    <div
   351→      className="relative"
   352→      style={{ width: size, height: size }}
   353→      role="meter"
   354→      aria-valuenow={Math.round(clamped)}
   355→      aria-valuemin={0}
   356→      aria-valuemax={100}
   357→      aria-label="Stock health percentage"
   358→    >
   359→      <svg width={size} height={size} className="-rotate-90">
   360→        <circle
   361→          cx={size / 2}
   362→          cy={size / 2}
   363→          r={radius}
   364→          fill="none"
   365→          stroke="var(--muted)"
   366→          strokeWidth={stroke}
   367→        />
   368→        <circle
   369→          cx={size / 2}
   370→          cy={size / 2}
   371→          r={radius}
   372→          fill="none"
   373→          stroke={color}
   374→          strokeWidth={stroke}
   375→          strokeLinecap="round"
   376→          strokeDasharray={circumference}
   377→          strokeDashoffset={dashOffset}
   378→          style={{
   379→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   380→          }}
   381→        />
   382→      </svg>
   383→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   384→        <span
   385→          className="text-3xl font-bold tabular-nums"
   386→          style={{ color }}
   387→        >
   388→          {Math.round(clamped)}%
   389→        </span>
   390→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   391→          Above Par
   392→        </span>
   393→      </div>
   394→    </div>
   395→  );
   396→}

   398→// ─── Donut Pie Label Renderer ───────────────────────────────────────────────
   399→
   400→/** Renders percentage labels INSIDE donut slices */
   401→function renderPiePercentLabel(props: {
   402→  cx: number;
   403→  cy: number;
   404→  midAngle: number;
   405→  innerRadius: number;
   406→  outerRadius: number;
   407→  percent: number;
   408→}) {
   409→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   410→  if (percent < 0.05) return null; // hide labels on slices < 5%
   411→
   412→  const RADIAN = Math.PI / 180;
   413→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   414→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   415→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   416→
   417→  return (
   418→    <text
   419→      x={x}
   420→      y={y}
   421→      fill="#fff"
   422→      textAnchor="middle"
   423→      dominantBaseline="central"
   424→      fontSize={12}
   425→      fontWeight={700}
   426→      style={{ pointerEvents: "none" }}
   427→    >
   428→      {(percent * 100).toFixed(0)}%
   429→    </text>
   430→  );
   431→}

   484→export function DashboardView({ onNavigate }: DashboardViewProps = {}) {
   485→  const [data, setData] = useState<DashboardData | null>(null);
   486→  const [allIngredients, setAllIngredients] = useState<IngredientListItem[]>(
   487→    []
   488→  );
   489→  const [loading, setLoading] = useState(true);
   490→  const [error, setError] = useState<string | null>(null);
   491→
   492→  // Monthly comparison state
   493→  const [currentMonthReport, setCurrentMonthReport] = useState<CostReportData | null>(null);
   494→  const [prevMonthReport, setPrevMonthReport] = useState<CostReportData | null>(null);
   495→
   496→  useEffect(() => {
   497→    async function fetchDashboard() {
   498→      try {
   499→        setLoading(true);
   500→        setError(null);
   501→        const [dashRes, ingRes] = await Promise.all([
   502→          fetch("/api/dashboard"),
   503→          fetch("/api/ingredients"),
   504→        ]);
   505→        if (!dashRes.ok) {
   506→          throw new Error(
   507→            `Failed to fetch dashboard data (${dashRes.status})`
   508→          );
   509→        }
   510→        const json = await dashRes.json();
   511→        setData(json);
   512→        // Ingredients fetch is best-effort (used for stock health gauge)
   513→        if (ingRes.ok) {
   514→          const ingJson = (await ingRes.json()) as IngredientListItem[];
   515→          setAllIngredients(ingJson);
   516→        }
   517→      } catch (err) {
   518→        console.error("Dashboard fetch error:", err);
   519→        setError(
   520→          err instanceof Error ? err.message : "Failed to load dashboard"
   521→        );
   522→      } finally {
   523→        setLoading(false);
   524→      }
   525→    }
   526→    fetchDashboard();
   527→  }, []);
   528→
   529→  // Fetch monthly comparison data
   530→  useEffect(() => {
   531→    async function fetchMonthlyComparison() {
   532→      try {
   533→        const now = new Date();
   534→        // Current month: first day to now
   535→        const curStart = new Date(now.getFullYear(), now.getMonth(), 1)
   536→          .toISOString()
   537→          .split("T")[0];
   538→        const curEnd = now.toISOString().split("T")[0];
   539→
   540→        // Previous month: first day to last day
   541→        const prevStart = new Date(now.getFullYear(), now.getMonth() - 1, 1)
   542→          .toISOString()
   543→          .split("T")[0];
   544→        const prevEnd = new Date(now.getFullYear(), now.getMonth(), 0)
   545→          .toISOString()
   546→          .split("T")[0];
   547→
   548→        const [curRes, prevRes] = await Promise.all([
   549→          fetch(
   550→            `/api/reports/cost?period=month&startDate=${curStart}&endDate=${curEnd}`
   551→          ),
   552→          fetch(
   553→            `/api/reports/cost?period=month&startDate=${prevStart}&endDate=${prevEnd}`
   554→          ),
   555→        ]);
   556→
   557→        if (curRes.ok) {
   558→          setCurrentMonthReport(await curRes.json());
   559→        }
   560→        if (prevRes.ok) {
   561→          setPrevMonthReport(await prevRes.json());
   562→        }
   563→      } catch (err) {
   564→        console.error("Monthly comparison fetch error:", err);
   565→      }
   566→    }
   567→    fetchMonthlyComparison();
   568→  }, []);
   569→
   570→  // ─── Loading State ──────────────────────────────────────────────────────
   571→
   572→  if (loading) {
   573→    return (
   574→      <div className="space-y-6">
   575→        <BannerSkeleton />

   577→          {Array.from({ length: 4 }).map((_, i) => (
   578→            <MetricCardSkeleton key={i} />
   579→          ))}
   580→        </div>
   581→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2">
   582→          <LargeCardSkeleton />
   583→          <LargeCardSkeleton />
   584→        </div>
   585→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2">
   586→          <LargeCardSkeleton />
   587→          <LargeCardSkeleton />
   588→        </div>
   589→        <div className="grid grid-cols-1 gap-4 md:grid-cols-2">
   590→          <LargeCardSkeleton />
   591→          <LargeCardSkeleton />
   592→        </div>
   593→        <LargeCardSkeleton />
   594→      </div>
   595→    );
   596→  }
   597→
   598→  // ─── Error State ────────────────────────────────────────────────────────
   599→
   600→  if (error || !data) {
   601→    return (
   602→      <div className="space-y-6">
   603→        <Card className="border-destructive/50">
   604→          <CardHeader>
   605→            <CardTitle className="flex items-center gap-2 text-destructive">
   606→              <AlertTriangle className="h-5 w-5" />
   607→              Error Loading Dashboard
   608→            </CardTitle>
   609→            <CardDescription>
   610→              {error ||
   611→                "Unable to load dashboard data. Please try again later."}
   612→            </CardDescription>
   613→          </CardHeader>
   614→          <CardContent>
   615→            <button
   616→              onClick={() => window.location.reload()}
   617→              className="inline-flex items-center justify-center rounded-md bg-primary px-4 py-2 text-sm font-medium text-primary-foreground hover:bg-primary/90"
   618→            >
   619→              Retry
   620→            </button>
   621→          </CardContent>
   622→        </Card>
   623→      </div>
   624→    );
   625→  }
   626→
   627→  // ─── Computed Values ────────────────────────────────────────────────────
   628→
