   118→// ─── Helpers ─────────────────────────────────────────────────────────────────
   119→
   120→function formatCurrency(amount: number): string {
   121→  return new Intl.NumberFormat("en-IN", {
   122→    style: "currency",
   123→    currency: "INR",
   124→    minimumFractionDigits: 2,
   125→    maximumFractionDigits: 2,
   126→  }).format(amount);
   127→}
   128→
   129→function formatDate(dateStr: string): string {
   130→  const d = new Date(dateStr);
   131→  const day = String(d.getDate()).padStart(2, "0");
   132→  const month = String(d.getMonth() + 1).padStart(2, "0");
   133→  const year = d.getFullYear();
   134→  return `${day}/${month}/${year}`;
   135→}
   136→
   137→function getTodayStr(): string {
   138→  const d = new Date();
   139→  return d.toISOString().split("T")[0];
   140→}
