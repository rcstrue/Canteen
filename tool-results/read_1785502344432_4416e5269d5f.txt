   100→   100→function getCategoryColor(category: string): string {
   101→   101→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   102→   102→  // Fallback: hash the category name to pick a consistent color
   103→   103→  let hash = 0;
   104→   104→  for (let i = 0; i < category.length; i++) {
   105→   105→    hash = (hash << 5) - hash + category.charCodeAt(i);
   106→   106→    hash |= 0;
   107→   107→  }
   108→   108→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   109→   109→}

   253→   253→interface TrendBadgeProps {
   254→   254→  /** Percentage change (positive = increase, negative = decrease) */
   255→   255→  pct: number | null;
   256→   256→  /** For cost metrics, "lower is better" → up arrow = red. For meals/positive metrics, "up is better" */
   257→   257→  lowerIsBetter?: boolean;

   261→   261→
   262→   262→function TrendBadge({ pct, lowerIsBetter = true, label }: TrendBadgeProps) {
   263→   263→  if (pct === null || !isFinite(pct)) {
   264→   264→    return (
   265→   265→      <div className="flex items-center gap-1 text-xs text-muted-foreground">
   266→   266→        <Minus className="h-3 w-3" />
   267→   267→        <span>{label}</span>
   268→   268→      </div>
   269→   269→    );
   270→   270→  }
   271→   271→
   272→   272→  const isUp = pct > 0;
   273→   273→  const isFlat = Math.abs(pct) < 0.5;
   274→   274→  const goodDirection = lowerIsBetter ? !isUp : isUp;
   275→   275→  const colorClass = isFlat
   276→   276→    ? "text-muted-foreground"
   277→   277→    : goodDirection
   278→   278→      ? "text-emerald-600 dark:text-emerald-400"
   279→   279→      : "text-rose-600 dark:text-rose-400";
   280→   280→  const Icon = isFlat ? Minus : isUp ? TrendingUp : TrendingDown;
   281→   281→
   282→   282→  return (
   283→   283→    <div className="flex items-center gap-1 text-xs">
   284→   284→      <Icon className={`h-3.5 w-3.5 ${colorClass}`} />
   285→   285→      <span className={`font-semibold tabular-nums ${colorClass}`}>
   286→   286→        {isFlat ? "0%" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   287→   287→      </span>
   288→   288→      <span className="text-muted-foreground">{label}</span>
   289→   289→    </div>
   290→   290→  );
   291→   291→}

   336→   336→
   337→   337→function CircularGauge({ percent }: { percent: number }) {
   338→   338→  const size = 168;
   339→   339→  const stroke = 14;
   340→   340→  const radius = (size - stroke) / 2;
   341→   341→  const circumference = 2 * Math.PI * radius;
   342→   342→  const clamped = Math.max(0, Math.min(100, percent));
   343→   343→  const dashOffset = circumference - (clamped / 100) * circumference;
   344→   344→
   345→   345→  // Color shifts: red (low) → amber (mid) → emerald (high)
   346→   346→  const color =
   347→   347→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   348→   348→
   349→   349→  return (
   350→   350→    <div
   351→   351→      className="relative"
   352→   352→      style={{ width: size, height: size }}
   353→   353→      role="meter"
   354→   354→      aria-valuenow={Math.round(clamped)}
   355→   355→      aria-valuemin={0}
   356→   356→      aria-valuemax={100}
   357→   357→      aria-label="Stock health percentage"
   358→   358→    >
   359→   359→      <svg width={size} height={size} className="-rotate-90">
   360→   360→        <circle
   361→   361→          cx={size / 2}
   362→   362→          cy={size / 2}
   363→   363→          r={radius}
   364→   364→          fill="none"
   365→   365→          stroke="var(--muted)"
   366→   366→          strokeWidth={stroke}
   367→   367→        />
   368→   368→        <circle
   369→   369→          cx={size / 2}
   370→   370→          cy={size / 2}
   371→   371→          r={radius}
   372→   372→          fill="none"
   373→   373→          stroke={color}
   374→   374→          strokeWidth={stroke}
   375→   375→          strokeLinecap="round"
   376→   376→          strokeDasharray={circumference}
   377→   377→          strokeDashoffset={dashOffset}
   378→   378→          style={{
   379→   379→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   380→   380→          }}
   381→   381→        />
   382→   382→      </svg>
   383→   383→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   384→   384→        <span
   385→   385→          className="text-3xl font-bold tabular-nums"
   386→   386→          style={{ color }}
   387→   387→        >
   388→   388→          {Math.round(clamped)}%
   389→   389→        </span>
   390→   390→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   391→   391→          Above Par
   392→   392→        </span>
   393→   393→      </div>
   394→   394→    </div>
   395→   395→  );
   396→   396→}

   399→   399→
   400→   400→/** Renders percentage labels INSIDE donut slices */
   401→   401→function renderPiePercentLabel(props: {
   402→   402→  cx: number;
   403→   403→  cy: number;
   404→   404→  midAngle: number;
   405→   405→  innerRadius: number;
   406→   406→  outerRadius: number;
   407→   407→  percent: number;
   408→   408→}) {
   409→   409→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   410→   410→  if (percent < 0.05) return null; // hide labels on slices < 5%
   411→   411→
   412→   412→  const RADIAN = Math.PI / 180;
   413→   413→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   414→   414→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   415→   415→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   416→   416→
   417→   417→  return (
   418→   418→    <text
   419→   419→      x={x}
   420→   420→      y={y}
   421→   421→      fill="#fff"
   422→   422→      textAnchor="middle"
   423→   423→      dominantBaseline="central"
   424→   424→      fontSize={12}
   425→   425→      fontWeight={700}
   426→   426→      style={{ pointerEvents: "none" }}
   427→   427→    >
   428→   428→      {(percent * 100).toFixed(0)}%
   429→   429→    </text>
   430→   430→  );
   431→   431→}
