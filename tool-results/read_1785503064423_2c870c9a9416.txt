   159→function formatINR(value: number): string {
   160→  const formatted = new Intl.NumberFormat("en-IN", {
   161→    minimumFractionDigits: 2,
   162→    maximumFractionDigits: 2,
   163→  }).format(value || 0);
   164→  return `₹${formatted}`;
   165→}
   166→
   167→function formatINRShort(value: number): string {
   168→  if (value >= 100000) return `₹${(value / 100000).toFixed(2)}L`;
   169→  if (value >= 1000) return `₹${(value / 1000).toFixed(1)}k`;
   170→  return formatINR(value);
   171→}

   193→function formatBytes(bytes: number): string {
   194→  if (bytes === 0) return "0 B";
   195→  const units = ["B", "KB", "MB", "GB"];
   196→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
   197→  const v = bytes / Math.pow(1024, i);
   198→  return `${v.toFixed(v < 10 && i > 0 ? 1 : 0)} ${units[i]}`;
   199→}

   201→function formatRelativeDate(iso: string | null): string {
   202→  if (!iso) return "Never";
   203→  try {
   204→    const date = new Date(iso);
   205→    const now = new Date();
   206→    const diffMs = now.getTime() - date.getTime();
   207→    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   208→    if (diffDays === 0) {
   209→      const diffHours = Math.floor(diffMs / (1000 * 60 * 60));
   210→      if (diffHours === 0) {
   211→        const diffMins = Math.floor(diffMs / (1000 * 60));
   212→        return diffMins <= 1 ? "Just now" : `${diffMins} minutes ago`;
   213→      }
   214→      return `${diffHours} hour${diffHours === 1 ? "" : "s"} ago`;
   215→    }
   216→    if (diffDays === 1) return "Yesterday";
   217→    if (diffDays < 7) return `${diffDays} days ago`;
   218→    return date.toLocaleDateString("en-IN", {
   219→      year: "numeric",
   220→      month: "short",
   221→      day: "numeric",
   222→    });
   223→  } catch {
   224→    return "Unknown";
   225→  }
   226→}
   227→
   228→function isStaleBackup(iso: string | null, days = 7): boolean {
   229→  if (!iso) return true;
   230→  try {
   231→    const date = new Date(iso);
   232→    const diffMs = Date.now() - date.getTime();
   233→    return diffMs > days * 24 * 60 * 60 * 1000;
   234→  } catch {
   235→    return true;
   236→  }
   237→}

   239→function validateBackupFile(parsed: unknown): {
   240→  valid: boolean;
   241→  error?: string;
   242→  file?: BackupFile;
   243→} {
   244→  if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
   245→    return { valid: false, error: "Invalid format: expected a JSON object." };
   246→  }
   247→  const obj = parsed as Record<string, unknown>;
   248→  const metadata = obj.metadata as Record<string, unknown> | undefined;
   249→  const data = obj.data as Record<string, unknown> | undefined;
   250→  if (!metadata || typeof metadata !== "object") {
   251→    return { valid: false, error: 'Missing "metadata" object.' };
   252→  }
   253→  if (!data || typeof data !== "object") {
   254→    return { valid: false, error: 'Missing "data" object.' };
   255→  }
   256→  // At least one collection must be an array
   257→  const collections = ["ingredients", "recipes", "stockMovements", "dailyMeals", "purchases", "expenses"];
   258→  const hasAny = collections.some((k) => Array.isArray((data as Record<string, unknown>)[k]));
   259→  if (!hasAny) {
   260→    return { valid: false, error: "Backup contains no recognizable data collections." };
   261→  }
   262→  return {
   263→    valid: true,
   264→    file: {
   265→      metadata: metadata as unknown as BackupMetadata,
   266→      data,
   267→    },
   268→  };
   269→}

   271→// ─── Circular Budget Gauge ───────────────────────────────────────────────────
   272→
   273→function BudgetGauge({ percent, label, spent, budget }: {
   274→  percent: number;
   275→  label: string;
   276→  spent: number;
   277→  budget: number;
   278→}) {
   279→  const size = 140;
   280→  const stroke = 12;
   281→  const radius = (size - stroke) / 2;
   282→  const circumference = 2 * Math.PI * radius;
   283→  const clamped = Math.max(0, Math.min(percent, 150)); // allow up to 150% for visual
   284→  const dashOffset = circumference - (Math.min(clamped, 100) / 100) * circumference;
   285→  const color = getBudgetColor(percent);
   286→
   287→  return (
   288→    <div className="flex flex-col items-center gap-2">
   289→      <div
   290→        className="relative"
   291→        style={{ width: size, height: size }}
   292→        role="meter"
   293→        aria-valuenow={Math.round(percent)}
   294→        aria-valuemin={0}
   295→        aria-valuemax={100}
   296→        aria-label={`${label} budget utilization`}
   297→      >
   298→        <svg width={size} height={size} className="-rotate-90">
   299→          <circle
   300→            cx={size / 2}
   301→            cy={size / 2}
   302→            r={radius}
   303→            fill="none"
   304→            stroke="var(--muted)"
   305→            strokeWidth={stroke}
   306→          />
   307→          <circle
   308→            cx={size / 2}
   309→            cy={size / 2}
   310→            r={radius}
   311→            fill="none"
   312→            stroke={color}
   313→            strokeWidth={stroke}
   314→            strokeLinecap="round"
   315→            strokeDasharray={circumference}
   316→            strokeDashoffset={dashOffset}
   317→            style={{
   318→              transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   319→            }}
   320→          />
   321→        </svg>
   322→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   323→          <span
   324→            className="text-2xl font-bold tabular-nums"
   325→            style={{ color }}
   326→          >
   327→            {Math.round(percent)}%
   328→          </span>
   329→          <span className="text-[10px] font-medium text-muted-foreground">
   330→            Used
   331→          </span>
   332→        </div>
   333→      </div>
   334→      <div className="text-center">
   335→        <p className="text-sm font-semibold">{label}</p>
   336→        <p className="text-xs text-muted-foreground tabular-nums">
   337→          {formatINRShort(spent)} / {formatINRShort(budget)}
   338→        </p>
   339→      </div>
   340→    </div>
   341→  );
   342→}
