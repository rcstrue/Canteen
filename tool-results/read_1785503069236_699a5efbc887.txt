   159→   159→function formatINR(value: number): string {
   160→   160→  const formatted = new Intl.NumberFormat("en-IN", {
   161→   161→    minimumFractionDigits: 2,
   162→   162→    maximumFractionDigits: 2,
   163→   163→  }).format(value || 0);
   164→   164→  return `₹${formatted}`;
   165→   165→}
   166→   166→
   167→   167→function formatINRShort(value: number): string {
   168→   168→  if (value >= 100000) return `₹${(value / 100000).toFixed(2)}L`;
   169→   169→  if (value >= 1000) return `₹${(value / 1000).toFixed(1)}k`;
   170→   170→  return formatINR(value);
   171→   171→}

   193→   193→function formatBytes(bytes: number): string {
   194→   194→  if (bytes === 0) return "0 B";
   195→   195→  const units = ["B", "KB", "MB", "GB"];
   196→   196→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
   197→   197→  const v = bytes / Math.pow(1024, i);
   198→   198→  return `${v.toFixed(v < 10 && i > 0 ? 1 : 0)} ${units[i]}`;
   199→   199→}

   201→   201→function formatRelativeDate(iso: string | null): string {
   202→   202→  if (!iso) return "Never";
   203→   203→  try {
   204→   204→    const date = new Date(iso);
   205→   205→    const now = new Date();
   206→   206→    const diffMs = now.getTime() - date.getTime();
   207→   207→    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   208→   208→    if (diffDays === 0) {
   209→   209→      const diffHours = Math.floor(diffMs / (1000 * 60 * 60));
   210→   210→      if (diffHours === 0) {
   211→   211→        const diffMins = Math.floor(diffMs / (1000 * 60));
   212→   212→        return diffMins <= 1 ? "Just now" : `${diffMins} minutes ago`;
   213→   213→      }
   214→   214→      return `${diffHours} hour${diffHours === 1 ? "" : "s"} ago`;
   215→   215→    }
   216→   216→    if (diffDays === 1) return "Yesterday";
   217→   217→    if (diffDays < 7) return `${diffDays} days ago`;
   218→   218→    return date.toLocaleDateString("en-IN", {
   219→   219→      year: "numeric",
   220→   220→      month: "short",
   221→   221→      day: "numeric",
   222→   222→    });
   223→   223→  } catch {
   224→   224→    return "Unknown";
   225→   225→  }
   226→   226→}

   228→   228→function isStaleBackup(iso: string | null, days = 7): boolean {
   229→   229→  if (!iso) return true;
   230→   230→  try {
   231→   231→    const date = new Date(iso);
   232→   232→    const diffMs = Date.now() - date.getTime();
   233→   233→    return diffMs > days * 24 * 60 * 60 * 1000;
   234→   234→  } catch {
   235→   235→    return true;
   236→   236→  }
   237→   237→}

   239→   239→function validateBackupFile(parsed: unknown): {
   240→   240→  valid: boolean;
   241→   241→  error?: string;
   242→   242→  file?: BackupFile;
   243→   243→} {
   244→   244→  if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
   245→   245→    return { valid: false, error: "Invalid format: expected a JSON object." };
   246→   246→  }
   247→   247→  const obj = parsed as Record<string, unknown>;
   248→   248→  const metadata = obj.metadata as Record<string, unknown> | undefined;
   249→   249→  const data = obj.data as Record<string, unknown> | undefined;
   250→   250→  if (!metadata || typeof metadata !== "object") {
   251→   251→    return { valid: false, error: 'Missing "metadata" object.' };
   252→   252→  }
   253→   253→  if (!data || typeof data !== "object") {
   254→   254→    return { valid: false, error: 'Missing "data" object.' };
   255→   255→  }
   256→   256→  // At least one collection must be an array
   257→   257→  const collections = ["ingredients", "recipes", "stockMovements", "dailyMeals", "purchases", "expenses"];
   258→   258→  const hasAny = collections.some((k) => Array.isArray((data as Record<string, unknown>)[k]));
   259→   259→  if (!hasAny) {
   260→   260→    return { valid: false, error: "Backup contains no recognizable data collections." };
   261→   261→  }
   262→   262→  return {
   263→   263→    valid: true,
   264→   264→    file: {
   265→   265→      metadata: metadata as unknown as BackupMetadata,
   266→   266→      data,
   267→   267→    },
   268→   268→  };
   269→   269→}

   271→   271→// ─── Circular Budget Gauge ───────────────────────────────────────────────────
   272→   272→
   273→   273→function BudgetGauge({ percent, label, spent, budget }: {
   274→   274→  percent: number;
   275→   275→  label: string;
   276→   276→  spent: number;
   277→   277→  budget: number;
   278→   278→}) {
   279→   279→  const size = 140;
   280→   280→  const stroke = 12;
   281→   281→  const radius = (size - stroke) / 2;
   282→   282→  const circumference = 2 * Math.PI * radius;
   283→   283→  const clamped = Math.max(0, Math.min(percent, 150)); // allow up to 150% for visual
   284→   284→  const dashOffset = circumference - (Math.min(clamped, 100) / 100) * circumference;
   285→   285→  const color = getBudgetColor(percent);
   286→   286→
   287→   287→  return (
   288→   288→    <div className="flex flex-col items-center gap-2">
   289→   289→      <div
   290→   290→        className="relative"
   291→   291→        style={{ width: size, height: size }}
   292→   292→        role="meter"
   293→   293→        aria-valuenow={Math.round(percent)}
   294→   294→        aria-valuemin={0}
   295→   295→        aria-valuemax={100}
   296→   296→        aria-label={`${label} budget utilization`}
   297→   297→      >
   298→   298→        <svg width={size} height={size} className="-rotate-90">
   299→   299→          <circle
   300→   300→            cx={size / 2}
   301→   301→            cy={size / 2}
   302→   302→            r={radius}
   303→   303→            fill="none"
   304→   304→            stroke="var(--muted)"
   305→   305→            strokeWidth={stroke}
   306→   306→          />
   307→   307→          <circle
   308→   308→            cx={size / 2}
   309→   309→            cy={size / 2}
   310→   310→            r={radius}
   311→   311→            fill="none"
   312→   312→            stroke={color}
   313→   313→            strokeWidth={stroke}
   314→   314→            strokeLinecap="round"
   315→   315→            strokeDasharray={circumference}
   316→   316→            strokeDashoffset={dashOffset}
   317→   317→            style={{
   318→   318→              transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   319→   319→            }}
   320→   320→          />
   321→   321→        </svg>
   322→   322→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   323→   323→          <span
   324→   324→            className="text-2xl font-bold tabular-nums"
   325→   325→            style={{ color }}
   326→   326→          >
   327→   327→            {Math.round(percent)}%
   328→   328→          </span>
   329→   329→          <span className="text-[10px] font-medium text-muted-foreground">
   330→   330→            Used
   331→   331→          </span>
   332→   332→        </div>
   333→   333→      </div>
   334→   334→      <div className="text-center">
   335→   335→        <p className="text-sm font-semibold">{label}</p>
   336→   336→        <p className="text-xs text-muted-foreground tabular-nums">
   337→   337→          {formatINRShort(spent)} / {formatINRShort(budget)}
   338→   338→        </p>
   339→   339→      </div>
   340→   340→    </div>
   341→   341→  );
   342→   342→}
