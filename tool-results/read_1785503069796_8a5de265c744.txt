   105→   105→function getCategoryColor(category: string): string {
   106→   106→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   107→   107→  // Fallback: hash the category name to pick a consistent color
   108→   108→  let hash = 0;
   109→   109→  for (let i = 0; i < category.length; i++) {
   110→   110→    hash = (hash << 5) - hash + category.charCodeAt(i);
   111→   111→    hash |= 0;
   112→   112→  }
   113→   113→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   114→   114→}

   262→   262→
   263→   263→interface TrendBadgeProps {
   264→   264→  /** Percentage change (positive = increase, negative = decrease) */
   265→   265→  pct: number | null;
   266→   266→  /** For cost metrics, "lower is better" → up arrow = red. For meals/positive metrics, "up is better" */
   267→   267→  lowerIsBetter?: boolean;
   268→   268→  /** Comparison label e.g. "vs avg day" */
   269→   269→  label: string;
   270→   270→}
   271→   271→
   272→   272→function TrendBadge({ pct, lowerIsBetter = true, label }: TrendBadgeProps) {
   273→   273→  if (pct === null || !isFinite(pct)) {
   274→   274→    return (
   275→   275→      <div className="flex items-center gap-1 text-xs text-muted-foreground">
   276→   276→        <Minus className="h-3.5 w-3.5" />
   277→   277→        <span>{label}</span>
   278→   278→      </div>
   279→   279→    );
   280→   280→  }
   281→   281→
   282→   282→  const isUp = pct > 0;
   283→   283→  const isFlat = Math.abs(pct) < 0.5;
   284→   284→  const goodDirection = lowerIsBetter ? !isUp : isUp;
   285→   285→  const colorClass = isFlat
   286→   286→    ? "text-muted-foreground"
   287→   287→    : goodDirection
   288→   288→      ? "text-emerald-600 dark:text-emerald-400"
   289→   289→      : "text-rose-600 dark:text-rose-400";
   290→   290→  const Icon = isFlat ? Minus : isUp ? TrendingUp : TrendingDown;
   291→   291→
   292→   292→  return (
   293→   293→    <div className="flex items-center gap-1.5 text-xs">
   294→   294→      <Icon className={`h-4 w-4 ${colorClass}`} />
   295→   295→      <span className={`font-bold tabular-nums ${colorClass}`}>
   296→   296→        {isFlat ? "0%" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   297→   297→      </span>
   298→   298→      <span className="text-muted-foreground">{label}</span>
   299→   299→    </div>
   300→   300→  );
   301→   301→}

   355→   355→function CircularGauge({ percent }: { percent: number }) {
   356→   356→  const size = 168;
   357→   357→  const stroke = 14;
   358→   358→  const radius = (size - stroke) / 2;
   359→   359→  const circumference = 2 * Math.PI * radius;
   360→   360→  const clamped = Math.max(0, Math.min(100, percent));
   361→   361→  const dashOffset = circumference - (clamped / 100) * circumference;
   362→   362→
   363→   363→  // Color shifts: red (low) → amber (mid) → emerald (high)
   364→   364→  const color =
   365→   365→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   366→   366→
   367→   367→  return (
   368→   368→    <div
   369→   369→      className="relative"
   370→   370→      style={{ width: size, height: size }}
   371→   371→      role="meter"
   372→   372→      aria-valuenow={Math.round(clamped)}
   373→   373→      aria-valuemin={0}
   374→   374→      aria-valuemax={100}
   375→   375→      aria-label="Stock health percentage"
   376→   376→    >
   377→   377→      <svg width={size} height={size} className="-rotate-90">
   378→   378→        <circle
   379→   379→          cx={size / 2}
   380→   380→          cy={size / 2}
   381→   381→          r={radius}
   382→   382→          fill="none"
   383→   383→          stroke="var(--muted)"
   384→   384→          strokeWidth={stroke}
   385→   385→        />
   386→   386→        <circle
   387→   387→          cx={size / 2}
   388→   388→          cy={size / 2}
   389→   389→          r={radius}
   390→   390→          fill="none"
   391→   391→          stroke={color}
   392→   392→          strokeWidth={stroke}
   393→   393→          strokeLinecap="round"
   394→   394→          strokeDasharray={circumference}
   395→   395→          strokeDashoffset={dashOffset}
   396→   396→          style={{
   397→   397→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   398→   398→          }}
   399→   399→        />
   400→   400→      </svg>
   401→   401→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   402→   402→        <span
   403→   403→          className="text-3xl font-bold tabular-nums"
   404→   404→          style={{ color }}
   405→   405→        >
   406→   406→          {Math.round(clamped)}%
   407→   407→        </span>
   408→   408→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   409→   409→          Above Par
   410→   410→        </span>
   411→   411→      </div>
   412→   412→    </div>
   413→   413→  );
   414→   414→}

   418→   418→/** Renders percentage labels INSIDE donut slices */
   419→   419→function renderPiePercentLabel(props: {
   420→   420→  cx: number;
   421→   421→  cy: number;
   422→   422→  midAngle: number;
   423→   423→  innerRadius: number;
   424→   424→  outerRadius: number;
   425→   425→  percent: number;
   426→   426→}) {
   427→   427→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   428→   428→  if (percent < 0.05) return null; // hide labels on slices < 5%
   429→   429→
   430→   430→  const RADIAN = Math.PI / 180;
   431→   431→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   432→   432→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   433→   433→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   434→   434→
   435→   435→  return (
   436→   436→    <text
   437→   437→      x={x}
   438→   438→      y={y}
   439→   439→      fill="#fff"
   440→   440→      textAnchor="middle"
   441→   441→      dominantBaseline="central"
   442→   442→      fontSize={12}
   443→   443→      fontWeight={700}
   444→   444→      style={{ pointerEvents: "none" }}
   445→   445→    >
   446→   446→      {(percent * 100).toFixed(0)}%
   447→   447→    </text>
   448→   448→  );
   449→   449→}
