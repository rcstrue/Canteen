   199→function getMealTypeStyle(mealType: string) {
   200→  return MEAL_TYPE_STYLES[mealType] ?? MEAL_TYPE_STYLES.default;
   201→}
   202→
   203→function formatRupee(amount: number): string {
   204→  return (
   205→    "₹" +
   206→    amount.toLocaleString("en-IN", {
   207→      minimumFractionDigits: 2,
   208→      maximumFractionDigits: 2,
   209→    })
   210→  );
   211→}
   212→
   213→function calcTotalIngredientCost(ingredients: RecipeIngredient[]): number {
   214→  return ingredients.reduce((sum, ri) => sum + ri.quantity * ri.ingredient.avgCost, 0);
   215→}
   216→
   217→function calcCostPerMeal(ingredients: RecipeIngredient[], baseServings: number): number {
   218→  if (baseServings <= 0) return 0;
   219→  return calcTotalIngredientCost(ingredients) / baseServings;
   220→}
   221→
   222→function calcCostFor600(ingredients: RecipeIngredient[], baseServings: number): number {
   223→  return calcCostPerMeal(ingredients, baseServings) * 600;
   224→}
   225→
   226→function calcCostBreakdown(ingredients: RecipeIngredient[]): {
   227→  entries: CategoryBreakdownEntry[];
   228→  total: number;
   229→} {
   230→  const byCategory: Record<string, number> = {};
   231→  let total = 0;
   232→  ingredients.forEach((ri) => {
   233→    const cost = ri.quantity * ri.ingredient.avgCost;
   234→    const cat = ri.ingredient.category || "Other";
   235→    byCategory[cat] = (byCategory[cat] || 0) + cost;
   236→    total += cost;
   237→  });
   238→  const entries = Object.entries(byCategory)
   239→    .map(([category, amount]) => ({
   240→      category,
   241→      amount,
   242→      percent: total > 0 ? (amount / total) * 100 : 0,
   243→    }))
   244→    .sort((a, b) => b.amount - a.amount);
   245→  return { entries, total };
   246→}
