   251→function formatCurrency(value: number): string {
   252→  return `₹${value.toFixed(2)}`;
   253→}
   254→
   255→function formatStockWithUnit(quantity: number, unit: string): string {
   256→  return `${quantity} ${unit}`;
   257→}
   258→
   259→function isLowStock(item: Ingredient): boolean {
   260→  return item.currentStock < item.minStock;
   261→}
   262→
   263→function formatDateDDMMYYYY(dateStr: string): string {
   264→  if (!dateStr) return "—";
   265→  const d = new Date(dateStr);
   266→  if (isNaN(d.getTime())) return "—";
   267→  const dd = String(d.getDate()).padStart(2, "0");
   268→  const mm = String(d.getMonth() + 1).padStart(2, "0");
   269→  const yyyy = d.getFullYear();
   270→  return `${dd}/${mm}/${yyyy}`;
   271→}
   272→
   273→function formatINR(value: number): string {
   274→  const formatted = new Intl.NumberFormat("en-IN", {
   275→    minimumFractionDigits: 2,
   276→    maximumFractionDigits: 2,
   277→  }).format(value || 0);
   278→  return `₹${formatted}`;
   279→}
