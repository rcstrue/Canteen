   129→
   130→function formatCurrency(amount: number): string {
   131→  return new Intl.NumberFormat("en-IN", {
   132→    style: "currency",
   133→    currency: "INR",
   134→    minimumFractionDigits: 2,
   135→    maximumFractionDigits: 2,
   136→  }).format(amount);
   137→}
   138→
   139→function formatDate(dateStr: string): string {
   140→  const d = new Date(dateStr);
   141→  const day = String(d.getDate()).padStart(2, "0");
   142→  const month = String(d.getMonth() + 1).padStart(2, "0");
   143→  const year = d.getFullYear();
   144→  return `${day}/${month}/${year}`;
   145→}
   146→
   147→function getTodayStr(): string {
   148→  const d = new Date();
   149→  return d.toISOString().split("T")[0];
   150→}
   151→
   152→function getMonthStartStr(): string {
   153→  const d = new Date();
   154→  return new Date(d.getFullYear(), d.getMonth(), 1).toISOString().split("T")[0];
   155→}

   794→                  Record Wastage
   795→                </>
