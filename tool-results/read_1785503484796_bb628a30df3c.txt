   137→// ─── Helpers ─────────────────────────────────────────────────────────────────
   138→
   139→function formatCurrency(amount: number): string {
   140→  return new Intl.NumberFormat("en-IN", {
   141→    style: "currency",
   142→    currency: "INR",
   143→    minimumFractionDigits: 0,
   144→    maximumFractionDigits: 2,
   145→  }).format(amount);
   146→}
   147→
   148→function formatDate(dateStr: string): string {
   149→  const d = new Date(dateStr);
   150→  const day = String(d.getDate()).padStart(2, "0");
   151→  const month = String(d.getMonth() + 1).padStart(2, "0");
   152→  const year = d.getFullYear();
   153→  return `${day}/${month}/${year}`;
   154→}
   155→
   156→function getTodayStr(): string {
   157→  const d = new Date();
   158→  return d.toISOString().split("T")[0];
   159→}
   160→
   161→function getMonthStartStr(): string {
   162→  const d = new Date();
   163→  d.setDate(1);
   164→  return d.toISOString().split("T")[0];
   165→}
