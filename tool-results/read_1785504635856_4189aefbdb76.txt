   170→   170→function formatINR(value: number): string {
   171→   171→  const formatted = new Intl.NumberFormat("en-IN", {
   172→   172→    minimumFractionDigits: 2,
   173→   173→    maximumFractionDigits: 2,
   174→   174→  }).format(value || 0);
   175→   175→  return `₹${formatted}`;
   176→   176→}
   177→   177→
   178→   178→function formatINRShort(value: number): string {
   179→   179→  if (value >= 100000) return `₹${(value / 100000).toFixed(2)}L`;
   180→   180→  if (value >= 1000) return `₹${(value / 1000).toFixed(1)}k`;
   181→   181→  return formatINR(value);
   182→   182→}

   204→   204→function formatBytes(bytes: number): string {
   205→   205→  if (bytes === 0) return "0 B";
   206→   206→  const units = ["B", "KB", "MB", "GB"];
   207→   207→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
   208→   208→  const v = bytes / Math.pow(1024, i);
   209→   209→  return `${v.toFixed(v < 10 && i > 0 ? 1 : 0)} ${units[i]}`;
   210→   210→}

   212→   212→function formatRelativeDate(iso: string | null): string {
   213→   213→  if (!iso) return "Never";
   214→   214→  try {
   215→   215→    const date = new Date(iso);
   216→   216→    const now = new Date();
   217→   217→    const diffMs = now.getTime() - date.getTime();
   218→   218→    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   219→   219→    if (diffDays === 0) {
   220→   220→      const diffHours = Math.floor(diffMs / (1000 * 60 * 60));
   221→   221→      if (diffHours === 0) {
   222→   222→        const diffMins = Math.floor(diffMs / (1000 * 60));
   223→   223→        return diffMins <= 1 ? "Just now" : `${diffMins} minutes ago`;
   224→   224→      }
   225→   225→      return `${diffHours} hour${diffHours === 1 ? "" : "s"} ago`;
   226→   226→    }
   227→   227→    if (diffDays === 1) return "Yesterday";
   228→   228→    if (diffDays < 7) return `${diffDays} days ago`;
   229→   229→    return date.toLocaleDateString("en-IN", {
   230→   230→      year: "numeric",
   231→   231→      month: "short",
   232→   232→      day: "numeric",
   233→   233→    });
   234→   234→  } catch {
   235→   235→    return "Unknown";
   236→   236→  }
   237→   237→}

   239→   239→function isStaleBackup(iso: string | null, days = 7): boolean {
   240→   240→  if (!iso) return true;
   241→   241→  try {
   242→   242→    const date = new Date(iso);
   243→   243→    const diffMs = Date.now() - date.getTime();
   244→   244→    return diffMs > days * 24 * 60 * 60 * 1000;
   245→   245→  } catch {
   246→   246→    return true;
   247→   247→  }
   248→   248→}

   250→   250→function validateBackupFile(parsed: unknown): {
   251→   251→  valid: boolean;
   252→   252→  error?: string;
   253→   253→  file?: BackupFile;
   254→   254→} {
   255→   255→  if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
   256→   256→    return { valid: false, error: "Invalid format: expected a JSON object." };
   257→   257→  }
   258→   258→  const obj = parsed as Record<string, unknown>;
   259→   259→  const metadata = obj.metadata as Record<string, unknown> | undefined;
   260→   260→  const data = obj.data as Record<string, unknown> | undefined;
   261→   261→  if (!metadata || typeof metadata !== "object") {
   262→   262→    return { valid: false, error: 'Missing "metadata" object.' };
   263→   263→  }
   264→   264→  if (!data || typeof data !== "object") {
   265→   265→    return { valid: false, error: 'Missing "data" object.' };
   266→   266→  }
   267→   267→  // At least one collection must be an array
   268→   268→  const collections = ["ingredients", "recipes", "stockMovements", "dailyMeals", "purchases", "expenses"];
   269→   269→  const hasAny = collections.some((k) => Array.isArray((data as Record<string, unknown>)[k]));
   270→   270→  if (!hasAny) {
   271→   271→    return { valid: false, error: "Backup contains no recognizable data collections." };
   272→   272→  }
   273→   273→  return {
   274→   274→    valid: true,
   275→   275→    file: {
   276→   276→      metadata: metadata as unknown as BackupMetadata,
   277→   277→      data,
   278→   278→    },
   279→   279→  };
   280→   280→}

   282→   282→// ─── Circular Budget Gauge ───────────────────────────────────────────────────

   284→   284→function BudgetGauge({ percent, label, spent, budget }: {
   285→   285→  percent: number;
   286→   286→  label: string;
   287→   287→  spent: number;
   288→   288→  budget: number;
   289→   289→}) {
   290→   290→  const size = 140;
   291→   291→  const stroke = 12;
   292→   292→  const radius = (size - stroke) / 2;
   293→   293→  const circumference = 2 * Math.PI * radius;
   294→   294→  const clamped = Math.max(0, Math.min(percent, 150)); // allow up to 150% for visual
   295→   295→  const dashOffset = circumference - (Math.min(clamped, 100) / 100) * circumference;
   296→   296→  const color = getBudgetColor(percent);
   297→   297→
   298→   298→  return (
   299→   299→    <div className="flex flex-col items-center gap-2">
   300→   300→      <div
   301→   301→        className="relative"
   302→   302→        style={{ width: size, height: size }}
   303→   303→        role="meter"
   304→   304→        aria-valuenow={Math.round(percent)}
   305→   305→        aria-valuemin={0}
   306→   306→        aria-valuemax={100}
   307→   307→        aria-label={`${label} budget utilization`}
   308→   308→      >
   309→   309→        <svg width={size} height={size} className="-rotate-90">
   310→   310→          <circle
   311→   311→            cx={size / 2}
   312→   312→            cy={size / 2}
   313→   313→            r={radius}
   314→   314→            fill="none"
   315→   315→            stroke="var(--muted)"
   316→   316→            strokeWidth={stroke}
   317→   317→          />
   318→   318→          <circle
   319→   319→            cx={size / 2}
   320→   320→            cy={size / 2}
   321→   321→            r={radius}
   322→   322→            fill="none"
   323→   323→            stroke={color}
   324→   324→            strokeWidth={stroke}
   325→   325→            strokeLinecap="round"
   326→   326→            strokeDasharray={circumference}
   327→   327→            strokeDashoffset={dashOffset}
   328→   328→            style={{
   329→   329→              transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   330→   330→            }}
   331→   331→          />
   332→   332→        </svg>
   333→   333→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   334→   334→          <span
   335→   335→            className="text-2xl font-bold tabular-nums"
   336→   336→            style={{ color }}
   337→   337→          >
   338→   338→            {Math.round(percent)}%
   339→   339→          </span>
   340→   340→          <span className="text-[10px] font-medium text-muted-foreground">
   341→   341→            Used
   342→   342→          </span>
   343→   343→        </div>
   344→   344→      </div>
   345→   345→      <div className="text-center">
   346→   346→        <p className="text-sm font-semibold">{label}</p>
   347→   347→        <p className="text-xs text-muted-foreground tabular-nums">
   348→   348→          {formatINRShort(spent)} / {formatINRShort(budget)}
   349→   349→        </p>
   350→   350→      </div>
   351→   351→    </div>
   352→   352→  );
   353→   353→}
