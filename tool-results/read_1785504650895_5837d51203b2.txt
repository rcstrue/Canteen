   170→function formatINR(value: number): string {
   171→  const formatted = new Intl.NumberFormat("en-IN", {
   172→    minimumFractionDigits: 2,
   173→    maximumFractionDigits: 2,
   174→  }).format(value || 0);
   175→  return `₹${formatted}`;
   176→}
   177→
   178→function formatINRShort(value: number): string {
   179→  if (value >= 100000) return `₹${(value / 100000).toFixed(2)}L`;
   180→  if (value >= 1000) return `₹${(value / 1000).toFixed(1)}k`;
   181→  return formatINR(value);
   182→}

   204→function formatBytes(bytes: number): string {
   205→  if (bytes === 0) return "0 B";
   206→  const units = ["B", "KB", "MB", "GB"];
   207→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
   208→  const v = bytes / Math.pow(1024, i);
   209→  return `${v.toFixed(v < 10 && i > 0 ? 1 : 0)} ${units[i]}`;
   210→}

   212→function formatRelativeDate(iso: string | null): string {
   213→  if (!iso) return "Never";
   214→  try {
   215→    const date = new Date(iso);
   216→    const now = new Date();
   217→    const diffMs = now.getTime() - date.getTime();
   218→    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   219→    if (diffDays === 0) {
   220→      const diffHours = Math.floor(diffMs / (1000 * 60 * 60));
   221→      if (diffHours === 0) {
   222→        const diffMins = Math.floor(diffMs / (1000 * 60));
   223→        return diffMins <= 1 ? "Just now" : `${diffMins} minutes ago`;
   224→      }
   225→      return `${diffHours} hour${diffHours === 1 ? "" : "s"} ago`;
   226→    }
   227→    if (diffDays === 1) return "Yesterday";
   228→    if (diffDays < 7) return `${diffDays} days ago`;
   229→    return date.toLocaleDateString("en-IN", {
   230→      year: "numeric",
   231→      month: "short",
   232→      day: "numeric",
   233→    });
   234→  } catch {
   235→    return "Unknown";
   236→  }
   237→}
   238→
   239→function isStaleBackup(iso: string | null, days = 7): boolean {
   240→  if (!iso) return true;
   241→  try {
   242→    const date = new Date(iso);
   243→    const diffMs = Date.now() - date.getTime();
   244→    return diffMs > days * 24 * 60 * 60 * 1000;
   245→  } catch {
   246→    return true;
   247→  }
   248→}

   250→function validateBackupFile(parsed: unknown): {
   251→  valid: boolean;
   252→  error?: string;
   253→  file?: BackupFile;
   254→} {
   255→  if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
   256→    return { valid: false, error: "Invalid format: expected a JSON object." };
   257→  }
   258→  const obj = parsed as Record<string, unknown>;
   259→  const metadata = obj.metadata as Record<string, unknown> | undefined;
   260→  const data = obj.data as Record<string, unknown> | undefined;
   261→  if (!metadata || typeof metadata !== "object") {
   262→    return { valid: false, error: 'Missing "metadata" object.' };
   263→  }
   264→  if (!data || typeof data !== "object") {
   265→    return { valid: false, error: 'Missing "data" object.' };
   266→  }
   267→  // At least one collection must be an array
   268→  const collections = ["ingredients", "recipes", "stockMovements", "dailyMeals", "purchases", "expenses"];
   269→  const hasAny = collections.some((k) => Array.isArray((data as Record<string, unknown>)[k]));
   270→  if (!hasAny) {
   271→    return { valid: false, error: "Backup contains no recognizable data collections." };
   272→  }
   273→  return {
   274→    valid: true,
   275→    file: {
   276→      metadata: metadata as unknown as BackupMetadata,
   277→      data,
   278→    },
   279→  };
   280→}

   282→// ─── Circular Budget Gauge ───────────────────────────────────────────────────
   283→
   284→function BudgetGauge({ percent, label, spent, budget }: {
   285→  percent: number;
   286→  label: string;
   287→  spent: number;
   288→  budget: number;
   289→}) {
   290→  const size = 140;
   291→  const stroke = 12;
   292→  const radius = (size - stroke) / 2;
   293→  const circumference = 2 * Math.PI * radius;
   294→  const clamped = Math.max(0, Math.min(percent, 150)); // allow up to 150% for visual
   295→  const dashOffset = circumference - (Math.min(clamped, 100) / 100) * circumference;
   296→  const color = getBudgetColor(percent);
   297→
   298→  return (
   299→    <div className="flex flex-col items-center gap-2">
   300→      <div
   301→        className="relative"
   302→        style={{ width: size, height: size }}
   303→        role="meter"
   304→        aria-valuenow={Math.round(percent)}
   305→        aria-valuemin={0}
   306→        aria-valuemax={100}
   307→        aria-label={`${label} budget utilization`}
   308→      >
   309→        <svg width={size} height={size} className="-rotate-90">
   310→          <circle
   311→            cx={size / 2}
   312→            cy={size / 2}
   313→            r={radius}
   314→            fill="none"
   315→            stroke="var(--muted)"
   316→            strokeWidth={stroke}
   317→          />
   318→          <circle
   319→            cx={size / 2}
   320→            cy={size / 2}
   321→            r={radius}
   322→            fill="none"
   323→            stroke={color}
   324→            strokeWidth={stroke}
   325→            strokeLinecap="round"
   326→            strokeDasharray={circumference}
   327→            strokeDashoffset={dashOffset}
   328→            style={{
   329→              transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   330→            }}
   331→          />
   332→        </svg>
   333→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   334→          <span
   335→            className="text-2xl font-bold tabular-nums"
   336→            style={{ color }}
   337→          >
   338→            {Math.round(percent)}%
   339→          </span>
   340→          <span className="text-[10px] font-medium text-muted-foreground">
   341→            Used
   342→          </span>
   343→        </div>
   344→      </div>
   345→      <div className="text-center">
   346→        <p className="text-sm font-semibold">{label}</p>
   347→        <p className="text-xs text-muted-foreground tabular-nums">
   348→          {formatINRShort(spent)} / {formatINRShort(budget)}
   349→        </p>
   350→      </div>
   351→    </div>
   352→  );
   353→}
