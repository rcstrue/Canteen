   107→function getCategoryColor(category: string): string {
   108→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   109→  // Fallback: hash the category name to pick a consistent color
   110→  let hash = 0;
   111→  for (let i = 0; i < category.length; i++) {
   112→    hash = (hash << 5) - hash + category.charCodeAt(i);
   113→    hash |= 0;
   114→  }
   115→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   116→}

   276→interface TrendBadgeProps {
   277→  /** Percentage change (positive = increase, negative = decrease) */
   278→  pct: number | null;
   279→  /** For cost metrics, "lower is better" → up arrow = red. For meals/positive metrics, "up is better" */
   280→  lowerIsBetter?: boolean;
   281→  /** Comparison label e.g. "vs avg day" */
   282→  label: string;
   283→}
   284→
   285→function TrendBadge({ pct, lowerIsBetter = true, label }: TrendBadgeProps) {
   286→  if (pct === null || !isFinite(pct)) {
   287→    return (
   288→      <div className="flex items-center gap-1 text-xs text-muted-foreground">
   289→        <Minus className="h-3.5 w-3.5" />
   290→        <span>{label}</span>
   291→      </div>
   292→    );
   293→  }
   294→
   295→  const isUp = pct > 0;
   296→  const isFlat = Math.abs(pct) < 0.5;
   297→  const goodDirection = lowerIsBetter ? !isUp : isUp;
   298→  const colorClass = isFlat
   299→    ? "text-muted-foreground"
   300→    : goodDirection
   301→      ? "text-emerald-600 dark:text-emerald-400"
   302→      : "text-rose-600 dark:text-rose-400";
   303→  const Icon = isFlat ? Minus : isUp ? TrendingUp : TrendingDown;
   304→
   305→  return (
   306→    <div className="flex items-center gap-1.5 text-xs">
   307→      <Icon className={`h-4 w-4 ${colorClass}`} />
   308→      <span className={`font-bold tabular-nums ${colorClass}`}>
   309→        {isFlat ? "0%" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   310→      </span>
   311→      <span className="text-muted-foreground">{label}</span>
   312→    </div>
   313→  );
   314→}

   368→function CircularGauge({ percent }: { percent: number }) {
   369→  const size = 168;
   370→  const stroke = 14;
   371→  const radius = (size - stroke) / 2;
   372→  const circumference = 2 * Math.PI * radius;
   373→  const clamped = Math.max(0, Math.min(100, percent));
   374→  const dashOffset = circumference - (clamped / 100) * circumference;
   375→
   376→  // Color shifts: red (low) → amber (mid) → emerald (high)
   377→  const color =
   378→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   379→
   380→  return (
   381→    <div
   382→      className="relative"
   383→      style={{ width: size, height: size }}
   384→      role="meter"
   385→      aria-valuenow={Math.round(clamped)}
   386→      aria-valuemin={0}
   387→      aria-valuemax={100}
   388→      aria-label="Stock health percentage"
   389→    >
   390→      <svg width={size} height={size} className="-rotate-90">
   391→        <circle
   392→          cx={size / 2}
   393→          cy={size / 2}
   394→          r={radius}
   395→          fill="none"
   396→          stroke="var(--muted)"
   397→          strokeWidth={stroke}
   398→        />
   399→        <circle
   400→          cx={size / 2}
   401→          cy={size / 2}
   402→          r={radius}
   403→          fill="none"
   404→          stroke={color}
   405→          strokeWidth={stroke}
   406→          strokeLinecap="round"
   407→          strokeDasharray={circumference}
   408→          strokeDashoffset={dashOffset}
   409→          style={{
   410→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   411→          }}
   412→        />
   413→      </svg>
   414→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   415→        <span
   416→          className="text-3xl font-bold tabular-nums"
   417→          style={{ color }}
   418→        >
   419→          {Math.round(clamped)}%
   420→        </span>
   421→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   422→          Above Par
   423→        </span>
   424→      </div>
   425→    </div>
   426→  );
   427→}

   429→// ─── Donut Pie Label Renderer ───────────────────────────────────────────────
   430→
   431→/** Renders percentage labels INSIDE donut slices */
   432→function renderPiePercentLabel(props: {
   433→  cx: number;
   434→  cy: number;
   435→  midAngle: number;
   436→  innerRadius: number;
   437→  outerRadius: number;
   438→  percent: number;
   439→}) {
   440→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   441→  if (percent < 0.05) return null; // hide labels on slices < 5%
   442→
   443→  const RADIAN = Math.PI / 180;
   444→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   445→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   446→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   447→
   448→  return (
   449→    <text
   450→      x={x}
   451→      y={y}
   452→      fill="#fff"
   453→      textAnchor="middle"
   454→      dominantBaseline="central"
   455→      fontSize={12}
   456→      fontWeight={700}
   457→      style={{ pointerEvents: "none" }}
   458→    >
   459→      {(percent * 100).toFixed(0)}%
   460→    </text>
   461→  );
   462→}

   514→export function DashboardView({ onNavigate }: DashboardViewProps = {}) {

   522→  // Monthly comparison state

   524→  const [prevMonthReport, setPrevMonthReport] = useState<CostReportData | null>(null);
   525→
   526→  // Budget state

   529→  useEffect(() => {
   530→    async function fetchDashboard() {
   531→      try {
   532→        setLoading(true);
   533→        setError(null);
   534→        const [dashRes, ingRes] = await Promise.all([
   535→          fetch("/api/dashboard"),
   536→          fetch("/api/ingredients"),
   537→        ]);
   538→        if (!dashRes.ok) {
   539→          throw new Error(
   540→            `Failed to fetch dashboard data (${dashRes.status})`
   541→          );
   542→        }
   543→        const json = await dashRes.json();
   544→        setData(json);
   545→        // Ingredients fetch is best-effort (used for stock health gauge)
   546→        if (ingRes.ok) {
   547→          const ingJson = (await ingRes.json()) as IngredientListItem[];
   548→          setAllIngredients(ingJson);

   550→      } catch (err) {
   551→        console.error("Dashboard fetch error:", err);
   552→        setError(
   553→          err instanceof Error ? err.message : "Failed to load dashboard"
   554→        );
   555→      } finally {
   556→        setLoading(false);
   557→      }
   558→    }
   559→    fetchDashboard();
   560→  }, []);
   561→
   562→  // Fetch monthly comparison data
   563→  useEffect(() => {
   564→    async function fetchMonthlyComparison() {
   565→      try {
   566→        const now = new Date();
   567→        // Current month: first day to now
   568→        const curStart = new Date(now.getFullYear(), now.getMonth(), 1)
   569→          .toISOString()
   570→          .split("T")[0];
   571→        const curEnd = now.toISOString().split("T")[0];
   572→
   573→        // Previous month: first day to last day
   574→        const prevStart = new Date(now.getFullYear(), now.getMonth() - 1, 1)
   575→          .toISOString()
   576→          .split("T")[0];
   577→        const prevEnd = new Date(now.getFullYear(), now.getMonth(), 0)
   578→          .toISOString()
   579→          .split("T")[0];
   580→
   581→        const [curRes, prevRes] = await Promise.all([
   582→          fetch(
   583→            `/api/reports/cost?period=month&startDate=${curStart}&endDate=${curEnd}`
   584→          ),
   585→          fetch(
   586→            `/api/reports/cost?period=month&startDate=${prevStart}&endDate=${prevEnd}`
   587→          ),
   588→        ]);
   589→
   590→        if (curRes.ok) {
   591→          setCurrentMonthReport(await curRes.json());
   592→        }
   593→        if (prevRes.ok) {
   594→          setPrevMonthReport(await prevRes.json());
   595→        }
   596→      } catch (err) {
   597→        console.error("Monthly comparison fetch error:", err);
   598→      }
   599→    }
   600→    fetchMonthlyComparison();
   601→  }, []);
   602→
   603→  // Fetch current month's budget
   604→  useEffect(() => {
   605→    async function fetchBudget() {
   606→      try {
   607→        const res = await fetch("/api/budgets");
   608→        if (res.ok) {
   609→          const budgets = (await res.json()) as BudgetRecord[];
   610→          const currentMonth = new Date().toISOString().slice(0, 7);
   611→          const current = budgets.find((b) => b.month === currentMonth);
   612→          if (current) {
   613→            setCurrentBudget(current);
   614→          }
   615→        }
   616→      } catch (err) {
   617→        console.error("Budget fetch error:", err);
   618→      }
   619→    }
   620→    fetchBudget();
   621→  }, []);
   622→
