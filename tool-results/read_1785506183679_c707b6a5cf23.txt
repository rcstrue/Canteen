   107→   107→function getCategoryColor(category: string): string {
   108→   108→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   109→   109→  // Fallback: hash the category name to pick a consistent color
   110→   110→  let hash = 0;
   111→   111→  for (let i = 0; i < category.length; i++) {
   112→   112→    hash = (hash << 5) - hash + category.charCodeAt(i);
   113→   113→    hash |= 0;
   114→   114→  }
   115→   115→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   116→   116→}

   275→   275→
   276→   276→interface TrendBadgeProps {
   277→   277→  /** Percentage change (positive = increase, negative = decrease) */
   278→   278→  pct: number | null;
   279→   279→  /** For cost metrics, "lower is better" → up arrow = red. For meals/positive metrics, "up is better" */
   280→   280→  lowerIsBetter?: boolean;
   281→   281→  /** Comparison label e.g. "vs avg day" */
   282→   282→  label: string;
   283→   283→}
   284→   284→
   285→   285→function TrendBadge({ pct, lowerIsBetter = true, label }: TrendBadgeProps) {
   286→   286→  if (pct === null || !isFinite(pct)) {
   287→   287→    return (
   288→   288→      <div className="flex items-center gap-1 text-xs text-muted-foreground">
   289→   289→        <Minus className="h-3.5 w-3.5" />
   290→   290→        <span>{label}</span>
   291→   291→      </div>
   292→   292→    );
   293→   293→  }
   294→   294→
   295→   295→  const isUp = pct > 0;
   296→   296→  const isFlat = Math.abs(pct) < 0.5;
   297→   297→  const goodDirection = lowerIsBetter ? !isUp : isUp;
   298→   298→  const colorClass = isFlat
   299→   299→    ? "text-muted-foreground"
   300→   300→    : goodDirection
   301→   301→      ? "text-emerald-600 dark:text-emerald-400"
   302→   302→      : "text-rose-600 dark:text-rose-400";
   303→   303→  const Icon = isFlat ? Minus : isUp ? TrendingUp : TrendingDown;
   304→   304→
   305→   305→  return (
   306→   306→    <div className="flex items-center gap-1.5 text-xs">
   307→   307→      <Icon className={`h-4 w-4 ${colorClass}`} />
   308→   308→      <span className={`font-bold tabular-nums ${colorClass}`}>
   309→   309→        {isFlat ? "0%" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   310→   310→      </span>
   311→   311→      <span className="text-muted-foreground">{label}</span>
   312→   312→    </div>
   313→   313→  );
   314→   314→}

   368→   368→function CircularGauge({ percent }: { percent: number }) {
   369→   369→  const size = 168;
   370→   370→  const stroke = 14;
   371→   371→  const radius = (size - stroke) / 2;
   372→   372→  const circumference = 2 * Math.PI * radius;
   373→   373→  const clamped = Math.max(0, Math.min(100, percent));
   374→   374→  const dashOffset = circumference - (clamped / 100) * circumference;
   375→   375→
   376→   376→  // Color shifts: red (low) → amber (mid) → emerald (high)
   377→   377→  const color =
   378→   378→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   379→   379→
   380→   380→  return (
   381→   381→    <div
   382→   382→      className="relative"
   383→   383→      style={{ width: size, height: size }}
   384→   384→      role="meter"
   385→   385→      aria-valuenow={Math.round(clamped)}
   386→   386→      aria-valuemin={0}
   387→   387→      aria-valuemax={100}
   388→   388→      aria-label="Stock health percentage"
   389→   389→    >
   390→   390→      <svg width={size} height={size} className="-rotate-90">
   391→   391→        <circle
   392→   392→          cx={size / 2}
   393→   393→          cy={size / 2}
   394→   394→          r={radius}
   395→   395→          fill="none"
   396→   396→          stroke="var(--muted)"
   397→   397→          strokeWidth={stroke}
   398→   398→        />
   399→   399→        <circle
   400→   400→          cx={size / 2}
   401→   401→          cy={size / 2}
   402→   402→          r={radius}
   403→   403→          fill="none"
   404→   404→          stroke={color}
   405→   405→          strokeWidth={stroke}
   406→   406→          strokeLinecap="round"
   407→   407→          strokeDasharray={circumference}
   408→   408→          strokeDashoffset={dashOffset}
   409→   409→          style={{
   410→   410→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   411→   411→          }}
   412→   412→        />
   413→   413→      </svg>
   414→   414→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   415→   415→        <span
   416→   416→          className="text-3xl font-bold tabular-nums"
   417→   417→          style={{ color }}
   418→   418→        >
   419→   419→          {Math.round(clamped)}%
   420→   420→        </span>
   421→   421→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   422→   422→          Above Par
   423→   423→        </span>
   424→   424→      </div>
   425→   425→    </div>
   426→   426→  );
   427→   427→}

   431→   431→/** Renders percentage labels INSIDE donut slices */
   432→   432→function renderPiePercentLabel(props: {
   433→   433→  cx: number;
   434→   434→  cy: number;
   435→   435→  midAngle: number;
   436→   436→  innerRadius: number;
   437→   437→  outerRadius: number;
   438→   438→  percent: number;
   439→   439→}) {
   440→   440→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   441→   441→  if (percent < 0.05) return null; // hide labels on slices < 5%
   442→   442→
   443→   443→  const RADIAN = Math.PI / 180;
   444→   444→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   445→   445→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   446→   446→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   447→   447→
   448→   448→  return (
   449→   449→    <text
   450→   450→      x={x}
   451→   451→      y={y}
   452→   452→      fill="#fff"
   453→   453→      textAnchor="middle"
   454→   454→      dominantBaseline="central"
   455→   455→      fontSize={12}
   456→   456→      fontWeight={700}
   457→   457→      style={{ pointerEvents: "none" }}
   458→   458→    >
   459→   459→      {(percent * 100).toFixed(0)}%
   460→   460→    </text>
   461→   461→  );
   462→   462→}
