   180→function hash(s: string): number {
   181→  let h = 0;
   182→  for (let i = 0; i < s.length; i++) h = (h << 5) - h + s.charCodeAt(i);
   183→  return h;
   184→}

   186→/** Format as DD/MM/YYYY (Indian format). */
   187→function formatDate(dateStr: string): string {
   188→  if (!dateStr) return '';
   189→  const d = new Date(dateStr);
   190→  if (isNaN(d.getTime())) return dateStr;
   191→  const dd = String(d.getDate()).padStart(2, '0');
   192→  const mm = String(d.getMonth() + 1).padStart(2, '0');
   193→  const yyyy = d.getFullYear();
   194→  return `${dd}/${mm}/${yyyy}`;
   195→}
   196→
   197→/** Short format DD/MM for chart axes. */
   198→function formatShortDate(dateStr: string): string {
   199→  if (!dateStr) return '';
   200→  const d = new Date(dateStr);
   201→  if (isNaN(d.getTime())) return dateStr;
   202→  const dd = String(d.getDate()).padStart(2, '0');
   203→  const mm = String(d.getMonth() + 1).padStart(2, '0');
   204→  return `${dd}/${mm}`;
   205→}
   206→
   207→/** Compute the previous-period date range (same duration immediately before `start`). */
   208→function getPreviousRange(start: string, end: string): { start: string; end: string } {
   209→  const s = new Date(start);
   210→  const e = new Date(end);
   211→  const duration = e.getTime() - s.getTime();
   212→  const prevEnd = new Date(s.getTime() - 1); // day before current start
   213→  const prevStart = new Date(prevEnd.getTime() - duration);
   214→  return {
   215→    start: prevStart.toISOString().split('T')[0],
   216→    end: prevEnd.toISOString().split('T')[0],
   217→  };
   218→}

  1629→  value: string;
  1630→  icon: React.ReactNode;
  1631→  iconBg: string;
  1632→  iconColor: string;
  1633→  subtitle?: string;
  1634→  trend?: 'up' | 'down';
  1635→  previousValue?: string;
  1636→  goodWhenDown?: boolean;
  1637→}) {
  1638→  // Compute delta vs previous value
  1639→  const delta = useMemo(() => {
  1640→    if (!previousValue) return null;
  1641→    const currNum = parseFloat(value.replace(/[^0-9.-]/g, '')) || 0;
  1642→    const prevNum = parseFloat(previousValue.replace(/[^0-9.-]/g, '')) || 0;
  1643→    if (prevNum === 0) return null;
  1644→    return ((currNum - prevNum) / Math.abs(prevNum)) * 100;
  1645→  }, [value, previousValue]);
  1646→
  1647→  const deltaPositive = delta !== null && delta > 0;
  1648→  const deltaIsGood =
  1649→    delta !== null
  1650→      ? goodWhenDown
  1651→        ? !deltaPositive
  1652→        : deltaPositive
  1653→      : null;
  1654→
  1655→  return (
  1656→    <Card className="transition-all hover:shadow-md">
  1657→      <CardContent className="p-4">
  1658→        <div className="flex items-center justify-between">
  1659→          <div className={`flex h-10 w-10 items-center justify-center rounded-lg ${iconBg}`}>
  1660→            <span className={iconColor}>{icon}</span>
  1661→          </div>
  1662→          {trend && (
  1663→            <span
  1664→              className={
  1665→                trend === 'up'
  1666→                  ? 'flex items-center text-rose-500 text-xs font-medium'
  1667→                  : 'flex items-center text-emerald-500 text-xs font-medium'
  1668→              }
  1669→            >
  1670→              {trend === 'up' ? (
  1671→                <ArrowUpRight className="h-3.5 w-3.5" />
  1672→              ) : (
  1673→                <ArrowDownRight className="h-3.5 w-3.5" />
  1674→              )}
  1675→            </span>
  1676→          )}
  1677→        </div>
  1678→        <div className="mt-3">
  1679→          <p className="text-sm text-muted-foreground">{title}</p>
  1680→          <p className="text-2xl font-bold tabular-nums">{value}</p>
  1681→          {previousValue && (
  1682→            <p className="text-xs text-muted-foreground mt-0.5">
  1683→              Prev: <span className="tabular-nums">{previousValue}</span>
  1684→              {delta !== null && (
  1685→                <span
  1686→                  className={`ml-1 font-medium ${
  1687→                    deltaIsGood ? 'text-emerald-600 dark:text-emerald-400' : 'text-rose-600 dark:text-rose-400'
  1688→                  }`}
  1689→                >
  1690→                  ({delta >= 0 ? '+' : ''}{delta.toFixed(1)}%)
  1691→                </span>
  1692→              )}
  1693→            </p>
  1694→          )}
  1695→          {subtitle && !previousValue && (
  1696→            <p className="text-xs text-muted-foreground mt-1">{subtitle}</p>
  1697→          )}
  1698→        </div>
  1699→      </CardContent>
  1700→    </Card>
  1701→  );
  1702→}

  1842→  cx,
  1843→  cy,
  1844→  midAngle,
  1845→  innerRadius,

  1869→      dominantBaseline="central"
