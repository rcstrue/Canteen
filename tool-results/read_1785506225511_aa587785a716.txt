   122→// ─── Helpers ─────────────────────────────────────────────────────────────────

   124→function formatCurrency(amount: number): string {
   125→  return new Intl.NumberFormat("en-IN", {
   126→    style: "currency",
   127→    currency: "INR",
   128→    minimumFractionDigits: 2,
   129→    maximumFractionDigits: 2,
   130→  }).format(amount);
   131→}
   132→
   133→function formatDate(dateStr: string): string {
   134→  const d = new Date(dateStr);
   135→  const day = String(d.getDate()).padStart(2, "0");
   136→  const month = String(d.getMonth() + 1).padStart(2, "0");
   137→  const year = d.getFullYear();
   138→  return `${day}/${month}/${year}`;
   139→}
   140→
   141→function getTodayStr(): string {
   142→  const d = new Date();
   143→  return d.toISOString().split("T")[0];
   144→}
   145→
   146→// Purchase status based on date (simulated since no status field in DB)
   147→type PurchaseStatus = "Pending" | "Received" | "Paid";
   148→
   149→function getPurchaseStatus(purchase: Purchase): PurchaseStatus {
   150→  const purchaseDate = new Date(purchase.date);
   151→  const today = new Date();
   152→  today.setHours(0, 0, 0, 0);
   153→  const diffDays = Math.floor((today.getTime() - purchaseDate.getTime()) / (1000 * 60 * 60 * 24));
   154→  if (diffDays <= 0) return "Pending";
   155→  if (diffDays <= 7) return "Received";
   156→  return "Paid";
   157→}
   158→

   184→}
