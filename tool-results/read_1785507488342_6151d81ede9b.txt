   210→function formatRupee(amount: number): string {
   211→  return (
   212→    "₹" +
   213→    amount.toLocaleString("en-IN", {
   214→      minimumFractionDigits: 2,
   215→      maximumFractionDigits: 2,
   216→    })
   217→  );
   218→}
   219→
   220→function calcTotalIngredientCost(ingredients: RecipeIngredient[]): number {
   221→  return ingredients.reduce((sum, ri) => sum + ri.quantity * ri.ingredient.avgCost, 0);
   222→}
   223→
   224→function calcCostPerMeal(ingredients: RecipeIngredient[], baseServings: number): number {
   225→  if (baseServings <= 0) return 0;
   226→  return calcTotalIngredientCost(ingredients) / baseServings;
   227→}
   228→
   229→function calcCostFor600(ingredients: RecipeIngredient[], baseServings: number): number {
   230→  return calcCostPerMeal(ingredients, baseServings) * 600;
   231→}
   232→
   233→function calcCostBreakdown(ingredients: RecipeIngredient[]): {
   234→  entries: CategoryBreakdownEntry[];
   235→  total: number;
   236→} {
   237→  const byCategory: Record<string, number> = {};
   238→  let total = 0;
   239→  ingredients.forEach((ri) => {
   240→    const cost = ri.quantity * ri.ingredient.avgCost;
   241→    const cat = ri.ingredient.category || "Other";
   242→    byCategory[cat] = (byCategory[cat] || 0) + cost;
   243→    total += cost;
   244→  });
   245→  const entries = Object.entries(byCategory)
   246→    .map(([category, amount]) => ({
   247→      category,
   248→      amount,
   249→      percent: total > 0 ? (amount / total) * 100 : 0,
   250→    }))
   251→    .sort((a, b) => b.amount - a.amount);
   252→  return { entries, total };
   253→}
   254→
   255→// Compare cost using lastPurchasePrice vs avgCost to show trend
   256→function calcCostTrend(ingredients: RecipeIngredient[]): {
   257→  direction: "up" | "down" | "stable";
   258→  percentChange: number;
   259→} {
   260→  let costByAvg = 0;
   261→  let costByLast = 0;
   262→  ingredients.forEach((ri) => {
   263→    costByAvg += ri.quantity * ri.ingredient.avgCost;
   264→    costByLast += ri.quantity * ri.ingredient.lastPurchasePrice;
   265→  });
   266→  if (costByAvg === 0) return { direction: "stable", percentChange: 0 };
   267→  const pctChange = ((costByLast - costByAvg) / costByAvg) * 100;
   268→  if (Math.abs(pctChange) < 2) return { direction: "stable", percentChange: pctChange };
   269→  return { direction: pctChange > 0 ? "up" : "down", percentChange: pctChange };
   270→}
