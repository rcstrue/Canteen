   110→
   111→function getCategoryColor(category: string): string {
   112→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   113→  // Fallback: hash the category name to pick a consistent color
   114→  let hash = 0;
   115→  for (let i = 0; i < category.length; i++) {
   116→    hash = (hash << 5) - hash + category.charCodeAt(i);
   117→    hash |= 0;
   118→  }
   119→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   120→}

   220→const LOW_STOCK_BANNER_DISMISS_KEY = "rcs-dashboard-lowstock-dismissed";
   221→const LOW_STOCK_BANNER_UPDATE_EVENT = "rcs-lowstock-update";
   222→
   223→/** SSR-safe subscription to a session-storage dismissal flag. Returns
   224→ *  `[isDismissed, dismiss]` — calling `dismiss` writes to sessionStorage and
   225→ *  notifies subscribers so the component re-renders without calling setState
   226→ *  synchronously inside an effect. */
   227→function useLowStockDismissed(): [boolean, () => void] {
   228→  const subscribe = useCallback((callback: () => void) => {
   229→    if (typeof window === "undefined") return () => {};
   230→    const handler = () => callback();
   231→    window.addEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   232→    window.addEventListener("storage", handler);
   233→    return () => {
   234→      window.removeEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   235→      window.removeEventListener("storage", handler);
   236→    };
   237→  }, []);
   238→
   239→  const getSnapshot = useCallback(() => {
   240→    try {
   241→      return window.sessionStorage.getItem(LOW_STOCK_BANNER_DISMISS_KEY) === "true";
   242→    } catch {
   243→      return false;
   244→    }
   245→  }, []);
   246→
   247→  const getServerSnapshot = useCallback(() => false, []);
   248→
   249→  const isDismissed = useSyncExternalStore(
   250→    subscribe,
   251→    getSnapshot,
   252→    getServerSnapshot
   253→  );
   254→
   255→  const dismiss = useCallback(() => {
   256→    try {
   257→      window.sessionStorage.setItem(LOW_STOCK_BANNER_DISMISS_KEY, "true");
   258→      window.dispatchEvent(new Event(LOW_STOCK_BANNER_UPDATE_EVENT));
   259→    } catch {
   260→      // Ignore write failure (private mode, etc.).
   261→    }
   262→  }, []);
   263→
   264→  return [isDismissed, dismiss];
   265→}

   826→function CircularGauge({ percent }: { percent: number }) {
   827→  const size = 168;
   828→  const stroke = 14;
   829→  const radius = (size - stroke) / 2;
   830→  const circumference = 2 * Math.PI * radius;
   831→  const clamped = Math.max(0, Math.min(100, percent));
   832→  const dashOffset = circumference - (clamped / 100) * circumference;
   833→
   834→  // Color shifts: red (low) → amber (mid) → emerald (high)
   835→  const color =
   836→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   837→
   838→  return (
   839→    <div
   840→      className="relative"
   841→      style={{ width: size, height: size }}
   842→      role="meter"
   843→      aria-valuenow={Math.round(clamped)}
   844→      aria-valuemin={0}
   845→      aria-valuemax={100}
   846→      aria-label="Stock health percentage"
   847→    >
   848→      <svg width={size} height={size} className="-rotate-90">
   849→        <circle
   850→          cx={size / 2}
   851→          cy={size / 2}
   852→          r={radius}
   853→          fill="none"
   854→          stroke="var(--muted)"
   855→          strokeWidth={stroke}
   856→        />
   857→        <circle
   858→          cx={size / 2}
   859→          cy={size / 2}
   860→          r={radius}
   861→          fill="none"
   862→          stroke={color}
   863→          strokeWidth={stroke}
   864→          strokeLinecap="round"
   865→          strokeDasharray={circumference}
   866→          strokeDashoffset={dashOffset}
   867→          style={{
   868→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   869→          }}
   870→        />
   871→      </svg>
   872→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   873→        <span
   874→          className="text-3xl font-bold tabular-nums"
   875→          style={{ color }}
   876→        >
   877→          {Math.round(clamped)}%
   878→        </span>
   879→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   880→          Above Par
   881→        </span>
   882→      </div>
   883→    </div>
   884→  );
   885→}

   887→// ─── Donut Pie Label Renderer ───────────────────────────────────────────────

   889→/** Renders percentage labels INSIDE donut slices */
   890→function renderPiePercentLabel(props: {
   891→  cx: number;
   892→  cy: number;
   893→  midAngle: number;
   894→  innerRadius: number;
   895→  outerRadius: number;
   896→  percent: number;
   897→}) {
   898→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   899→  if (percent < 0.05) return null; // hide labels on slices < 5%
   900→
   901→  const RADIAN = Math.PI / 180;
   902→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   903→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   904→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   905→
   906→  return (
   907→    <text
   908→      x={x}
   909→      y={y}
   910→      fill="#fff"
   911→      textAnchor="middle"
   912→      dominantBaseline="central"
   913→      fontSize={12}
   914→      fontWeight={700}
   915→      style={{ pointerEvents: "none" }}
   916→    >
   917→      {(percent * 100).toFixed(0)}%
   918→    </text>
   919→  );
   920→}
