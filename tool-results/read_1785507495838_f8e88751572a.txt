   220→   220→   220→function calcTotalIngredientCost(ingredients: RecipeIngredient[]): number {
   221→   221→   221→  return ingredients.reduce((sum, ri) => sum + ri.quantity * ri.ingredient.avgCost, 0);
   222→   222→   222→}
   223→   223→   223→
   224→   224→   224→function calcCostPerMeal(ingredients: RecipeIngredient[], baseServings: number): number {
   225→   225→   225→  if (baseServings <= 0) return 0;
   226→   226→   226→  return calcTotalIngredientCost(ingredients) / baseServings;
   227→   227→   227→}
   228→   228→   228→
   229→   229→   229→function calcCostFor600(ingredients: RecipeIngredient[], baseServings: number): number {
   230→   230→   230→  return calcCostPerMeal(ingredients, baseServings) * 600;
   231→   231→   231→}
   232→   232→   232→
   233→   233→   233→function calcCostBreakdown(ingredients: RecipeIngredient[]): {
   234→   234→   234→  entries: CategoryBreakdownEntry[];
   235→   235→   235→  total: number;
   236→   236→   236→} {
   237→   237→   237→  const byCategory: Record<string, number> = {};
   238→   238→   238→  let total = 0;
   239→   239→   239→  ingredients.forEach((ri) => {
   240→   240→   240→    const cost = ri.quantity * ri.ingredient.avgCost;
   241→   241→   241→    const cat = ri.ingredient.category || "Other";
   242→   242→   242→    byCategory[cat] = (byCategory[cat] || 0) + cost;
   243→   243→   243→    total += cost;
   244→   244→   244→  });
   245→   245→   245→  const entries = Object.entries(byCategory)
   246→   246→   246→    .map(([category, amount]) => ({
   247→   247→   247→      category,
   248→   248→   248→      amount,
   249→   249→   249→      percent: total > 0 ? (amount / total) * 100 : 0,
   250→   250→   250→    }))
   251→   251→   251→    .sort((a, b) => b.amount - a.amount);
   252→   252→   252→  return { entries, total };
   253→   253→   253→}
   254→   254→   254→
   255→   255→   255→// Compare cost using lastPurchasePrice vs avgCost to show trend
   256→   256→   256→function calcCostTrend(ingredients: RecipeIngredient[]): {
   257→   257→   257→  direction: "up" | "down" | "stable";
   258→   258→   258→  percentChange: number;
   259→   259→   259→} {
   260→   260→   260→  let costByAvg = 0;
   261→   261→   261→  let costByLast = 0;
   262→   262→   262→  ingredients.forEach((ri) => {
   263→   263→   263→    costByAvg += ri.quantity * ri.ingredient.avgCost;
   264→   264→   264→    costByLast += ri.quantity * ri.ingredient.lastPurchasePrice;
   265→   265→   265→  });
   266→   266→   266→  if (costByAvg === 0) return { direction: "stable", percentChange: 0 };
   267→   267→   267→  const pctChange = ((costByLast - costByAvg) / costByAvg) * 100;
   268→   268→   268→  if (Math.abs(pctChange) < 2) return { direction: "stable", percentChange: pctChange };
   269→   269→   269→  return { direction: pctChange > 0 ? "up" : "down", percentChange: pctChange };
   270→   270→   270→}
