   126→function formatCurrency(amount: number): string {
   127→  return new Intl.NumberFormat("en-IN", {
   128→    style: "currency",
   129→    currency: "INR",
   130→    minimumFractionDigits: 2,
   131→    maximumFractionDigits: 2,
   132→  }).format(amount);
   133→}
   134→
   135→function formatDate(dateStr: string): string {
   136→  const d = new Date(dateStr);
   137→  const day = String(d.getDate()).padStart(2, "0");
   138→  const month = String(d.getMonth() + 1).padStart(2, "0");
   139→  const year = d.getFullYear();
   140→  return `${day}/${month}/${year}`;
   141→}
   142→
   143→function getTodayStr(): string {
   144→  const d = new Date();
   145→  return d.toISOString().split("T")[0];
   146→}
   147→
   148→/** Build a friendly PUR-XXXX invoice number from a purchase id. */
   149→function getInvoiceNumber(purchase: Purchase): string {
   150→  if (purchase.invoiceNo && purchase.invoiceNo.trim()) {
   151→    return purchase.invoiceNo.trim();
   152→  }
   153→  // Derive a deterministic short id from the purchase id
   154→  const short = purchase.id.replace(/[^a-z0-9]/gi, "").slice(-6).toUpperCase();
   155→  return `PUR-${short || "0000"}`;
   156→}
   157→
   158→// Purchase status based on date (simulated since no status field in DB)
   159→type PurchaseStatus = "Pending" | "Received" | "Paid";
   160→
   161→function getPurchaseStatus(purchase: Purchase): PurchaseStatus {
   162→  const purchaseDate = new Date(purchase.date);
   163→  const today = new Date();
   164→  today.setHours(0, 0, 0, 0);
   165→  const diffDays = Math.floor((today.getTime() - purchaseDate.getTime()) / (1000 * 60 * 60 * 24));
   166→  if (diffDays <= 0) return "Pending";
   167→  if (diffDays <= 7) return "Received";
   168→  return "Paid";
   169→}
