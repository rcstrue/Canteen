   138→function formatCurrency(amount: number): string {
   139→  return new Intl.NumberFormat("en-IN", {
   140→    style: "currency",
   141→    currency: "INR",
   142→    minimumFractionDigits: 2,
   143→    maximumFractionDigits: 2,
   144→  }).format(amount);
   145→}
   146→
   147→function formatCurrencyShort(amount: number): string {
   148→  if (amount >= 10000000) return `₹${(amount / 10000000).toFixed(2)}Cr`;
   149→  if (amount >= 100000) return `₹${(amount / 100000).toFixed(2)}L`;
   150→  if (amount >= 1000) return `₹${(amount / 1000).toFixed(1)}k`;
   151→  return formatCurrency(amount);
   152→}
   153→
   154→function formatDate(dateStr: string): string {
   155→  const d = new Date(dateStr);
   156→  const day = String(d.getDate()).padStart(2, "0");
   157→  const month = String(d.getMonth() + 1).padStart(2, "0");
   158→  const year = d.getFullYear();
   159→  return `${day}/${month}/${year}`;
   160→}

   212→  const [detailOpen, setDetailOpen] = useState(false);

   215→  const [selectedDetail, setSelectedDetail] = useState<SupplierDetail | null>(null);

   234→    setLoading(true);
   235→    try {
   236→      const params = new URLSearchParams();
   237→      if (search) params.set("search", search);
   238→      if (categoryFilter && categoryFilter !== "All") params.set("category", categoryFilter);
   239→

   242→        const data = await res.json();
   243→        setSuppliers(Array.isArray(data) ? data : []);
   244→      } else {
   245→        toast({
   246→          title: "Error",
   247→          description: "Failed to fetch suppliers",
   248→          variant: "destructive",
   249→        });
   250→      }
   251→    } catch (err) {
   252→      console.error("Error fetching suppliers:", err);
   253→      toast({
   254→        title: "Error",
   255→        description: "Failed to fetch suppliers",
   256→        variant: "destructive",
   257→      });

   259→      setLoading(false);
   260→    }

   264→    fetchSuppliers();
   265→  }, [fetchSuppliers]);

   270→    arr.sort((a, b) => {
   271→      let cmp = 0;

   273→        case "name":
   274→          cmp = a.name.localeCompare(b.name);
   275→          break;
   276→        case "category":
   277→          cmp = (a.category || "").localeCompare(b.category || "");
   278→          break;
   279→        case "ingredientCount":
   280→          cmp = a.ingredientCount - b.ingredientCount;
   281→          break;
   282→        case "totalPurchaseValue":
   283→          cmp = a.totalPurchaseValue - b.totalPurchaseValue;
   284→          break;
   285→        case "createdAt":
   286→          cmp = new Date(a.createdAt).getTime() - new Date(b.createdAt).getTime();
   287→          break;
   288→      }
   289→      return sortDir === "asc" ? cmp : -cmp;
   290→    });
   291→    return arr;
   292→  }, [suppliers, sortField, sortDir]);

   295→    if (sortField === field) {
   296→      setSortDir(sortDir === "asc" ? "desc" : "asc");

   298→      setSortField(field);
   299→      setSortDir("asc");
   300→    }
   301→  };
   302→
   303→  const SortIcon = ({ field }: { field: SortField }) => {

   305→    return sortDir === "asc" ? (
   306→      <ChevronUp className="ml-1 inline h-3 w-3" />

   308→      <ChevronDown className="ml-1 inline h-3 w-3" />
   309→    );
   310→  };
   311→
   312→  // ── Summary stats ───────────────────────────────────────────────────────
   313→  const summary = useMemo(() => {
   314→    const total = suppliers.length;
   315→    const totalPurchaseValue = suppliers.reduce(
   316→      (sum, s) => sum + s.totalPurchaseValue,
   317→      0
   318→    );
   319→    // Active = supplier with at least 1 ingredient OR 1 purchase
   320→    const active = suppliers.filter(
   321→      (s) => s.ingredientCount > 0 || s.purchaseCount > 0
   322→    ).length;
   323→    return { total, totalPurchaseValue, active };
   324→  }, [suppliers]);
   325→
   326→  // ── Form actions ────────────────────────────────────────────────────────

   329→    setFormErrors({});
   330→    setEditingId(null);
   331→    setFormOpen(true);
   332→  };
   333→

   336→      name: supplier.name,
   337→      contactPerson: supplier.contactPerson || "",
   338→      phone: supplier.phone || "",
   339→      email: supplier.email || "",
   340→      address: supplier.address || "",
   341→      gstin: supplier.gstin || "",
   342→      category: supplier.category || "",
   343→      notes: supplier.notes || "",
   344→    });
   345→    setFormErrors({});
   346→    setEditingId(supplier.id);
   347→    setFormOpen(true);
   348→  };
   349→
   350→  const validateForm = (): boolean => {
   351→    const errs: Partial<Record<keyof SupplierFormValues, string>> = {};
   352→    if (!form.name.trim()) errs.name = "Supplier name is required";
   353→    if (form.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) {

   355→    }
   356→    if (form.gstin && form.gstin.trim().length !== 15) {
   357→      errs.gstin = "GSTIN must be 15 characters";
   358→    }
   359→    setFormErrors(errs);
   360→    return Object.keys(errs).length === 0;
   361→  };
   362→
   363→  const handleSubmit = async () => {
   364→    if (!validateForm()) return;
   365→    setSubmitting(true);
   366→    try {
   367→      const payload = {
   368→        name: form.name.trim(),
   369→        contactPerson: form.contactPerson.trim() || null,
   370→        phone: form.phone.trim() || null,
   371→        email: form.email.trim() || null,
   372→        address: form.address.trim() || null,
   373→        gstin: form.gstin.trim().toUpperCase() || null,
   374→        category: form.category.trim() || null,
   375→        notes: form.notes.trim() || null,
   376→      };
   377→
   378→      const url = editingId
   379→        ? `/api/suppliers/${editingId}`
   380→        : `/api/suppliers`;
   381→      const method = editingId ? "PUT" : "POST";
   382→
   383→      const res = await fetch(url, {
   384→        method,
   385→        headers: { "Content-Type": "application/json" },
   386→        body: JSON.stringify(payload),
   387→      });
   388→
   389→      if (res.ok) {
   390→        toast({
   391→          title: editingId ? "Supplier updated" : "Supplier created",
   392→          description: `${payload.name} has been ${editingId ? "updated" : "added"} successfully.`,
   393→        });
   394→        setFormOpen(false);
   395→        await fetchSuppliers();
   396→      } else {
   397→        const data = await res.json();
   398→        toast({
   399→          title: "Failed",
   400→          description: data.error || "Operation failed",
   401→          variant: "destructive",
   402→        });
   403→      }
   404→    } catch (err) {
   405→      console.error("Error submitting supplier:", err);
   406→      toast({
   407→        title: "Error",
   408→        description: "Failed to save supplier",
   409→        variant: "destructive",
   410→      });
   411→    } finally {
   412→      setSubmitting(false);
   413→    }
   414→  };
   415→
   416→  // ── Detail view ──────────────────────────────────────────────────────────
   417→  const handleViewDetail = async (supplier: Supplier) => {
   418→    setDetailLoading(true);
   419→    setDetailOpen(true);
   420→    try {
   421→      const res = await fetch(`/api/suppliers/${supplier.id}`);
   422→      if (res.ok) {
   423→        const data = await res.json();
   424→        setSelectedDetail(data);
   425→      } else {
   426→        toast({
   427→          title: "Error",
   428→          description: "Failed to load supplier details",
   429→          variant: "destructive",
   430→        });
   431→      }
   432→    } catch (err) {
   433→      console.error("Error fetching supplier detail:", err);
   434→    } finally {
   435→      setDetailLoading(false);
   436→    }
   437→  };

   439→  // ── Delete ──────────────────────────────────────────────────────────────
   440→  const confirmDelete = (id: string) => {
   441→    setDeleteId(id);
   442→    setDeleteOpen(true);
   443→  };

   445→  const handleDelete = async () => {
   446→    if (!deleteId) return;
   447→    setDeleting(true);

   449→      const res = await fetch(`/api/suppliers/${deleteId}`, {
   450→        method: "DELETE",

   454→          title: "Supplier deleted",
   455→          description: "The supplier has been removed. Linked items were detached.",

   458→        setDeleteId(null);
   459→        await fetchSuppliers();
   460→      } else {
   461→        const data = await res.json();
   462→        toast({
   463→          title: "Failed",
   464→          description: data.error || "Failed to delete supplier",
   465→          variant: "destructive",
   466→        });
   467→      }
   468→    } catch (err) {
   469→      console.error("Error deleting supplier:", err);
   470→      toast({
   471→        title: "Error",
   472→        description: "Failed to delete supplier",
   473→        variant: "destructive",
   474→      });
   475→    } finally {
   476→      setDeleting(false);
   477→    }
   478→  };

   484→        title: "Nothing to export",

   491→      Name: s.name,
   492→      "Contact Person": s.contactPerson || "",
   493→      Phone: s.phone || "",
   494→      Email: s.email || "",
   495→      Address: s.address || "",
   496→      GSTIN: s.gstin || "",
   497→      Category: s.category || "",
   498→      "Ingredient Count": String(s.ingredientCount),
   499→      "Purchase Count": String(s.purchaseCount),
   500→      "Total Purchase Value": String(s.totalPurchaseValue),
   501→      "Created At": formatDate(s.createdAt),

   504→    downloadCSV(`suppliers-${today}.csv`, rows);

   506→      title: "Export successful",

   508→    });
   509→  };
   510→
   511→  // ── Render ───────────────────────────────────────────────────────────────

   515→      {/* Header */}
   516→      <div className="flex flex-col gap-4 sm:flex-row sm:items-center sm:justify-between">

   538→          <Button
   539→            onClick={openAddForm}

   541→          >
   542→            <Plus className="h-4 w-4" />
   543→            <span className="hidden sm:inline">Add Supplier</span>
   544→            <span className="sm:hidden">Add</span>
   545→          </Button>
   546→        </div>
   547→      </div>
   548→
   549→      {/* Summary Cards */}
   550→      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   551→        <Card className="card-hover border-amber-200/60 dark:border-amber-800/30">
   552→          <CardContent className="flex items-center gap-4 p-4">
   553→            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-lg bg-amber-100 dark:bg-amber-900/30">
   554→              <Building2 className="h-6 w-6 text-amber-600 dark:text-amber-400" />
   555→            </div>

   560→          </CardContent>
   561→        </Card>
   562→
   563→        <Card className="card-hover border-amber-200/60 dark:border-amber-800/30">
   564→          <CardContent className="flex items-center gap-4 p-4">
   565→            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-lg bg-emerald-100 dark:bg-emerald-900/30">
   566→              <IndianRupee className="h-6 w-6 text-emerald-600 dark:text-emerald-400" />
   567→            </div>
   568→            <div className="min-w-0">
   569→              <p className="text-2xl font-bold">{formatCurrencyShort(summary.totalPurchaseValue)}</p>
   570→              <p className="text-xs text-muted-foreground">Total Purchase Value</p>
   571→            </div>
   572→          </CardContent>
   573→        </Card>
   574→
   575→        <Card className="card-hover border-amber-200/60 dark:border-amber-800/30">
   576→          <CardContent className="flex items-center gap-4 p-4">
   577→            <div className="flex h-12 w-12 shrink-0 items-center justify-center rounded-lg bg-green-100 dark:bg-green-900/30">
   578→              <Package className="h-6 w-6 text-green-600 dark:text-green-400" />

   581→              <p className="text-2xl font-bold">{summary.active}</p>

  1300→    </div>
