   262→function isLowStock(item: Ingredient): boolean {
   263→  return item.currentStock < item.minStock;
   264→}
   265→
   266→type StockHealth = "OK" | "LOW" | "CRITICAL";
   267→
   268→function getStockHealth(item: Ingredient): StockHealth {
   269→  if (item.minStock === 0) return "OK";
   270→  const ratio = item.currentStock / item.minStock;
   271→  if (ratio < 1) return "CRITICAL";    // currentStock < minStock → red
   272→  if (ratio < 1.2) return "LOW";       // 80-100% of minStock (near par) → amber
   273→  return "OK";                          // currentStock >= minStock → green
   274→}

   311→function formatDateDDMMYYYY(dateStr: string): string {
   312→  if (!dateStr) return "—";
   313→  const d = new Date(dateStr);
   314→  if (isNaN(d.getTime())) return "—";
   315→  const dd = String(d.getDate()).padStart(2, "0");
   316→  const mm = String(d.getMonth() + 1).padStart(2, "0");
   317→  const yyyy = d.getFullYear();
   318→  return `${dd}/${mm}/${yyyy}`;
   319→}
   320→
   321→function formatINR(value: number): string {
   322→  const formatted = new Intl.NumberFormat("en-IN", {
   323→    minimumFractionDigits: 2,
   324→    maximumFractionDigits: 2,
   325→  }).format(value || 0);
   326→  return `₹${formatted}`;
   327→}
