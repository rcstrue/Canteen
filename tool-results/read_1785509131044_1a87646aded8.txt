   218→function formatINR(value: number): string {
   219→  const formatted = new Intl.NumberFormat("en-IN", {
   220→    minimumFractionDigits: 2,
   221→    maximumFractionDigits: 2,
   222→  }).format(value || 0);
   223→  return `₹${formatted}`;
   224→}
   225→
   226→function formatINRShort(value: number): string {
   227→  if (value >= 100000) return `₹${(value / 100000).toFixed(2)}L`;
   228→  if (value >= 1000) return `₹${(value / 1000).toFixed(1)}k`;
   229→  return formatINR(value);
   230→}

   252→function formatBytes(bytes: number): string {
   253→  if (bytes === 0) return "0 B";
   254→  const units = ["B", "KB", "MB", "GB"];
   255→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
   256→  const v = bytes / Math.pow(1024, i);
   257→  return `${v.toFixed(v < 10 && i > 0 ? 1 : 0)} ${units[i]}`;
   258→}

   260→function formatRelativeDate(iso: string | null): string {
   261→  if (!iso) return "Never";
   262→  try {
   263→    const date = new Date(iso);
   264→    const now = new Date();
   265→    const diffMs = now.getTime() - date.getTime();
   266→    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   267→    if (diffDays === 0) {
   268→      const diffHours = Math.floor(diffMs / (1000 * 60 * 60));
   269→      if (diffHours === 0) {
   270→        const diffMins = Math.floor(diffMs / (1000 * 60));
   271→        return diffMins <= 1 ? "Just now" : `${diffMins} minutes ago`;
   272→      }
   273→      return `${diffHours} hour${diffHours === 1 ? "" : "s"} ago`;
   274→    }
   275→    if (diffDays === 1) return "Yesterday";
   276→    if (diffDays < 7) return `${diffDays} days ago`;
   277→    return date.toLocaleDateString("en-IN", {
   278→      year: "numeric",
   279→      month: "short",
   280→      day: "numeric",
   281→    });
   282→  } catch {
   283→    return "Unknown";
   284→  }
   285→}
   286→
   287→function isStaleBackup(iso: string | null, days = 7): boolean {
   288→  if (!iso) return true;
   289→  try {
   290→    const date = new Date(iso);
   291→    const diffMs = Date.now() - date.getTime();
   292→    return diffMs > days * 24 * 60 * 60 * 1000;
   293→  } catch {
   294→    return true;
   295→  }
   296→}

   298→function validateBackupFile(parsed: unknown): {
   299→  valid: boolean;
   300→  error?: string;
   301→  file?: BackupFile;
   302→} {
   303→  if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
   304→    return { valid: false, error: "Invalid format: expected a JSON object." };
   305→  }
   306→  const obj = parsed as Record<string, unknown>;
   307→  const metadata = obj.metadata as Record<string, unknown> | undefined;
   308→  const data = obj.data as Record<string, unknown> | undefined;
   309→  if (!metadata || typeof metadata !== "object") {
   310→    return { valid: false, error: 'Missing "metadata" object.' };
   311→  }
   312→  if (!data || typeof data !== "object") {
   313→    return { valid: false, error: 'Missing "data" object.' };
   314→  }
   315→  // At least one collection must be an array
   316→  const collections = ["ingredients", "recipes", "stockMovements", "dailyMeals", "purchases", "expenses", "suppliers", "users"];
   317→  const hasAny = collections.some((k) => Array.isArray((data as Record<string, unknown>)[k]));
   318→  if (!hasAny) {
   319→    return { valid: false, error: "Backup contains no recognizable data collections." };
   320→  }
   321→  return {
   322→    valid: true,
   323→    file: {
   324→      metadata: metadata as unknown as BackupMetadata,
   325→      data,
   326→    },
   327→  };
   328→}

   330→// ─── Circular Budget Gauge ───────────────────────────────────────────────────
   331→
   332→function BudgetGauge({ percent, label, spent, budget }: {
   333→  percent: number;
   334→  label: string;
   335→  spent: number;
   336→  budget: number;
   337→}) {
   338→  const size = 140;
   339→  const stroke = 12;
   340→  const radius = (size - stroke) / 2;
   341→  const circumference = 2 * Math.PI * radius;
   342→  const clamped = Math.max(0, Math.min(percent, 150)); // allow up to 150% for visual
   343→  const dashOffset = circumference - (Math.min(clamped, 100) / 100) * circumference;
   344→  const color = getBudgetColor(percent);
   345→
   346→  return (
   347→    <div className="flex flex-col items-center gap-2">
   348→      <div
   349→        className="relative"
   350→        style={{ width: size, height: size }}
   351→        role="meter"
   352→        aria-valuenow={Math.round(percent)}
   353→        aria-valuemin={0}
   354→        aria-valuemax={100}
   355→        aria-label={`${label} budget utilization`}
   356→      >
   357→        <svg width={size} height={size} className="-rotate-90">
   358→          <circle
   359→            cx={size / 2}
   360→            cy={size / 2}
   361→            r={radius}
   362→            fill="none"
   363→            stroke="var(--muted)"
   364→            strokeWidth={stroke}
   365→          />
   366→          <circle
   367→            cx={size / 2}
   368→            cy={size / 2}
   369→            r={radius}
   370→            fill="none"
   371→            stroke={color}
   372→            strokeWidth={stroke}
   373→            strokeLinecap="round"
   374→            strokeDasharray={circumference}
   375→            strokeDashoffset={dashOffset}
   376→            style={{
   377→              transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   378→            }}
   379→          />
   380→        </svg>
   381→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   382→          <span
   383→            className="text-2xl font-bold tabular-nums"
   384→            style={{ color }}
   385→          >
   386→            {Math.round(percent)}%
   387→          </span>
   388→          <span className="text-[10px] font-medium text-muted-foreground">
   389→            Used
   390→          </span>
   391→        </div>
   392→      </div>
   393→      <div className="text-center">
   394→        <p className="text-sm font-semibold">{label}</p>
   395→        <p className="text-xs text-muted-foreground tabular-nums">
   396→          {formatINRShort(spent)} / {formatINRShort(budget)}
   397→        </p>
   398→      </div>
   399→    </div>
   400→  );
   401→}
