   110→
   111→function getCategoryColor(category: string): string {
   112→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   113→  // Fallback: hash the category name to pick a consistent color
   114→  let hash = 0;
   115→  for (let i = 0; i < category.length; i++) {
   116→    hash = (hash << 5) - hash + category.charCodeAt(i);
   117→    hash |= 0;
   118→  }
   119→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   120→}

   218→// ─── Low-Stock Alert Banner ────────────────────────────────────────────────
   219→
   220→const LOW_STOCK_BANNER_DISMISS_KEY = "rcs-dashboard-lowstock-dismissed";
   221→const LOW_STOCK_BANNER_UPDATE_EVENT = "rcs-lowstock-update";
   222→
   223→/** SSR-safe subscription to a session-storage dismissal flag. Returns
   224→ *  `[isDismissed, dismiss]` — calling `dismiss` writes to sessionStorage and
   225→ *  notifies subscribers so the component re-renders without calling setState
   226→ *  synchronously inside an effect. */
   227→function useLowStockDismissed(): [boolean, () => void] {
   228→  const subscribe = useCallback((callback: () => void) => {
   229→    if (typeof window === "undefined") return () => {};
   230→    const handler = () => callback();
   231→    window.addEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   232→    window.addEventListener("storage", handler);
   233→    return () => {
   234→      window.removeEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   235→      window.removeEventListener("storage", handler);
   236→    };
   237→  }, []);
   238→
   239→  const getSnapshot = useCallback(() => {
   240→    try {
   241→      return window.sessionStorage.getItem(LOW_STOCK_BANNER_DISMISS_KEY) === "true";
   242→    } catch {
   243→      return false;
   244→    }
   245→  }, []);
   246→
   247→  const getServerSnapshot = useCallback(() => false, []);
   248→
   249→  const isDismissed = useSyncExternalStore(
   250→    subscribe,
   251→    getSnapshot,
   252→    getServerSnapshot
   253→  );
   254→
   255→  const dismiss = useCallback(() => {
   256→    try {
   257→      window.sessionStorage.setItem(LOW_STOCK_BANNER_DISMISS_KEY, "true");
   258→      window.dispatchEvent(new Event(LOW_STOCK_BANNER_UPDATE_EVENT));
   259→    } catch {
   260→      // Ignore write failure (private mode, etc.).
   261→    }
   262→  }, []);
   263→
   264→  return [isDismissed, dismiss];
   265→}

   842→function CircularGauge({ percent }: { percent: number }) {
   843→  const size = 168;
   844→  const stroke = 14;
   845→  const radius = (size - stroke) / 2;
   846→  const circumference = 2 * Math.PI * radius;
   847→  const clamped = Math.max(0, Math.min(100, percent));
   848→  const dashOffset = circumference - (clamped / 100) * circumference;
   849→
   850→  // Color shifts: red (low) → amber (mid) → emerald (high)
   851→  const color =
   852→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   853→
   854→  return (
   855→    <div
   856→      className="relative"
   857→      style={{ width: size, height: size }}
   858→      role="meter"
   859→      aria-valuenow={Math.round(clamped)}
   860→      aria-valuemin={0}
   861→      aria-valuemax={100}
   862→      aria-label="Stock health percentage"
   863→    >
   864→      <svg width={size} height={size} className="-rotate-90">
   865→        <circle
   866→          cx={size / 2}
   867→          cy={size / 2}
   868→          r={radius}
   869→          fill="none"
   870→          stroke="var(--muted)"
   871→          strokeWidth={stroke}
   872→        />
   873→        <circle
   874→          cx={size / 2}
   875→          cy={size / 2}
   876→          r={radius}
   877→          fill="none"
   878→          stroke={color}
   879→          strokeWidth={stroke}
   880→          strokeLinecap="round"
   881→          strokeDasharray={circumference}
   882→          strokeDashoffset={dashOffset}
   883→          style={{
   884→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   885→          }}
   886→        />
   887→      </svg>
   888→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   889→        <span
   890→          className="text-3xl font-bold tabular-nums"
   891→          style={{ color }}
   892→        >
   893→          {Math.round(clamped)}%
   894→        </span>
   895→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   896→          Above Par
   897→        </span>
   898→      </div>
   899→    </div>
   900→  );
   901→}

   905→/** Renders percentage labels INSIDE donut slices */
   906→function renderPiePercentLabel(props: {
   907→  cx: number;
   908→  cy: number;
   909→  midAngle: number;
   910→  innerRadius: number;
   911→  outerRadius: number;
   912→  percent: number;
   913→}) {
   914→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   915→  if (percent < 0.05) return null; // hide labels on slices < 5%
   916→
   917→  const RADIAN = Math.PI / 180;
   918→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   919→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   920→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   921→
   922→  return (
   923→    <text
   924→      x={x}
   925→      y={y}
   926→      fill="#fff"
   927→      textAnchor="middle"
   928→      dominantBaseline="central"
   929→      fontSize={12}
   930→      fontWeight={700}
   931→      style={{ pointerEvents: "none" }}
   932→    >
   933→      {(percent * 100).toFixed(0)}%
   934→    </text>
   935→  );
