   111→   111→function getCategoryColor(category: string): string {
   112→   112→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   113→   113→  // Fallback: hash the category name to pick a consistent color
   114→   114→  let hash = 0;
   115→   115→  for (let i = 0; i < category.length; i++) {
   116→   116→    hash = (hash << 5) - hash + category.charCodeAt(i);
   117→   117→    hash |= 0;
   118→   118→  }
   119→   119→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   120→   120→}

   218→   218→// ─── Low-Stock Alert Banner ────────────────────────────────────────────────
   219→   219→
   220→   220→const LOW_STOCK_BANNER_DISMISS_KEY = "rcs-dashboard-lowstock-dismissed";
   221→   221→const LOW_STOCK_BANNER_UPDATE_EVENT = "rcs-lowstock-update";
   222→   222→
   223→   223→/** SSR-safe subscription to a session-storage dismissal flag. Returns
   224→   224→ *  `[isDismissed, dismiss]` — calling `dismiss` writes to sessionStorage and
   225→   225→ *  notifies subscribers so the component re-renders without calling setState
   226→   226→ *  synchronously inside an effect. */
   227→   227→function useLowStockDismissed(): [boolean, () => void] {
   228→   228→  const subscribe = useCallback((callback: () => void) => {
   229→   229→    if (typeof window === "undefined") return () => {};
   230→   230→    const handler = () => callback();
   231→   231→    window.addEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   232→   232→    window.addEventListener("storage", handler);
   233→   233→    return () => {
   234→   234→      window.removeEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   235→   235→      window.removeEventListener("storage", handler);
   236→   236→    };
   237→   237→  }, []);
   238→   238→
   239→   239→  const getSnapshot = useCallback(() => {
   240→   240→    try {
   241→   241→      return window.sessionStorage.getItem(LOW_STOCK_BANNER_DISMISS_KEY) === "true";
   242→   242→    } catch {
   243→   243→      return false;
   244→   244→    }
   245→   245→  }, []);
   246→   246→
   247→   247→  const getServerSnapshot = useCallback(() => false, []);
   248→   248→
   249→   249→  const isDismissed = useSyncExternalStore(
   250→   250→    subscribe,
   251→   251→    getSnapshot,
   252→   252→    getServerSnapshot
   253→   253→  );
   254→   254→
   255→   255→  const dismiss = useCallback(() => {
   256→   256→    try {
   257→   257→      window.sessionStorage.setItem(LOW_STOCK_BANNER_DISMISS_KEY, "true");
   258→   258→      window.dispatchEvent(new Event(LOW_STOCK_BANNER_UPDATE_EVENT));
   259→   259→    } catch {
   260→   260→      // Ignore write failure (private mode, etc.).
   261→   261→    }
   262→   262→  }, []);
   263→   263→
   264→   264→  return [isDismissed, dismiss];
   265→   265→}

   842→   842→function CircularGauge({ percent }: { percent: number }) {
   843→   843→  const size = 168;
   844→   844→  const stroke = 14;
   845→   845→  const radius = (size - stroke) / 2;
   846→   846→  const circumference = 2 * Math.PI * radius;
   847→   847→  const clamped = Math.max(0, Math.min(100, percent));
   848→   848→  const dashOffset = circumference - (clamped / 100) * circumference;
   849→   849→
   850→   850→  // Color shifts: red (low) → amber (mid) → emerald (high)
   851→   851→  const color =
   852→   852→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   853→   853→
   854→   854→  return (
   855→   855→    <div
   856→   856→      className="relative"
   857→   857→      style={{ width: size, height: size }}
   858→   858→      role="meter"
   859→   859→      aria-valuenow={Math.round(clamped)}
   860→   860→      aria-valuemin={0}
   861→   861→      aria-valuemax={100}
   862→   862→      aria-label="Stock health percentage"
   863→   863→    >
   864→   864→      <svg width={size} height={size} className="-rotate-90">
   865→   865→        <circle
   866→   866→          cx={size / 2}
   867→   867→          cy={size / 2}
   868→   868→          r={radius}
   869→   869→          fill="none"
   870→   870→          stroke="var(--muted)"
   871→   871→          strokeWidth={stroke}
   872→   872→        />
   873→   873→        <circle
   874→   874→          cx={size / 2}
   875→   875→          cy={size / 2}
   876→   876→          r={radius}
   877→   877→          fill="none"
   878→   878→          stroke={color}
   879→   879→          strokeWidth={stroke}
   880→   880→          strokeLinecap="round"
   881→   881→          strokeDasharray={circumference}
   882→   882→          strokeDashoffset={dashOffset}
   883→   883→          style={{
   884→   884→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   885→   885→          }}
   886→   886→        />

   905→   905→/** Renders percentage labels INSIDE donut slices */
   906→   906→function renderPiePercentLabel(props: {
   907→   907→  cx: number;
   908→   908→  cy: number;
   909→   909→  midAngle: number;
   910→   910→  innerRadius: number;
   911→   911→  outerRadius: number;
   912→   912→  percent: number;
   913→   913→}) {
   914→   914→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   915→   915→  if (percent < 0.05) return null; // hide labels on slices < 5%
   916→   916→
   917→   917→  const RADIAN = Math.PI / 180;
   918→   918→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   919→   919→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   920→   920→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   921→   921→
   922→   922→  return (
   923→   923→    <text
   924→   924→      x={x}
   925→   925→      y={y}
   926→   926→      fill="#fff"
   927→   927→      textAnchor="middle"
   928→   928→      dominantBaseline="central"
   929→   929→      fontSize={12}
   930→   930→      fontWeight={700}
   931→   931→      style={{ pointerEvents: "none" }}
   932→   932→    >
   933→   933→      {(percent * 100).toFixed(0)}%
   934→   934→    </text>
   935→   935→  );
