   215→function formatRupee(amount: number): string {
   216→  return (
   217→    "₹" +
   218→    amount.toLocaleString("en-IN", {
   219→      minimumFractionDigits: 2,
   220→      maximumFractionDigits: 2,
   221→    })
   222→  );
   223→}

   225→function calcTotalIngredientCost(ingredients: RecipeIngredient[]): number {
   226→  return ingredients.reduce((sum, ri) => sum + ri.quantity * ri.ingredient.avgCost, 0);
   227→}
   228→
   229→function calcCostPerMeal(ingredients: RecipeIngredient[], baseServings: number): number {
   230→  if (baseServings <= 0) return 0;
   231→  return calcTotalIngredientCost(ingredients) / baseServings;
   232→}
   233→
   234→function calcCostFor600(ingredients: RecipeIngredient[], baseServings: number): number {
   235→  return calcCostPerMeal(ingredients, baseServings) * 600;
   236→}
   237→
   238→function calcCostBreakdown(ingredients: RecipeIngredient[]): {
   239→  entries: CategoryBreakdownEntry[];
   240→  total: number;
   241→} {
   242→  const byCategory: Record<string, number> = {};
   243→  let total = 0;
   244→  ingredients.forEach((ri) => {
   245→    const cost = ri.quantity * ri.ingredient.avgCost;
   246→    const cat = ri.ingredient.category || "Other";
   247→    byCategory[cat] = (byCategory[cat] || 0) + cost;
   248→    total += cost;
   249→  });
   250→  const entries = Object.entries(byCategory)
   251→    .map(([category, amount]) => ({
   252→      category,
   253→      amount,
   254→      percent: total > 0 ? (amount / total) * 100 : 0,
   255→    }))
   256→    .sort((a, b) => b.amount - a.amount);
   257→  return { entries, total };
   258→}
   259→
   260→// Compare cost using lastPurchasePrice vs avgCost to show trend
   261→function calcCostTrend(ingredients: RecipeIngredient[]): {
   262→  direction: "up" | "down" | "stable";
   263→  percentChange: number;
   264→} {
   265→  let costByAvg = 0;
   266→  let costByLast = 0;
   267→  ingredients.forEach((ri) => {
   268→    costByAvg += ri.quantity * ri.ingredient.avgCost;
   269→    costByLast += ri.quantity * ri.ingredient.lastPurchasePrice;
   270→  });
   271→  if (costByAvg === 0) return { direction: "stable", percentChange: 0 };
   272→  const pctChange = ((costByLast - costByAvg) / costByAvg) * 100;
   273→  if (Math.abs(pctChange) < 2) return { direction: "stable", percentChange: pctChange };
   274→  return { direction: pctChange > 0 ? "up" : "down", percentChange: pctChange };
   275→}

   288→/**
   289→ * Renders a recipe image. If no image is provided, shows a beautiful
   290→ * gradient placeholder with a UtensilsCrossed icon and the recipe's
   291→ * first letter as fallback.
   292→ */
   293→function RecipeImage({
   294→  imageUrl,
   295→  name,
   296→  className,
   297→  imgClassName,
   298→  eager = false,
   299→  rounded = "rounded-t-xl",
   300→}: RecipeImageProps) {
   301→  const firstLetter = (name || "?").trim().charAt(0).toUpperCase() || "?";
   302→  return (
   303→    <div
   304→      className={cn(
   305→        "relative overflow-hidden bg-gradient-to-br from-amber-100 via-orange-100 to-amber-50 dark:from-amber-900/40 dark:via-orange-900/30 dark:to-amber-950/20",
   306→        rounded,
   307→        className
   308→      )}
   309→    >
   310→      {imageUrl ? (
   311→        <img
   312→          src={imageUrl}
   313→          alt={name}
   314→          loading={eager ? "eager" : "lazy"}
   315→          className={cn(
   316→            "h-full w-full object-cover transition-transform duration-300 group-hover:scale-105",
   317→            imgClassName
   318→          )}
   319→        />
   320→      ) : (
   321→        <div className="flex h-full w-full flex-col items-center justify-center gap-1">
   322→          <UtensilsCrossed className="h-8 w-8 text-orange-400/70 dark:text-orange-500/50" />
   323→          <span className="bg-gradient-to-br from-orange-500 to-amber-500 bg-clip-text text-3xl font-black text-transparent dark:from-orange-300 dark:to-amber-300">
   324→            {firstLetter}
   325→          </span>
   326→        </div>
   327→      )}
   328→    </div>
   329→  );
   330→}
