   215→   215→function formatRupee(amount: number): string {
   216→   216→  return (
   217→   217→    "₹" +
   218→   218→    amount.toLocaleString("en-IN", {
   219→   219→      minimumFractionDigits: 2,
   220→   220→      maximumFractionDigits: 2,
   221→   221→    })
   222→   222→  );
   223→   223→}

   225→   225→function calcTotalIngredientCost(ingredients: RecipeIngredient[]): number {
   226→   226→  return ingredients.reduce((sum, ri) => sum + ri.quantity * ri.ingredient.avgCost, 0);
   227→   227→}
   228→   228→
   229→   229→function calcCostPerMeal(ingredients: RecipeIngredient[], baseServings: number): number {
   230→   230→  if (baseServings <= 0) return 0;
   231→   231→  return calcTotalIngredientCost(ingredients) / baseServings;
   232→   232→}
   233→   233→
   234→   234→function calcCostFor600(ingredients: RecipeIngredient[], baseServings: number): number {
   235→   235→  return calcCostPerMeal(ingredients, baseServings) * 600;
   236→   236→}
   237→   237→
   238→   238→function calcCostBreakdown(ingredients: RecipeIngredient[]): {
   239→   239→  entries: CategoryBreakdownEntry[];
   240→   240→  total: number;
   241→   241→} {
   242→   242→  const byCategory: Record<string, number> = {};
   243→   243→  let total = 0;
   244→   244→  ingredients.forEach((ri) => {
   245→   245→    const cost = ri.quantity * ri.ingredient.avgCost;
   246→   246→    const cat = ri.ingredient.category || "Other";
   247→   247→    byCategory[cat] = (byCategory[cat] || 0) + cost;
   248→   248→    total += cost;
   249→   249→  });
   250→   250→  const entries = Object.entries(byCategory)
   251→   251→    .map(([category, amount]) => ({
   252→   252→      category,
   253→   253→      amount,
   254→   254→      percent: total > 0 ? (amount / total) * 100 : 0,
   255→   255→    }))
   256→   256→    .sort((a, b) => b.amount - a.amount);
   257→   257→  return { entries, total };
   258→   258→}

   260→   260→// Compare cost using lastPurchasePrice vs avgCost to show trend
   261→   261→function calcCostTrend(ingredients: RecipeIngredient[]): {
   262→   262→  direction: "up" | "down" | "stable";
   263→   263→  percentChange: number;
   264→   264→} {
   265→   265→  let costByAvg = 0;
   266→   266→  let costByLast = 0;
   267→   267→  ingredients.forEach((ri) => {
   268→   268→    costByAvg += ri.quantity * ri.ingredient.avgCost;
   269→   269→    costByLast += ri.quantity * ri.ingredient.lastPurchasePrice;
   270→   270→  });
   271→   271→  if (costByAvg === 0) return { direction: "stable", percentChange: 0 };
   272→   272→  const pctChange = ((costByLast - costByAvg) / costByAvg) * 100;
   273→   273→  if (Math.abs(pctChange) < 2) return { direction: "stable", percentChange: pctChange };
   274→   274→  return { direction: pctChange > 0 ? "up" : "down", percentChange: pctChange };
   275→   275→}
