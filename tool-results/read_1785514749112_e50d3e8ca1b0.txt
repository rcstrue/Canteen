   125→
   126→function getCategoryColor(category: string): string {
   127→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   128→  // Fallback: hash the category name to pick a consistent color
   129→  let hash = 0;
   130→  for (let i = 0; i < category.length; i++) {
   131→    hash = (hash << 5) - hash + category.charCodeAt(i);
   132→    hash |= 0;
   133→  }
   134→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   135→}

   262→// ─── Low-Stock Alert Banner ────────────────────────────────────────────────
   263→
   264→const LOW_STOCK_BANNER_DISMISS_KEY = "rcs-dashboard-lowstock-dismissed";
   265→const LOW_STOCK_BANNER_UPDATE_EVENT = "rcs-lowstock-update";
   266→
   267→/** SSR-safe subscription to a session-storage dismissal flag. Returns
   268→ *  `[isDismissed, dismiss]` — calling `dismiss` writes to sessionStorage and
   269→ *  notifies subscribers so the component re-renders without calling setState
   270→ *  synchronously inside an effect. */
   271→function useLowStockDismissed(): [boolean, () => void] {
   272→  const subscribe = useCallback((callback: () => void) => {
   273→    if (typeof window === "undefined") return () => {};
   274→    const handler = () => callback();
   275→    window.addEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   276→    window.addEventListener("storage", handler);
   277→    return () => {
   278→      window.removeEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   279→      window.removeEventListener("storage", handler);
   280→    };
   281→  }, []);
   282→
   283→  const getSnapshot = useCallback(() => {
   284→    try {
   285→      return window.sessionStorage.getItem(LOW_STOCK_BANNER_DISMISS_KEY) === "true";
   286→    } catch {
   287→      return false;
   288→    }
   289→  }, []);
   290→
   291→  const getServerSnapshot = useCallback(() => false, []);
   292→
   293→  const isDismissed = useSyncExternalStore(
   294→    subscribe,
   295→    getSnapshot,
   296→    getServerSnapshot
   297→  );
   298→
   299→  const dismiss = useCallback(() => {
   300→    try {
   301→      window.sessionStorage.setItem(LOW_STOCK_BANNER_DISMISS_KEY, "true");
   302→      window.dispatchEvent(new Event(LOW_STOCK_BANNER_UPDATE_EVENT));
   303→    } catch {
   304→      // Ignore write failure (private mode, etc.).
   305→    }
   306→  }, []);
   307→
   308→  return [isDismissed, dismiss];
   309→}

   808→
   809→interface TrendBadgeProps {
   810→  /** Percentage change (positive = increase, negative = decrease) */
   811→  pct: number | null;
   812→  /** For cost metrics, "lower is better" → up arrow = red. For meals/positive metrics, "up is better" */
   813→  lowerIsBetter?: boolean;
   814→  /** Comparison label e.g. "vs avg day" */
   815→  label: string;
   816→}
   817→
   818→function TrendBadge({ pct, lowerIsBetter = true, label }: TrendBadgeProps) {
   819→  if (pct === null || !isFinite(pct)) {
   820→    return (
   821→      <div className="flex items-center gap-1 text-xs text-muted-foreground">
   822→        <Minus className="h-3.5 w-3.5" />
   823→        <span>{label}</span>
   824→      </div>
   825→    );
   826→  }
   827→
   828→  const isUp = pct > 0;
   829→  const isFlat = Math.abs(pct) < 0.5;
   830→  const goodDirection = lowerIsBetter ? !isUp : isUp;
   831→  const colorClass = isFlat
   832→    ? "text-muted-foreground"
   833→    : goodDirection
   834→      ? "text-emerald-600 dark:text-emerald-400"
   835→      : "text-rose-600 dark:text-rose-400";
   836→  const Icon = isFlat ? Minus : isUp ? TrendingUp : TrendingDown;
   837→
   838→  return (
   839→    <div className="flex items-center gap-1.5 text-xs">
   840→      <Icon className={`h-4 w-4 ${colorClass}`} />
   841→      <span className={`font-bold tabular-nums ${colorClass}`}>
   842→        {isFlat ? "0%" : `${isUp ? "+" : ""}${pct.toFixed(1)}%`}
   843→      </span>
   844→      <span className="text-muted-foreground">{label}</span>
   845→    </div>
   846→  );
   847→}

   908→function CircularGauge({ percent }: { percent: number }) {
   909→  const size = 168;
   910→  const stroke = 14;
   911→  const radius = (size - stroke) / 2;
   912→  const circumference = 2 * Math.PI * radius;
   913→  const clamped = Math.max(0, Math.min(100, percent));
   914→  const dashOffset = circumference - (clamped / 100) * circumference;
   915→
   916→  // Color shifts: red (low) → amber (mid) → emerald (high)
   917→  const color =
   918→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   919→
   920→  return (
   921→    <div
   922→      className="relative"
   923→      style={{ width: size, height: size }}
   924→      role="meter"
   925→      aria-valuenow={Math.round(clamped)}
   926→      aria-valuemin={0}
   927→      aria-valuemax={100}
   928→      aria-label="Stock health percentage"
   929→    >
   930→      <svg width={size} height={size} className="-rotate-90">
   931→        <circle
   932→          cx={size / 2}
   933→          cy={size / 2}
   934→          r={radius}
   935→          fill="none"
   936→          stroke="var(--muted)"
   937→          strokeWidth={stroke}
   938→        />
   939→        <circle
   940→          cx={size / 2}
   941→          cy={size / 2}
   942→          r={radius}
   943→          fill="none"
   944→          stroke={color}
   945→          strokeWidth={stroke}
   946→          strokeLinecap="round"
   947→          strokeDasharray={circumference}
   948→          strokeDashoffset={dashOffset}
   949→          style={{
   950→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   951→          }}
   952→        />
   953→      </svg>
   954→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   955→        <span
   956→          className="text-3xl font-bold tabular-nums"
   957→          style={{ color }}
   958→        >
   959→          {Math.round(clamped)}%
   960→        </span>
   961→        <span className="mt-0.5 text-xs font-medium text-muted-foreground">
   962→          Above Par
   963→        </span>
   964→      </div>
   965→    </div>
   966→  );
   967→}

   969→// ─── Donut Pie Label Renderer ───────────────────────────────────────────────
   970→
   971→/** Renders percentage labels INSIDE donut slices */
   972→function renderPiePercentLabel(props: {
   973→  cx: number;
   974→  cy: number;
   975→  midAngle: number;
   976→  innerRadius: number;
   977→  outerRadius: number;
   978→  percent: number;
   979→}) {
   980→  const { cx, cy, midAngle, innerRadius, outerRadius, percent } = props;
   981→  if (percent < 0.05) return null; // hide labels on slices < 5%
   982→
   983→  const RADIAN = Math.PI / 180;
   984→  const radius = innerRadius + (outerRadius - innerRadius) * 0.5;
   985→  const x = cx + radius * Math.cos(-midAngle * RADIAN);
   986→  const y = cy + radius * Math.sin(-midAngle * RADIAN);
   987→
   988→  return (
   989→    <text
   990→      x={x}
   991→      y={y}
   992→      fill="#fff"
   993→      textAnchor="middle"
   994→      dominantBaseline="central"
   995→      fontSize={12}
   996→      fontWeight={700}
   997→      style={{ pointerEvents: "none" }}
   998→    >
   999→      {(percent * 100).toFixed(0)}%
  1000→    </text>
  1001→  );
  1002→}
