   263→function isLowStock(item: Ingredient): boolean {
   264→  return item.currentStock < item.minStock;
   265→}
   266→
   267→type StockHealth = "OK" | "LOW" | "CRITICAL";
   268→
   269→function getStockHealth(item: Ingredient): StockHealth {
   270→  if (item.minStock === 0) return "OK";
   271→  const ratio = item.currentStock / item.minStock;
   272→  if (ratio < 1) return "CRITICAL";    // currentStock < minStock → red
   273→  if (ratio < 1.2) return "LOW";       // 80-100% of minStock (near par) → amber
   274→  return "OK";                          // currentStock >= minStock → green
   275→}

   312→function formatDateDDMMYYYY(dateStr: string): string {
   313→  if (!dateStr) return "—";
   314→  const d = new Date(dateStr);
   315→  if (isNaN(d.getTime())) return "—";
   316→  const dd = String(d.getDate()).padStart(2, "0");
   317→  const mm = String(d.getMonth() + 1).padStart(2, "0");
   318→  const yyyy = d.getFullYear();
   319→  return `${dd}/${mm}/${yyyy}`;
   320→}
   321→
   322→function formatINR(value: number): string {
   323→  const formatted = new Intl.NumberFormat("en-IN", {
   324→    minimumFractionDigits: 2,
   325→    maximumFractionDigits: 2,
   326→  }).format(value || 0);
   327→  return `₹${formatted}`;
   328→}
