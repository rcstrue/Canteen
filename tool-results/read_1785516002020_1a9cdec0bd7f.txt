   139→
   140→function getCategoryColor(category: string): string {
   141→  if (CATEGORY_COLORS[category]) return CATEGORY_COLORS[category];
   142→  // Fallback: hash the category name to pick a consistent color
   143→  let hash = 0;
   144→  for (let i = 0; i < category.length; i++) {
   145→    hash = (hash << 5) - hash + category.charCodeAt(i);
   146→    hash |= 0;
   147→  }
   148→  return CHART_COLORS[Math.abs(hash) % CHART_COLORS.length];
   149→}

   276→// ─── Low-Stock Alert Banner ────────────────────────────────────────────────
   277→
   278→const LOW_STOCK_BANNER_DISMISS_KEY = "rcs-dashboard-lowstock-dismissed";
   279→const LOW_STOCK_BANNER_UPDATE_EVENT = "rcs-lowstock-update";
   280→
   281→/** SSR-safe subscription to a session-storage dismissal flag. Returns
   282→ *  `[isDismissed, dismiss]` — calling `dismiss` writes to sessionStorage and
   283→ *  notifies subscribers so the component re-renders without calling setState
   284→ *  synchronously inside an effect. */
   285→function useLowStockDismissed(): [boolean, () => void] {
   286→  const subscribe = useCallback((callback: () => void) => {
   287→    if (typeof window === "undefined") return () => {};
   288→    const handler = () => callback();
   289→    window.addEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   290→    window.addEventListener("storage", handler);
   291→    return () => {
   292→      window.removeEventListener(LOW_STOCK_BANNER_UPDATE_EVENT, handler);
   293→      window.removeEventListener("storage", handler);
   294→    };
   295→  }, []);
   296→
   297→  const getSnapshot = useCallback(() => {
   298→    try {
   299→      return window.sessionStorage.getItem(LOW_STOCK_BANNER_DISMISS_KEY) === "true";
   300→    } catch {
   301→      return false;
   302→    }
   303→  }, []);
   304→
   305→  const getServerSnapshot = useCallback(() => false, []);
   306→
   307→  const isDismissed = useSyncExternalStore(
   308→    subscribe,
   309→    getSnapshot,
   310→    getServerSnapshot
   311→  );
   312→
   313→  const dismiss = useCallback(() => {
   314→    try {
   315→      window.sessionStorage.setItem(LOW_STOCK_BANNER_DISMISS_KEY, "true");
   316→      window.dispatchEvent(new Event(LOW_STOCK_BANNER_UPDATE_EVENT));
   317→    } catch {
   318→      // Ignore write failure (private mode, etc.).
   319→    }
   320→  }, []);
   321→
   322→  return [isDismissed, dismiss];
   323→}

   868→      ? "text-emerald-600 dark:text-emerald-400"

   874→      <Icon className={`h-4 w-4 ${colorClass}`} />
   875→      <span className={`font-bold tabular-nums ${colorClass}`}>

   944→function CircularGauge({ percent }: { percent: number }) {
   945→  const size = 168;
   946→  const stroke = 14;
   947→  const radius = (size - stroke) / 2;
   948→  const circumference = 2 * Math.PI * radius;
   949→  const clamped = Math.max(0, Math.min(100, percent));
   950→  const dashOffset = circumference - (clamped / 100) * circumference;
   951→
   952→  // Color shifts: red (low) → amber (mid) → emerald (high)
   953→  const color =
   954→    clamped >= 80 ? "#10b981" : clamped >= 60 ? "#f59e0b" : "#f43f5e";
   955→
   956→  return (
   957→    <div
   958→      className="relative"
   959→      style={{ width: size, height: size }}
   960→      role="meter"
   961→      aria-valuenow={Math.round(clamped)}
   962→      aria-valuemin={0}
   963→      aria-valuemax={100}
   964→      aria-label="Stock health percentage"
   965→    >
   966→      <svg width={size} height={size} className="-rotate-90">
   967→        <circle
   968→          cx={size / 2}
   969→          cy={size / 2}
   970→          r={radius}
   971→          fill="none"
   972→          stroke="var(--muted)"
   973→          strokeWidth={stroke}
   974→        />
   975→        <circle
   976→          cx={size / 2}
   977→          cy={size / 2}
   978→          r={radius}
   979→          fill="none"
   980→          stroke={color}
   981→          strokeWidth={stroke}
   982→          strokeLinecap="round"
   983→          strokeDasharray={circumference}
   984→          strokeDashoffset={dashOffset}
   985→          style={{
   986→            transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   987→          }}
   988→        />
   989→      </svg>
   990→      <div className="absolute inset-0 flex flex-col items-center justify-center">
   991→        <span
   992→          className="text-3xl font-bold tabular-nums"
   993→          style={{ color }}
   994→        >
   995→          {Math.round(clamped)}%

  1049→function getDateRangeForPreset(preset: DateRangePreset): DateRange {
  1050→  const now = new Date();
  1051→  switch (preset) {
  1052→    case "today":
  1053→      return { from: startOfDay(now), to: endOfDay(now) };
  1054→    case "week":
  1055→      return { from: startOfWeek(now, { weekStartsOn: 0 }), to: now };
  1056→    case "month":
  1057→      return { from: startOfMonth(now), to: endOfMonth(now) };
  1058→    default:
  1059→      return { from: startOfDay(now), to: endOfDay(now) };
  1060→  }
  1061→}

  1068→      return "This Week";
  1069→    case "month":
  1070→      return "This Month";
  1071→    case "custom":
  1072→      if (range?.from && range?.to) {
  1073→        return `${format(range.from, "dd MMM")} – ${format(range.to, "dd MMM")}`;
  1074→      }
  1075→      return "Custom Range";
  1076→    default:
  1077→      return "Today";
