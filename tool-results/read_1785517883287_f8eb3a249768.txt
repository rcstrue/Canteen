   143→// ─── Helpers ─────────────────────────────────────────────────────────────────
   144→
   145→function formatCurrency(amount: number): string {
   146→  return new Intl.NumberFormat("en-IN", {
   147→    style: "currency",
   148→    currency: "INR",
   149→    minimumFractionDigits: 0,
   150→    maximumFractionDigits: 2,
   151→  }).format(amount);
   152→}
   153→
   154→function formatDate(dateStr: string): string {
   155→  const d = new Date(dateStr);
   156→  const day = String(d.getDate()).padStart(2, "0");
   157→  const month = String(d.getMonth() + 1).padStart(2, "0");
   158→  const year = d.getFullYear();
   159→  return `${day}/${month}/${year}`;
   160→}
   161→
   162→function getTodayStr(): string {
   163→  const d = new Date();
   164→  return d.toISOString().split("T")[0];
   165→}
   166→
   167→function getMonthStartStr(): string {
   168→  const d = new Date();
   169→  d.setDate(1);
   170→  return d.toISOString().split("T")[0];
   171→}

   344→  async function handleSubmit() {
   345→    if (!formDate || !formCategory || !formAmount || Number(formAmount) <= 0) return;

   349→      const body = {
   350→        date: formDate,

   352→        amount: Number(formAmount),
   353→        description: formDescription || null,

   364→        const res = await fetch("/api/expenses", {
   365→          method: "POST",
   366→          headers: { "Content-Type": "application/json" },
   367→          body: JSON.stringify(body),
