   188→function getCurrentMonthStr(): string {
   189→  const d = new Date();
   190→  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
   191→}
   192→
   193→/** Utilization color band — emerald / amber / orange / red. */
   194→function getUtilBand(pct: number): {
   195→  color: string;
   196→  bg: string;
   197→  text: string;
   198→  ring: string;
   199→  label: string;
   200→} {
   201→  if (pct > 100) {
   202→    return {
   203→      color: '#ef4444',
   204→      bg: 'bg-red-500',
   205→      text: 'text-red-600 dark:text-red-400',
   206→      ring: 'ring-red-200 dark:ring-red-900/40',
   207→      label: 'Over Budget',
   208→    };
   209→  }
   210→  if (pct >= 80) {
   211→    return {
   212→      color: '#f97316',
   213→      bg: 'bg-orange-500',
   214→      text: 'text-orange-600 dark:text-orange-400',
   215→      ring: 'ring-orange-200 dark:ring-orange-900/40',
   216→      label: 'Critical',
   217→    };
   218→  }
   219→  if (pct >= 60) {
   220→    return {
   221→      color: '#f59e0b',
   222→      bg: 'bg-amber-500',
   223→      text: 'text-amber-600 dark:text-amber-400',
   224→      ring: 'ring-amber-200 dark:ring-amber-900/40',
   225→      label: 'Caution',
   226→    };
   227→  }
   228→  return {
   229→    color: '#10b981',
   230→    bg: 'bg-emerald-500',
   231→    text: 'text-emerald-600 dark:text-emerald-400',
   232→    ring: 'ring-emerald-200 dark:ring-emerald-900/40',
   233→    label: 'On Track',
   234→  };
   235→}
   236→
   237→function formatINR(amount: number, withDecimals = false): string {
   238→  return withDecimals ? inrFmt2.format(amount) : inrFmt.format(amount);
   239→}
   240→
   241→function formatINRShort(amount: number): string {
   242→  if (Math.abs(amount) >= 10000000) {
   243→    return `₹${(amount / 10000000).toFixed(2)}Cr`;
   244→  }
   245→  if (Math.abs(amount) >= 100000) {
   246→    return `₹${(amount / 100000).toFixed(2)}L`;
   247→  }
   248→  if (Math.abs(amount) >= 1000) {
   249→    return `₹${(amount / 1000).toFixed(1)}k`;
   250→  }
   251→  return `₹${amount.toFixed(0)}`;
   252→}

  1293→function SetBudgetForm({
  1294→  existingBudget,
  1295→  defaultMonth,
  1296→  saving,
  1297→  onSave,
  1298→  onCancel,
  1299→}: {
  1300→  existingBudget: BudgetRow | null;
  1301→  defaultMonth: string;
  1302→  saving: boolean;
  1303→  onSave: (payload: SetBudgetPayload) => Promise<void>;
  1304→  onCancel: () => void;
  1305→}) {
  1306→  // Lazy initializers — runs once per mount. Parent controls remount via `key`
  1307→  // (changes whenever existingBudget changes after a save).
  1308→  const [formMonth, setFormMonth] = useState<string>(
  1309→    () => existingBudget?.month ?? defaultMonth
  1310→  );
  1311→  const [foodBudget, setFoodBudget] = useState<string>(
  1312→    () => (existingBudget ? String(existingBudget.foodBudget) : '')
  1313→  );
  1314→  const [operatingBudget, setOperatingBudget] = useState<string>(
  1315→    () => (existingBudget ? String(existingBudget.operatingBudget) : '')
  1316→  );
  1317→  const [alertThreshold, setAlertThreshold] = useState<string>(
  1318→    () => (existingBudget ? String(existingBudget.alertThreshold) : '80')
  1319→  );
  1320→
  1321→  const totalBudget = useMemo(() => {
  1322→    const f = parseFloat(foodBudget) || 0;
  1323→    const o = parseFloat(operatingBudget) || 0;
  1324→    return f + o;
  1325→  }, [foodBudget, operatingBudget]);
  1326→
  1327→  const canSubmit =
  1328→    !!formMonth &&
  1329→    parseFloat(foodBudget) >= 0 &&
  1330→    parseFloat(operatingBudget) >= 0 &&
  1331→    parseFloat(alertThreshold) >= 0 &&
  1332→    parseFloat(alertThreshold) <= 100 &&
  1333→    !saving;
  1334→
  1335→  const handleSubmit = async (e: React.FormEvent) => {
  1336→    e.preventDefault();
  1337→    if (!canSubmit) return;
  1338→    await onSave({
  1339→      month: formMonth,
  1340→      foodBudget: parseFloat(foodBudget) || 0,
  1341→      operatingBudget: parseFloat(operatingBudget) || 0,
  1342→      totalBudget,
  1343→      alertThreshold: parseFloat(alertThreshold) || 80,
  1344→    });
  1345→  };
  1346→
  1347→  return (
  1348→    <>
  1349→      <DialogHeader>
  1350→        <DialogTitle className="flex items-center gap-2">
  1351→          <PiggyBank className="h-5 w-5 text-amber-600 dark:text-amber-400" />
  1352→          {existingBudget ? 'Update Budget' : 'Set Monthly Budget'}
  1353→        </DialogTitle>
  1354→        <DialogDescription>
  1355→          Define the food and operating budgets for the selected month. The
  1356→          total budget is auto-calculated as Food + Operating.
  1357→        </DialogDescription>
  1358→      </DialogHeader>
  1359→        <form onSubmit={handleSubmit} className="space-y-4">
  1360→          <div className="space-y-2">
  1361→            <Label htmlFor="budget-month">Month</Label>
  1362→            <Input
  1363→              id="budget-month"
  1364→              type="month"
  1365→              value={formMonth}
  1366→              onChange={(e) => setFormMonth(e.target.value)}
  1367→              required
  1368→              disabled={!!existingBudget}
  1369→            />
  1370→            {existingBudget && (
  1371→              <p className="text-xs text-muted-foreground">
  1372→                Month cannot be changed for an existing budget.
  1373→              </p>
  1374→            )}
  1375→          </div>
  1376→          <div className="space-y-2">
  1377→            <Label htmlFor="food-budget">Food Budget (₹)</Label>
  1378→            <Input
  1379→              id="food-budget"
  1380→              type="number"
  1381→              min="0"
  1382→              step="0.01"
  1383→              placeholder="0.00"
  1384→              value={foodBudget}
  1385→              onChange={(e) => setFoodBudget(e.target.value)}
  1386→              required
  1387→            />
  1388→          </div>
  1389→          <div className="space-y-2">
  1390→            <Label htmlFor="operating-budget">Operating Budget (₹)</Label>
  1391→            <Input
  1392→              id="operating-budget"
  1393→              type="number"
  1394→              min="0"
  1395→              step="0.01"
  1396→              placeholder="0.00"
  1397→              value={operatingBudget}
  1398→              onChange={(e) => setOperatingBudget(e.target.value)}
  1399→              required
  1400→            />
  1401→            <p className="text-xs text-muted-foreground">
  1402→              Includes food cost + other operating expenses (gas, electricity,
  1403→              etc.)
  1404→            </p>
  1405→          </div>
  1406→          <div className="space-y-2">
  1407→            <Label htmlFor="total-budget">Total Budget (auto-calculated)</Label>
  1408→            <Input
  1409→              id="total-budget"
  1410→              type="text"
  1411→              readOnly
  1412→              value={formatINR(totalBudget, true)}
  1413→              className="bg-muted/50 font-semibold"
  1414→            />
  1415→          </div>
  1416→          <div className="space-y-2">
  1417→            <Label htmlFor="alert-threshold">Alert Threshold (%)</Label>
  1418→            <Input
  1419→              id="alert-threshold"
  1420→              type="number"
  1421→              min="0"
  1422→              max="100"
  1423→              step="1"
  1424→              value={alertThreshold}
  1425→              onChange={(e) => setAlertThreshold(e.target.value)}
  1426→              required
  1427→            />
  1428→            <p className="text-xs text-muted-foreground">
  1429→              Trigger a utilization alert once spend crosses this percentage.
  1430→            </p>
  1431→          </div>
  1432→          <DialogFooter>
  1433→            <Button
  1434→              type="button"
  1435→              variant="outline"
  1436→              onClick={onCancel}
  1437→              disabled={saving}
  1438→            >
  1439→              Cancel
  1440→            </Button>
  1441→            <Button
  1442→              type="submit"
  1443→              disabled={!canSubmit}
  1444→              className="bg-amber-600 hover:bg-amber-700 text-white"
  1445→            >
  1446→              {saving ? (
  1447→                <>
  1448→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
  1449→                  Saving...
  1450→                </>

  1452→                <>
  1453→                  <CheckCircle2 className="mr-2 h-4 w-4" />
  1454→                  Update Budget
  1455→                </>
  1456→              ) : (
  1457→                <>
  1458→                  <Plus className="mr-2 h-4 w-4" />
  1459→                  Set Budget
  1460→                </>
  1461→              )}
  1462→            </Button>
  1463→          </DialogFooter>
  1464→        </form>
  1465→    </>
  1466→  );
  1467→}
