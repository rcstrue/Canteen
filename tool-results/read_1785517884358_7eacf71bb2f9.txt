   238→function formatCurrency(amount: number): string {
   239→  return new Intl.NumberFormat("en-IN", {
   240→    style: "currency",
   241→    currency: "INR",
   242→    minimumFractionDigits: 2,
   243→    maximumFractionDigits: 2,
   244→  }).format(amount);
   245→}
   246→
   247→function formatCurrencyShort(amount: number): string {
   248→  if (amount >= 10000000) return `₹${(amount / 10000000).toFixed(2)}Cr`;
   249→  if (amount >= 100000) return `₹${(amount / 100000).toFixed(2)}L`;
   250→  if (amount >= 1000) return `₹${(amount / 1000).toFixed(1)}k`;
   251→  return formatCurrency(amount);
   252→}
   253→
   254→function formatDate(dateStr: string): string {
   255→  const d = new Date(dateStr);
   256→  const day = String(d.getDate()).padStart(2, "0");
   257→  const month = String(d.getMonth() + 1).padStart(2, "0");
   258→  const year = d.getFullYear();
   259→  return `${day}/${month}/${year}`;
   260→}
