   280→function calcTotalIngredientCost(ingredients: RecipeIngredient[]): number {
   281→  return ingredients.reduce((sum, ri) => sum + ri.quantity * ri.ingredient.avgCost, 0);
   282→}

   284→function calcCostPerMeal(ingredients: RecipeIngredient[], baseServings: number): number {
   285→  if (baseServings <= 0) return 0;
   286→  return calcTotalIngredientCost(ingredients) / baseServings;
   287→}

   293→function calcCostBreakdown(ingredients: RecipeIngredient[]): {
   294→  entries: CategoryBreakdownEntry[];
   295→  total: number;
   296→} {
   297→  const byCategory: Record<string, number> = {};
   298→  let total = 0;
   299→  ingredients.forEach((ri) => {
   300→    const cost = ri.quantity * ri.ingredient.avgCost;
   301→    const cat = ri.ingredient.category || "Other";
   302→    byCategory[cat] = (byCategory[cat] || 0) + cost;
   303→    total += cost;
   304→  });
   305→  const entries = Object.entries(byCategory)
   306→    .map(([category, amount]) => ({
   307→      category,
   308→      amount,
   309→      percent: total > 0 ? (amount / total) * 100 : 0,
   310→    }))
   311→    .sort((a, b) => b.amount - a.amount);
   312→  return { entries, total };
   313→}

   315→// Compare cost using lastPurchasePrice vs avgCost to show trend
   316→function calcCostTrend(ingredients: RecipeIngredient[]): {
   317→  direction: "up" | "down" | "stable";
   318→  percentChange: number;
   319→} {
   320→  let costByAvg = 0;
   321→  let costByLast = 0;
   322→  ingredients.forEach((ri) => {
   323→    costByAvg += ri.quantity * ri.ingredient.avgCost;
   324→    costByLast += ri.quantity * ri.ingredient.lastPurchasePrice;
   325→  });
   326→  if (costByAvg === 0) return { direction: "stable", percentChange: 0 };
   327→  const pctChange = ((costByLast - costByAvg) / costByAvg) * 100;
   328→  if (Math.abs(pctChange) < 2) return { direction: "stable", percentChange: pctChange };
   329→  return { direction: pctChange > 0 ? "up" : "down", percentChange: pctChange };
   330→}

   332→// Format ISO date as DD/MM for compact chart axis labels.
   333→function formatDateShort(iso: string): string {
   334→  try {
   335→    const d = new Date(iso);
   336→    const dd = String(d.getDate()).padStart(2, "0");
   337→    const mm = String(d.getMonth() + 1).padStart(2, "0");
   338→    return `${dd}/${mm}`;
   339→  } catch {
   340→    return "—";
   341→  }
   342→}
   343→
   344→// Format ISO date as "DD Mon YYYY, HH:MM" for table rows.
   345→function formatDateTime(iso: string): string {
   346→  try {
   347→    const d = new Date(iso);
   348→    return d.toLocaleString("en-IN", {
   349→      day: "2-digit",
   350→      month: "short",
   351→      year: "numeric",
   352→      hour: "2-digit",
   353→      minute: "2-digit",
   354→      hour12: true,
   355→    });
   356→  } catch {
   357→    return iso;
   358→  }
   359→}

   446→// ─── Latest Cost Variance Badge (used in recipe list/grid) ──────
   447→
   448→/**
   449→ * Compact badge shown next to recipe cards/rows when there's a meaningful
   450→ * cost variance in the latest RecipeCostHistory snapshot.
   451→ * - ⬆ red badge if variance > +5%
   452→ * - ⬇ green badge if variance < -5%
   453→ * - nothing rendered if within ±5% or no history
   454→ */
   455→function LatestCostVarianceBadge({
   456→  variance,
   457→}: {
   458→  variance: LatestCostVariance | null | undefined;
   459→}) {
   460→  if (!variance) return null;
   461→  if (variance.direction === "none") return null;
   462→  if (Math.abs(variance.percentage) < 5) return null;
   463→
   464→  const isUp = variance.percentage > 0;
   465→  const tooltipText =
   466→    variance.previous != null
   467→      ? `Cost changed from ${formatRupee(variance.previous)} to ${formatRupee(
   468→          variance.current
   469→        )} on ${formatDateShort(variance.recordedAt)}`
   470→      : `Cost recorded at ${formatRupee(variance.current)} on ${formatDateShort(
   471→          variance.recordedAt
   472→        )}`;
   473→
   474→  return (
   475→    <Tooltip>
   476→      <TooltipTrigger asChild>
   477→        <span
   478→          className={cn(
   479→            "inline-flex items-center gap-0.5 rounded-full px-1.5 py-0.5 text-[10px] font-semibold",
   480→            isUp
   481→              ? "bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-300"
   482→              : "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300"
   483→          )}
   484→        >
   485→          {isUp ? (
   486→            <TrendingUp className="h-3 w-3" />
   487→          ) : (
   488→            <TrendingDown className="h-3 w-3" />
   489→          )}
   490→          {Math.abs(variance.percentage).toFixed(0)}%
   491→        </span>
   492→      </TooltipTrigger>
   493→      <TooltipContent side="top" className="text-xs">
   494→        {tooltipText}
   495→      </TooltipContent>
   496→    </Tooltip>
   497→  );
   498→}

   511→function CostHistorySection({
   512→  recipeId,
   513→  liveCostPerServing,
   514→  costHistory,
   515→  loading,
   516→  recording,
   517→  onRecord,
   518→}: CostHistorySectionProps) {
   519→  // Use the live computed value as the "current cost" so it always reflects
   520→  // the latest ingredient prices — even before a snapshot is recorded.
   521→  const currentCostPerServing = liveCostPerServing;
   522→  const currentSnapshot = costHistory?.current ?? null;
   523→  const previousSnapshot = costHistory?.previous ?? null;
   524→  const variance = costHistory?.variance ?? {
   525→    absolute: 0,
   526→    percentage: 0,
   527→    direction: "none" as const,
   528→  };
   529→
   530→  // Show variance badge relative to the most recent snapshot (since the live
   531→  // number may differ slightly from the last snapshot). If no snapshots yet,
   532→  // there's no variance to show.
   533→  const showVarianceBadge =
   534→    !!currentSnapshot &&
   535→    !!previousSnapshot &&
   536→    variance.direction !== "none";
   537→
   538→  const varianceBadgeClasses =
   539→    variance.direction === "up"
   540→      ? "bg-red-100 text-red-700 dark:bg-red-900/40 dark:text-red-300"
   541→      : variance.direction === "down"
   542→      ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/40 dark:text-emerald-300"
   543→      : "bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-300";
   544→
   545→  // Alert banner trigger: >10% increase vs the previous snapshot.
   546→  const showAlertBanner =
   547→    !!currentSnapshot &&
   548→    !!previousSnapshot &&
   549→    variance.direction === "up" &&
   550→    variance.percentage > 10;
   551→
   552→  const trendData = (costHistory?.trend ?? []).map((t) => ({
   553→    date: formatDateShort(t.date),
   554→    cost: t.cost,
   555→    rawDate: t.date,
   556→  }));
   557→
   558→  return (
   559→    <div className="space-y-3">
   560→      {/* Alert banner — slides in when variance > 10% increase */}
   561→      <AnimatePresence>
   562→        {showAlertBanner && previousSnapshot && currentSnapshot && (
   563→          <motion.div
   564→            key="cost-alert-banner"
   565→            initial={{ opacity: 0, y: -8, height: 0 }}
   566→            animate={{ opacity: 1, y: 0, height: "auto" }}
   567→            exit={{ opacity: 0, y: -8, height: 0 }}
   568→            transition={{ duration: 0.25, ease: "easeOut" }}
   569→            className="overflow-hidden"
   570→          >
   571→            <div className="flex items-start gap-2 rounded-lg border border-amber-300 bg-amber-50 px-3 py-2 text-sm dark:border-amber-800 dark:bg-amber-950/30">
   572→              <AlertTriangle className="mt-0.5 h-4 w-4 shrink-0 text-amber-600 dark:text-amber-400" />
   573→              <div className="flex-1">
   574→                <p className="font-medium text-amber-900 dark:text-amber-200">
   575→                  ⚠ Cost increased by {Math.abs(variance.percentage).toFixed(1)}%
   576→                  since last snapshot
   577→                </p>
   578→                <p className="text-xs text-amber-700 dark:text-amber-300">
   579→                  {formatRupee(previousSnapshot.costPerServing)} →{" "}
   580→                  {formatRupee(currentSnapshot.costPerServing)} per serving.
   581→                  Review ingredient prices.
   582→                </p>
   583→              </div>
   584→            </div>
   585→          </motion.div>
   586→        )}
   587→      </AnimatePresence>
   588→
   589→      <div className="rounded-lg border p-4 space-y-4">
   590→        {/* Header */}
   591→        <div className="flex flex-wrap items-center justify-between gap-2">
   592→          <p className="text-sm font-medium flex items-center gap-2">
   593→            <History className="h-4 w-4 text-orange-600 dark:text-orange-400" />
   594→            Cost History
   595→          </p>
   596→          <Button
   597→            size="sm"
   598→            variant="outline"
   599→            className="h-8 gap-1.5 text-xs"
   600→            disabled={recording || loading}
   601→            onClick={onRecord}
   602→          >
   603→            {recording ? (
   604→              <Loader2 className="h-3.5 w-3.5 animate-spin" />
   605→            ) : (
   606→              <Camera className="h-3.5 w-3.5" />
   607→            )}
   608→            Record Snapshot
   609→          </Button>
   610→        </div>
