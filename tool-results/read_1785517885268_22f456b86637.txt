   252→
   253→// Calculate estimated cost for a meal based on its recipe
   254→const estimateMealCost = (meal: DailyMeal): number => {
   255→  if (!meal.recipe?.ingredients || meal.recipe.ingredients.length === 0) return 0;
   256→  const recipe = meal.recipe;
   257→  const ratio = meal.mealsServed / (recipe.baseServings || 1);
   258→  return recipe.ingredients.reduce(
   259→    (sum, ri) => sum + ratio * ri.quantity * (ri.ingredient.avgCost || 0),
   260→    0
   261→  );
   262→};
