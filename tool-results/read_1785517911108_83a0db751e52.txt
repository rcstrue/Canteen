   219→function formatINR(value: number): string {
   220→  const formatted = new Intl.NumberFormat("en-IN", {
   221→    minimumFractionDigits: 2,
   222→    maximumFractionDigits: 2,
   223→  }).format(value || 0);
   224→  return `₹${formatted}`;
   225→}
   226→
   227→function formatINRShort(value: number): string {
   228→  if (value >= 100000) return `₹${(value / 100000).toFixed(2)}L`;
   229→  if (value >= 1000) return `₹${(value / 1000).toFixed(1)}k`;
   230→  return formatINR(value);
   231→}

   253→function formatBytes(bytes: number): string {
   254→  if (bytes === 0) return "0 B";
   255→  const units = ["B", "KB", "MB", "GB"];
   256→  const i = Math.floor(Math.log(bytes) / Math.log(1024));
   257→  const v = bytes / Math.pow(1024, i);
   258→  return `${v.toFixed(v < 10 && i > 0 ? 1 : 0)} ${units[i]}`;
   259→}

   261→function formatRelativeDate(iso: string | null): string {
   262→  if (!iso) return "Never";
   263→  try {
   264→    const date = new Date(iso);
   265→    const now = new Date();
   266→    const diffMs = now.getTime() - date.getTime();
   267→    const diffDays = Math.floor(diffMs / (1000 * 60 * 60 * 24));
   268→    if (diffDays === 0) {
   269→      const diffHours = Math.floor(diffMs / (1000 * 60 * 60));
   270→      if (diffHours === 0) {
   271→        const diffMins = Math.floor(diffMs / (1000 * 60));
   272→        return diffMins <= 1 ? "Just now" : `${diffMins} minutes ago`;
   273→      }
   274→      return `${diffHours} hour${diffHours === 1 ? "" : "s"} ago`;
   275→    }
   276→    if (diffDays === 1) return "Yesterday";
   277→    if (diffDays < 7) return `${diffDays} days ago`;
   278→    return date.toLocaleDateString("en-IN", {
   279→      year: "numeric",
   280→      month: "short",
   281→      day: "numeric",
   282→    });
   283→  } catch {
   284→    return "Unknown";
   285→  }
   286→}
   287→
   288→function isStaleBackup(iso: string | null, days = 7): boolean {
   289→  if (!iso) return true;
   290→  try {
   291→    const date = new Date(iso);
   292→    const diffMs = Date.now() - date.getTime();
   293→    return diffMs > days * 24 * 60 * 60 * 1000;
   294→  } catch {
   295→    return true;
   296→  }
   297→}

   299→function validateBackupFile(parsed: unknown): {
   300→  valid: boolean;
   301→  error?: string;
   302→  file?: BackupFile;
   303→} {
   304→  if (typeof parsed !== "object" || parsed === null || Array.isArray(parsed)) {
   305→    return { valid: false, error: "Invalid format: expected a JSON object." };
   306→  }
   307→  const obj = parsed as Record<string, unknown>;
   308→  const metadata = obj.metadata as Record<string, unknown> | undefined;
   309→  const data = obj.data as Record<string, unknown> | undefined;
   310→  if (!metadata || typeof metadata !== "object") {
   311→    return { valid: false, error: 'Missing "metadata" object.' };
   312→  }
   313→  if (!data || typeof data !== "object") {
   314→    return { valid: false, error: 'Missing "data" object.' };
   315→  }
   316→  // At least one collection must be an array
   317→  const collections = ["ingredients", "recipes", "stockMovements", "dailyMeals", "purchases", "expenses", "suppliers", "users"];
   318→  const hasAny = collections.some((k) => Array.isArray((data as Record<string, unknown>)[k]));
   319→  if (!hasAny) {
   320→    return { valid: false, error: "Backup contains no recognizable data collections." };
   321→  }
   322→  return {
   323→    valid: true,
   324→    file: {
   325→      metadata: metadata as unknown as BackupMetadata,
   326→      data,
   327→    },
   328→  };
   329→}

   331→// ─── Circular Budget Gauge ───────────────────────────────────────────────────
   332→
   333→function BudgetGauge({ percent, label, spent, budget }: {
   334→  percent: number;
   335→  label: string;
   336→  spent: number;
   337→  budget: number;
   338→}) {
   339→  const size = 140;
   340→  const stroke = 12;
   341→  const radius = (size - stroke) / 2;
   342→  const circumference = 2 * Math.PI * radius;
   343→  const clamped = Math.max(0, Math.min(percent, 150)); // allow up to 150% for visual
   344→  const dashOffset = circumference - (Math.min(clamped, 100) / 100) * circumference;
   345→  const color = getBudgetColor(percent);
   346→
   347→  return (
   348→    <div className="flex flex-col items-center gap-2">
   349→      <div
   350→        className="relative"
   351→        style={{ width: size, height: size }}
   352→        role="meter"
   353→        aria-valuenow={Math.round(percent)}
   354→        aria-valuemin={0}
   355→        aria-valuemax={100}
   356→        aria-label={`${label} budget utilization`}
   357→      >
   358→        <svg width={size} height={size} className="-rotate-90">
   359→          <circle
   360→            cx={size / 2}
   361→            cy={size / 2}
   362→            r={radius}
   363→            fill="none"
   364→            stroke="var(--muted)"
   365→            strokeWidth={stroke}
   366→          />
   367→          <circle
   368→            cx={size / 2}
   369→            cy={size / 2}
   370→            r={radius}
   371→            fill="none"
   372→            stroke={color}
   373→            strokeWidth={stroke}
   374→            strokeLinecap="round"
   375→            strokeDasharray={circumference}
   376→            strokeDashoffset={dashOffset}
   377→            style={{
   378→              transition: "stroke-dashoffset 0.8s ease, stroke 0.4s ease",
   379→            }}
   380→          />
   381→        </svg>
   382→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   383→          <span
   384→            className="text-2xl font-bold tabular-nums"
   385→            style={{ color }}
   386→          >
   387→            {Math.round(percent)}%
   388→          </span>
   389→          <span className="text-[10px] font-medium text-muted-foreground">
   390→            Used
   391→          </span>
   392→        </div>
   393→      </div>
   394→      <div className="text-center">
   395→        <p className="text-sm font-semibold">{label}</p>
   396→        <p className="text-xs text-muted-foreground tabular-nums">
   397→          {formatINRShort(spent)} / {formatINRShort(budget)}
   398→        </p>
   399→      </div>
   400→    </div>
   401→  );
   402→}
