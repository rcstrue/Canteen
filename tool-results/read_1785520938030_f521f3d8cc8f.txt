   256→function formatStockWithUnit(quantity: number, unit: string): string {
   257→  return `${quantity} ${unit}`;
   258→}
   259→
   260→function isLowStock(item: Ingredient): boolean {
   261→  return item.currentStock < item.minStock;
   262→}
   263→
   264→type StockHealth = "OK" | "LOW" | "CRITICAL";
   265→
   266→function getStockHealth(item: Ingredient): StockHealth {
   267→  if (item.minStock === 0) return "OK";
   268→  const ratio = item.currentStock / item.minStock;
   269→  if (ratio < 1) return "CRITICAL";      // currentStock < minStock → red
   270→  if (ratio < 1.5) return "LOW";         // between minStock and 1.5× minStock → amber
   271→  return "OK";                            // currentStock > 1.5× minStock → green
   272→}
   273→

   309→function formatDateDDMMYYYY(dateStr: string): string {
   310→  if (!dateStr) return "—";
   311→  const d = new Date(dateStr);
   312→  if (isNaN(d.getTime())) return "—";
   313→  const dd = String(d.getDate()).padStart(2, "0");
   314→  const mm = String(d.getMonth() + 1).padStart(2, "0");
   315→  const yyyy = d.getFullYear();
   316→  return `${dd}/${mm}/${yyyy}`;
   317→}
